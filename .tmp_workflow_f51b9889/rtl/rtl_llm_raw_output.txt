---BEGIN adaptive_aero_control_register_map.v---
module adaptive_aero_control_register_map (
    clk,
    reset_n,
    cfg_valid,
    cfg_write,
    cfg_read,
    cfg_addr,
    cfg_wdata,
    cfg_ready,
    cfg_rdata,
    cfg_rvalid,
    enable,
    clear_faults,
    operating_mode,
    stream_velocity_setpoint,
    timeout_cycles,
    stale_age_limit,
    actuator_min,
    actuator_max,
    fallback_command,
    last_txn_id,
    last_rxn_id,
    sticky_fault_bits,
    request_pending,
    watchdog_expired,
    fault_status,
    response_age,
    clamp_active,
    fallback_active
);
    input clk;
    input reset_n;
    input cfg_valid;
    input cfg_write;
    input cfg_read;
    input [7:0] cfg_addr;
    input [31:0] cfg_wdata;
    output cfg_ready;
    output [31:0] cfg_rdata;
    output cfg_rvalid;
    output enable;
    output clear_faults;
    output [1:0] operating_mode;
    output [15:0] stream_velocity_setpoint;
    output [15:0] timeout_cycles;
    output [15:0] stale_age_limit;
    output [15:0] actuator_min;
    output [15:0] actuator_max;
    output [15:0] fallback_command;
    output [15:0] last_txn_id;
    output [15:0] last_rxn_id;
    output [15:0] sticky_fault_bits;
    input request_pending;
    input watchdog_expired;
    input fault_status;
    input [15:0] response_age;
    input clamp_active;
    input fallback_active;

    reg enable_r;
    reg clear_faults_r;
    reg [1:0] operating_mode_r;
    reg [15:0] stream_velocity_setpoint_r;
    reg [15:0] timeout_cycles_r;
    reg [15:0] stale_age_limit_r;
    reg [15:0] actuator_min_r;
    reg [15:0] actuator_max_r;
    reg [15:0] fallback_command_r;
    reg [15:0] last_txn_id_r;
    reg [15:0] last_rxn_id_r;
    reg [15:0] sticky_fault_bits_r;
    reg [31:0] cfg_rdata_r;
    reg cfg_rvalid_r;
    reg cfg_ready_r;

    assign cfg_ready = cfg_ready_r;
    assign cfg_rdata = cfg_rdata_r;
    assign cfg_rvalid = cfg_rvalid_r;
    assign enable = enable_r;
    assign clear_faults = clear_faults_r;
    assign operating_mode = operating_mode_r;
    assign stream_velocity_setpoint = stream_velocity_setpoint_r;
    assign timeout_cycles = timeout_cycles_r;
    assign stale_age_limit = stale_age_limit_r;
    assign actuator_min = actuator_min_r;
    assign actuator_max = actuator_max_r;
    assign fallback_command = fallback_command_r;
    assign last_txn_id = last_txn_id_r;
    assign last_rxn_id = last_rxn_id_r;
    assign sticky_fault_bits = sticky_fault_bits_r;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            enable_r <= 1'b0;
            clear_faults_r <= 1'b0;
            operating_mode_r <= 2'b00;
            stream_velocity_setpoint_r <= 16'h0000;
            timeout_cycles_r <= 16'd1024;
            stale_age_limit_r <= 16'd32;
            actuator_min_r <= 16'h0000;
            actuator_max_r <= 16'd4095;
            fallback_command_r <= 16'h0000;
            last_txn_id_r <= 16'h0000;
            last_rxn_id_r <= 16'h0000;
            sticky_fault_bits_r <= 16'h0001;
            cfg_rdata_r <= 32'h00000000;
            cfg_rvalid_r <= 1'b0;
            cfg_ready_r <= 1'b1;
        end else begin
            clear_faults_r <= 1'b0;
            cfg_rvalid_r <= cfg_valid & cfg_read;
            cfg_ready_r <= 1'b1;

            if (cfg_valid && cfg_write) begin
                case (cfg_addr)
                    8'h00: begin
                        enable_r <= cfg_wdata[0];
                        clear_faults_r <= cfg_wdata[1];
                        operating_mode_r <= cfg_wdata[3:2];
                    end
                    8'h04: stream_velocity_setpoint_r <= cfg_wdata[15:0];
                    8'h08: timeout_cycles_r <= cfg_wdata[15:0];
                    8'h0C: stale_age_limit_r <= cfg_wdata[15:0];
                    8'h10: actuator_min_r <= cfg_wdata[15:0];
                    8'h14: actuator_max_r <= cfg_wdata[15:0];
                    8'h18: fallback_command_r <= cfg_wdata[15:0];
                    default: begin
                    end
                endcase
            end

            if (request_pending)
                last_txn_id_r <= last_txn_id_r;
            if (watchdog_expired)
                sticky_fault_bits_r[3] <= 1'b1;
            if (fault_status)
                sticky_fault_bits_r[0] <= 1'b1;
            if (clamp_active)
                sticky_fault_bits_r[4] <= 1'b1;
            if (fallback_active)
                sticky_fault_bits_r[5] <= 1'b1;

            if (cfg_valid && cfg_write && cfg_addr == 8'h00 && cfg_wdata[1] && !fault_status && !watchdog_expired) begin
                sticky_fault_bits_r <= 16'h0000;
            end else if (cfg_valid && cfg_write && cfg_addr == 8'h00 && cfg_wdata[1]) begin
                sticky_fault_bits_r[0] <= sticky_fault_bits_r[0];
                sticky_fault_bits_r[3] <= sticky_fault_bits_r[3];
            end

            if (cfg_valid && cfg_write && cfg_addr == 8'h1C)
                last_txn_id_r <= last_txn_id_r;
            if (cfg_valid && cfg_write && cfg_addr == 8'h20)
                last_rxn_id_r <= last_rxn_id_r;
            if (cfg_valid && cfg_write && cfg_addr == 8'h24)
                sticky_fault_bits_r <= sticky_fault_bits_r;
            if (cfg_valid && cfg_write && cfg_addr == 8'h28)
                last_txn_id_r <= last_txn_id_r;
            if (cfg_valid && cfg_write && cfg_addr == 8'h2C)
                last_rxn_id_r <= last_rxn_id_r;
            if (cfg_valid && cfg_write && cfg_addr == 8'h30)
                response_age <= response_age;
        end
    end

    always @(*) begin
        cfg_rdata_r = 32'h00000000;
        case (cfg_addr)
            8'h00: cfg_rdata_r = {28'h0000000, operating_mode_r, clear_faults_r, enable_r};
            8'h04: cfg_rdata_r = {16'h0000, stream_velocity_setpoint_r};
            8'h08: cfg_rdata_r = {16'h0000, timeout_cycles_r};
            8'h0C: cfg_rdata_r = {16'h0000, stale_age_limit_r};
            8'h10: cfg_rdata_r = {16'h0000, actuator_min_r};
            8'h14: cfg_rdata_r = {16'h0000, actuator_max_r};
            8'h18: cfg_rdata_r = {16'h0000, fallback_command_r};
            8'h1C: cfg_rdata_r = {16'h0000, last_txn_id_r};
            8'h20: cfg_rdata_r = {16'h0000, last_rxn_id_r};
            8'h24: cfg_rdata_r = {15'h0000, fault_status, sticky_fault_bits_r[1], request_pending, watchdog_expired, clamp_active, fallback_active, 1'b0};
            8'h28: cfg_rdata_r = {16'h0000, last_txn_id_r};
            8'h2C: cfg_rdata_r = {16'h0000, last_rxn_id_r};
            8'h30: cfg_rdata_r = {16'h0000, response_age};
            8'h34: cfg_rdata_r = {27'h0000000, clamp_active, fallback_active, 1'b0, request_pending, watchdog_expired, 1'b0};
            8'h38: cfg_rdata_r = {actuator_max_r, actuator_min_r};
            8'h3C: cfg_rdata_r = {stale_age_limit_r, timeout_cycles_r};
            default: cfg_rdata_r = 32'h00000000;
        endcase
    end
endmodule
---END adaptive_aero_control_register_map.v---

---BEGIN adaptive_aero_control_transport.v---
module adaptive_aero_control_transport (
    clk,
    reset_n,
    enable,
    operating_mode,
    stream_velocity_setpoint,
    timeout_cycles,
    stale_age_limit,
    fallback_command,
    req_valid,
    req_ready,
    req_data,
    rsp_valid,
    rsp_ready,
    rsp_data,
    request_pending,
    watchdog_expired,
    response_age,
    last_txn_id,
    last_rxn_id,
    candidate_cmd,
    candidate_valid,
    fallback_active,
    parser_error,
    fault_status,
    explicit_fault_in,
    clear_faults
);
    input clk;
    input reset_n;
    input enable;
    input [1:0] operating_mode;
    input [15:0] stream_velocity_setpoint;
    input [15:0] timeout_cycles;
    input [15:0] stale_age_limit;
    input [15:0] fallback_command;
    output req_valid;
    input req_ready;
    output [63:0] req_data;
    input rsp_valid;
    output rsp_ready;
    input [63:0] rsp_data;
    output request_pending;
    output watchdog_expired;
    output [15:0] response_age;
    output [15:0] last_txn_id;
    output [15:0] last_rxn_id;
    output [15:0] candidate_cmd;
    output candidate_valid;
    output fallback_active;
    output parser_error;
    output fault_status;
    input explicit_fault_in;
    input clear_faults;

    reg req_valid_r;
    reg [63:0] req_data_r;
    reg rsp_ready_r;
    reg request_pending_r;
    reg watchdog_expired_r;
    reg [15:0] response_age_r;
    reg [15:0] last_txn_id_r;
    reg [15:0] last_rxn_id_r;
    reg [15:0] candidate_cmd_r;
    reg candidate_valid_r;
    reg fallback_active_r;
    reg parser_error_r;
    reg fault_status_r;
    reg [15:0] txn_counter_r;
    reg [15:0] watchdog_counter_r;

    assign req_valid = req_valid_r;
    assign req_data = req_data_r;
    assign rsp_ready = rsp_ready_r;
    assign request_pending = request_pending_r;
    assign watchdog_expired = watchdog_expired_r;
    assign response_age = response_age_r;
    assign last_txn_id = last_txn_id_r;
    assign last_rxn_id = last_rxn_id_r;
    assign candidate_cmd = candidate_cmd_r;
    assign candidate_valid = candidate_valid_r;
    assign fallback_active = fallback_active_r;
    assign parser_error = parser_error_r;
    assign fault_status = fault_status_r;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            req_valid_r <= 1'b0;
            req_data_r <= 64'h0000000000000000;
            rsp_ready_r <= 1'b1;
            request_pending_r <= 1'b0;
            watchdog_expired_r <= 1'b0;
            response_age_r <= 16'h0000;
            last_txn_id_r <= 16'h0000;
            last_rxn_id_r <= 16'h0000;
            candidate_cmd_r <= 16'h0000;
            candidate_valid_r <= 1'b0;
            fallback_active_r <= 1'b1;
            parser_error_r <= 1'b0;
            fault_status_r <= 1'b1;
            txn_counter_r <= 16'h0001;
            watchdog_counter_r <= 16'h0000;
        end else begin
            rsp_ready_r <= 1'b1;
            req_valid_r <= enable & !request_pending_r;
            req_data_r <= {16'hA5A5, operating_mode, stream_velocity_setpoint, txn_counter_r, fallback_command, 8'h5A, 8'hC3};
            fallback_active_r <= 1'b0;
            parser_error_r <= 1'b0;

            if (clear_faults && !explicit_fault_in && !watchdog_expired_r && !parser_error_r)
                fault_status_r <= 1'b0;
            if (explicit_fault_in)
                fault_status_r <= 1'b1;

            if (enable && !request_pending_r && !fault_status_r) begin
                request_pending_r <= 1'b1;
                last_txn_id_r <= txn_counter_r;
                txn_counter_r <= txn_counter_r + 16'h0001;
                watchdog_counter_r <= 16'h0000;
                response_age_r <= 16'h0000;
            end else if (request_pending_r) begin
                if (watchdog_counter_r < timeout_cycles) begin
                    watchdog_counter_r <= watchdog_counter_r + 16'h0001;
                    response_age_r <= response_age_r + 16'h0001;
                end else begin
                    watchdog_expired_r <= 1'b1;
                    fault_status_r <= 1'b1;
                    fallback_active_r <= 1'b1;
                    candidate_valid_r <= 1'b0;
                    request_pending_r <= 1'b0;
                end
            end

            if (rsp_valid) begin
                last_rxn_id_r <= rsp_data[31:16];
                response_age_r <= rsp_data[15:0];
                if ((rsp_data[63:48] == last_txn_id_r) &&
                    (rsp_data[47:32] == 16'h0001) &&
                    (operating_mode <= 2'b10) &&
                    (rsp_data[15:0] <= stale_age_limit) &&
                    !explicit_fault_in) begin
                    candidate_cmd_r <= rsp_data[31:16];
                    candidate_valid_r <= 1'b1;
                    request_pending_r <= 1'b0;
                    watchdog_expired_r <= 1'b0;
                    fallback_active_r <= 1'b0;
                    parser_error_r <= 1'b0;
                end else begin
                    candidate_valid_r <= 1'b0;
                    parser_error_r <= 1'b1;
                    fault_status_r <= 1'b1;
                    fallback_active_r <= 1'b1;
                end
            end else if (request_pending_r && watchdog_counter_r >= timeout_cycles) begin
                watchdog_expired_r <= 1'b1;
                fault_status_r <= 1'b1;
                fallback_active_r <= 1'b1;
            end

            if (!enable) begin
                req_valid_r <= 1'b0;
                request_pending_r <= 1'b0;
                candidate_valid_r <= 1'b0;
                fallback_active_r <= 1'b1;
            end

            if (clear_faults && !explicit_fault_in && !watchdog_expired_r && !parser_error_r && !request_pending_r)
                fault_status_r <= 1'b0;
        end
    end
endmodule
---END adaptive_aero_control_transport.v---

---BEGIN adaptive_aero_control_actuator.v---
module adaptive_aero_control_actuator (
    clk,
    reset_n,
    enable,
    actuator_min,
    actuator_max,
    fallback_command,
    candidate_cmd,
    candidate_valid,
    fault_status,
    fallback_active,
    clamp_active,
    act_cmd,
    act_enable,
    fallback_active_out
);
    input clk;
    input reset_n;
    input enable;
    input [15:0] actuator_min;
    input [15:0] actuator_max;
    input [15:0] fallback_command;
    input [15:0] candidate_cmd;
    input candidate_valid;
    input fault_status;
    input fallback_active;
    output clamp_active;
    output [15:0] act_cmd;
    output act_enable;
    output fallback_active_out;

    reg clamp_active_r;
    reg [15:0] act_cmd_r;
    reg act_enable_r;
    reg fallback_active_out_r;

    assign clamp_active = clamp_active_r;
    assign act_cmd = act_cmd_r;
    assign act_enable = act_enable_r;
    assign fallback_active_out = fallback_active_out_r;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            clamp_active_r <= 1'b0;
            act_cmd_r <= fallback_command;
            act_enable_r <= 1'b0;
            fallback_active_out_r <= 1'b1;
        end else begin
            if (!enable || fault_status || !candidate_valid || fallback_active) begin
                act_cmd_r <= fallback_command;
                act_enable_r <= 1'b0;
                fallback_active_out_r <= 1'b1;
                clamp_active_r <= 1'b0;
            end else begin
                if (candidate_cmd < actuator_min) begin
                    act_cmd_r <= actuator_min;
                    clamp_active_r <= 1'b1;
                end else if (candidate_cmd > actuator_max) begin
                    act_cmd_r <= actuator_max;
                    clamp_active_r <= 1'b1;
                end else begin
                    act_cmd_r <= candidate_cmd;
                    clamp_active_r <= 1'b0;
                end
                act_enable_r <= 1'b1;
                fallback_active_out_r <= 1'b0;
            end
        end
    end
endmodule
---END adaptive_aero_control_actuator.v---
