module csr_block(
    clk,
    reset_n,
    host_valid,
    host_ready,
    host_addr,
    host_wdata,
    host_rdata,
    host_we,
    host_re,
    cfg_launch,
    cfg_timeout_cycles,
    cfg_freshness_cycles,
    cfg_flow_velocity,
    cfg_geom_length,
    cfg_job_id,
    cfg_sequence_id,
    cfg_fault_clear,
    cfg_irq_enable,
    cfg_enable_latched_fault,
    cfg_act_min,
    cfg_act_max,
    cfg_slew_limit,
    cfg_geom_format_id,
    cfg_model_id,
    status_busy,
    status_fault,
    status_timeout,
    status_valid_response,
    status_fallback_active
);
input clk;
input reset_n;
input host_valid;
output host_ready;
input [7:0] host_addr;
input [63:0] host_wdata;
output [63:0] host_rdata;
input host_we;
input host_re;
output cfg_launch;
output [23:0] cfg_timeout_cycles;
output [23:0] cfg_freshness_cycles;
output [15:0] cfg_flow_velocity;
output [15:0] cfg_geom_length;
output [15:0] cfg_job_id;
output [15:0] cfg_sequence_id;
output cfg_fault_clear;
output cfg_irq_enable;
output cfg_enable_latched_fault;
output [15:0] cfg_act_min;
output [15:0] cfg_act_max;
output [15:0] cfg_slew_limit;
output [7:0] cfg_geom_format_id;
output [7:0] cfg_model_id;
input status_busy;
input status_fault;
input status_timeout;
input status_valid_response;
input status_fallback_active;
reg host_ready;
reg [63:0] host_rdata;
reg cfg_launch;
reg [23:0] cfg_timeout_cycles;
reg [23:0] cfg_freshness_cycles;
reg [15:0] cfg_flow_velocity;
reg [15:0] cfg_geom_length;
reg [15:0] cfg_job_id;
reg [15:0] cfg_sequence_id;
reg cfg_fault_clear;
reg cfg_irq_enable;
reg cfg_enable_latched_fault;
reg [15:0] cfg_act_min;
reg [15:0] cfg_act_max;
reg [15:0] cfg_slew_limit;
reg [7:0] cfg_geom_format_id;
reg [7:0] cfg_model_id;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        host_ready <= 1'b0;
        host_rdata <= 64'h0;
        cfg_launch <= 1'b0;
        cfg_timeout_cycles <= 24'h0;
        cfg_freshness_cycles <= 24'h0;
        cfg_flow_velocity <= 16'h0;
        cfg_geom_length <= 16'h0;
        cfg_job_id <= 16'h0;
        cfg_sequence_id <= 16'h0;
        cfg_fault_clear <= 1'b0;
        cfg_irq_enable <= 1'b0;
        cfg_enable_latched_fault <= 1'b0;
        cfg_act_min <= 16'h0;
        cfg_act_max <= 16'h0;
        cfg_slew_limit <= 16'h0;
        cfg_geom_format_id <= 8'h0;
        cfg_model_id <= 8'h0;
    end else begin
        host_ready <= host_valid;
        cfg_launch <= 1'b0;
        cfg_fault_clear <= 1'b0;
        if (host_valid && host_we) begin
            case (host_addr[5:0])
                6'h00: begin
                    cfg_launch <= host_wdata[0];
                    cfg_fault_clear <= host_wdata[1];
                    cfg_irq_enable <= host_wdata[2];
                    cfg_enable_latched_fault <= host_wdata[3];
                end
                6'h08: cfg_timeout_cycles <= host_wdata[23:0];
                6'h10: cfg_freshness_cycles <= host_wdata[23:0];
                6'h18: begin
                    cfg_flow_velocity <= host_wdata[15:0];
                    cfg_geom_format_id <= host_wdata[23:16];
                    cfg_model_id <= host_wdata[31:24];
                end
                6'h20: cfg_geom_length <= host_wdata[15:0];
                6'h28: begin
                    cfg_job_id <= host_wdata[15:0];
                    cfg_sequence_id <= host_wdata[31:16];
                end
                6'h30: begin
                    cfg_act_min <= host_wdata[15:0];
                    cfg_act_max <= host_wdata[31:16];
                    cfg_slew_limit <= host_wdata[47:32];
                end
                default: begin
                end
            endcase
        end
        if (host_valid && host_re) begin
            case (host_addr[5:0])
                6'h00: host_rdata <= {60'h0, cfg_enable_latched_fault, cfg_irq_enable, cfg_fault_clear, 1'b0};
                6'h08: host_rdata <= {40'h0, cfg_timeout_cycles};
                6'h10: host_rdata <= {40'h0, cfg_freshness_cycles};
                6'h18: host_rdata <= {32'h0, cfg_model_id, cfg_geom_format_id, cfg_flow_velocity};
                6'h20: host_rdata <= {48'h0, cfg_geom_length};
                6'h28: host_rdata <= {32'h0, cfg_sequence_id, cfg_job_id};
                6'h30: host_rdata <= {16'h0, cfg_slew_limit, cfg_act_max, cfg_act_min};
                6'h38: host_rdata <= {59'h0, status_fallback_active, status_valid_response, status_timeout, status_fault, status_busy};
                default: host_rdata <= 64'h0;
            endcase
        end
    end
end
endmodule

module geometry_staging(
    clk,
    reset_n,
    geom_valid,
    geom_ready,
    geom_data,
    geom_keep,
    geom_last,
    cfg_geom_length,
    geom_capture_start,
    geom_capture_done,
    geom_capture_error,
    geom_stream_count,
    geom_crc_seed,
    geom_crc_value
);
input clk;
input reset_n;
input geom_valid;
output geom_ready;
input [63:0] geom_data;
input [7:0] geom_keep;
input geom_last;
input [15:0] cfg_geom_length;
input geom_capture_start;
output geom_capture_done;
output geom_capture_error;
output [15:0] geom_stream_count;
input [31:0] geom_crc_seed;
output [31:0] geom_crc_value;
reg geom_ready;
reg geom_capture_done;
reg geom_capture_error;
reg [15:0] geom_stream_count;
reg [31:0] geom_crc_value;
reg [31:0] crc_accum;
reg [15:0] accepted_count;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        geom_ready <= 1'b0;
        geom_capture_done <= 1'b0;
        geom_capture_error <= 1'b0;
        geom_stream_count <= 16'h0;
        geom_crc_value <= 32'h0;
        crc_accum <= 32'h0;
        accepted_count <= 16'h0;
    end else begin
        geom_capture_done <= 1'b0;
        geom_capture_error <= 1'b0;
        if (geom_capture_start) begin
            accepted_count <= 16'h0;
            crc_accum <= geom_crc_seed;
            geom_ready <= 1'b1;
        end
        if (geom_valid && geom_ready) begin
            accepted_count <= accepted_count + 16'd1;
            crc_accum <= crc_accum ^ geom_data[31:0] ^ {24'h0, geom_keep};
            if (geom_last || ((accepted_count + 16'd1) >= cfg_geom_length)) begin
                geom_capture_done <= 1'b1;
                geom_ready <= 1'b0;
                geom_stream_count <= accepted_count + 16'd1;
                geom_crc_value <= crc_accum ^ geom_data[31:0] ^ {24'h0, geom_keep};
            end
        end
        if (geom_valid && !geom_ready) begin
            geom_capture_error <= 1'b1;
        end
    end
end
endmodule

input [15:0] cfg_flow_velocity;
input [15:0] cfg_geom_length;

input request_ready;

reg request_valid;
reg [63:0] request_data;
reg [7:0] request_keep;
reg request_last;
reg request_accepted;
reg request_checksum_ok;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        request_valid <= 1'b0;
        request_data <= 64'h0;
        request_keep <= 8'h0;
        request_last <= 1'b0;
        request_accepted <= 1'b0;
        request_checksum_ok <= 1'b0;
    end else begin
        request_accepted <= 1'b0;
        if (request_fire) begin
            request_valid <= 1'b1;
            request_data <= {cfg_model_id, cfg_geom_format_id, cfg_flow_velocity, cfg_geom_length, cfg_job_id, cfg_sequence_id};
            request_keep <= 8'hff;
            request_last <= 1'b1;
            request_checksum_ok <= (geom_stream_count == cfg_geom_length) && (geom_crc_value != 32'h0);
        end
        if (request_valid && request_ready) begin
            request_valid <= 1'b0;
            request_accepted <= 1'b1;
        end
    end
end

        transport_error <= 1'b0;
        response_pending <= 1'b0;

    end else begin
        request_accepted <= 1'b0;
        worker_rsp_valid <= 1'b0;

module safety_logic(
    clk,
    reset_n,
    act_mode_raw,
    act_setpoint_raw,
    act_enable_raw,
    act_fallback_raw,
    act_summary_valid,
    cfg_act_min,
    cfg_act_max,
    cfg_slew_limit,
    cfg_enable_latched_fault,
    status_timeout,
    status_fault,
    status_transport_error,
    status_invalid_response,
    status_stale_response,
    host_abort,
    act_mode,
    act_setpoint,
    act_enable,
    act_fallback,
    fallback_active,
    fault_latched
);
input clk;
input reset_n;
input [2:0] act_mode_raw;
input [15:0] act_setpoint_raw;
input act_enable_raw;
input act_fallback_raw;
input act_summary_valid;
input [15:0] cfg_act_min;
input [15:0] cfg_act_max;
input [15:0] cfg_slew_limit;
input cfg_enable_latched_fault;
input status_timeout;
input status_fault;
input status_transport_error;
input status_invalid_response;
input status_stale_response;
input host_abort;
output [2:0] act_mode;
output [15:0] act_setpoint;
output act_enable;
output act_fallback;
output fallback_active;
output fault_latched;
reg [2:0] act_mode;
reg [15:0] act_setpoint;
reg act_enable;
reg act_fallback;
reg fallback_active;
reg fault_latched;
reg [15:0] prev_setpoint;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        act_mode <= 3'b000;
        act_setpoint <= 16'h0000;
        act_enable <= 1'b0;
        act_fallback <= 1'b1;
        fallback_active <= 1'b1;
        fault_latched <= 1'b0;
        prev_setpoint <= 16'h0000;
    end else begin
        if (host_abort || status_timeout || status_fault || status_transport_error || status_invalid_response || status_stale_response) begin
            fallback_active <= 1'b1;
            act_fallback <= 1'b1;
            act_enable <= 1'b0;
            if (cfg_enable_latched_fault) fault_latched <= 1'b1;
        end else if (act_summary_valid) begin
            fallback_active <= act_fallback_raw;
            act_fallback <= act_fallback_raw;
            act_mode <= act_mode_raw;
            if (act_setpoint_raw < cfg_act_min) act_setpoint <= cfg_act_min;
            else if (act_setpoint_raw > cfg_act_max) act_setpoint <= cfg_act_max;
            else if (cfg_slew_limit != 16'h0 && ((act_setpoint_raw > prev_setpoint) ? (act_setpoint_raw - prev_setpoint) : (prev_setpoint - act_setpoint_raw)) > cfg_slew_limit) begin
                if (act_setpoint_raw > prev_setpoint) act_setpoint <= prev_setpoint + cfg_slew_limit;
                else act_setpoint <= prev_setpoint - cfg_slew_limit;
            end else begin
                act_setpoint <= act_setpoint_raw;
            end
            act_enable <= act_enable_raw;
            prev_setpoint <= act_setpoint;
            if (!cfg_enable_latched_fault) fault_latched <= 1'b0;
        end
    end
end
endmodule
