module adaptive_aero_control_transport (
    clk,
    rst_n,
    cmd_in_valid,
    cmd_in_ready,
    cmd_in_data,
    cmd_out_valid,
    cmd_out_ready,
    cmd_out_data,
    cfg_enable,
    cfg_timeout_threshold,
    cfg_stale_threshold,
    request_seq_out,
    last_accepted_seq_out,
    response_candidate_cmd,
    response_seq,
    response_age,
    response_valid,
    response_duplicate,
    response_out_of_order,
    response_stale,
    request_issued_pulse,
    response_received_pulse,
    status_flags_in
);
    input clk;
    input rst_n;
    input cmd_in_valid;
    output cmd_in_ready;
    input [127:0] cmd_in_data;
    output cmd_out_valid;
    input cmd_out_ready;
    output [127:0] cmd_out_data;
    input cfg_enable;
    input [15:0] cfg_timeout_threshold;
    input [15:0] cfg_stale_threshold;
    output [15:0] request_seq_out;
    output [15:0] last_accepted_seq_out;
    output [15:0] response_candidate_cmd;
    output [15:0] response_seq;
    output [15:0] response_age;
    output response_valid;
    output response_duplicate;
    output response_out_of_order;
    output response_stale;
    output request_issued_pulse;
    output response_received_pulse;
    input [15:0] status_flags_in;

    reg cmd_in_ready_r;
    reg cmd_out_valid_r;
    reg [127:0] cmd_out_data_r;
    reg [15:0] request_seq_r;
    reg [15:0] last_accepted_seq_r;
    reg [15:0] response_candidate_cmd_r;
    reg [15:0] response_seq_r;
    reg [15:0] response_age_r;
    reg response_valid_r;
    reg response_duplicate_r;
    reg response_out_of_order_r;
    reg response_stale_r;
    reg request_issued_pulse_r;
    reg response_received_pulse_r;
    reg [15:0] watchdog_age_r;
    reg [15:0] safe_status_hold_r;

    reg [15:0] seq_in;
    reg [15:0] age_in;
    reg [15:0] cmd_in_field;
    reg [15:0] flags_in;
    reg [15:0] handle_in;
    reg packet_valid_in;
    reg resp_duplicate_next;
    reg resp_ooo_next;
    reg resp_stale_next;
    reg resp_accept_next;
    reg request_issue_next;
    reg response_receive_next;

    assign cmd_in_ready = cmd_in_ready_r;
    assign cmd_out_valid = cmd_out_valid_r;
    assign cmd_out_data = cmd_out_data_r;
    assign request_seq_out = request_seq_r;
    assign last_accepted_seq_out = last_accepted_seq_r;
    assign response_candidate_cmd = response_candidate_cmd_r;
    assign response_seq = response_seq_r;
    assign response_age = response_age_r;
    assign response_valid = response_valid_r;
    assign response_duplicate = response_duplicate_r;
    assign response_out_of_order = response_out_of_order_r;
    assign response_stale = response_stale_r;
    assign request_issued_pulse = request_issued_pulse_r;
    assign response_received_pulse = response_received_pulse_r;

    always @(*) begin
        cmd_in_ready_r = 1'b0;
        cmd_out_valid_r = 1'b0;
        cmd_out_data_r = 128'h0;
        seq_in = cmd_in_data[15:0];
        age_in = cmd_in_data[31:16];
        cmd_in_field = cmd_in_data[47:32];
        flags_in = cmd_in_data[63:48];
        handle_in = cmd_in_data[79:64];
        packet_valid_in = cmd_in_data[80];
        resp_duplicate_next = 1'b0;
        resp_ooo_next = 1'b0;
        resp_stale_next = 1'b0;
        resp_accept_next = 1'b0;
        request_issue_next = 1'b0;
        response_receive_next = 1'b0;
        if (cfg_enable) begin
            cmd_in_ready_r = 1'b1;
            cmd_out_valid_r = 1'b1;
            cmd_out_data_r = {16'hA5C3, request_seq_r, last_accepted_seq_r, status_flags_in, cfg_timeout_threshold, cfg_stale_threshold, 32'h0, 32'h0, 16'h0001};
            if (cmd_in_valid) begin
                response_receive_next = 1'b1;
                if (packet_valid_in) begin
                    resp_accept_next = 1'b1;
                    if (seq_in == last_accepted_seq_r) begin
                        resp_accept_next = 1'b0;
                        resp_duplicate_next = 1'b1;
                    end else begin
                        if (seq_in <= last_accepted_seq_r) begin
                            resp_accept_next = 1'b0;
                            resp_ooo_next = 1'b1;
                        end
                        if (age_in > cfg_stale_threshold) begin
                            resp_accept_next = 1'b0;
                            resp_stale_next = 1'b1;
                        end
                    end
                end
                request_issue_next = 1'b1;
            end
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            request_seq_r <= 16'h0000;
            last_accepted_seq_r <= 16'h0000;
            response_candidate_cmd_r <= 16'h0000;
            response_seq_r <= 16'h0000;
            response_age_r <= 16'h0000;
            response_valid_r <= 1'b0;
            response_duplicate_r <= 1'b0;
            response_out_of_order_r <= 1'b0;
            response_stale_r <= 1'b0;
            request_issued_pulse_r <= 1'b0;
            response_received_pulse_r <= 1'b0;
            watchdog_age_r <= 16'h0000;
            safe_status_hold_r <= 16'h0000;
        end else begin
            request_issued_pulse_r <= 1'b0;
            response_received_pulse_r <= 1'b0;
            response_valid_r <= 1'b0;
            response_duplicate_r <= 1'b0;
            response_out_of_order_r <= 1'b0;
            response_stale_r <= 1'b0;
            if (cfg_enable) begin
                if (cmd_in_valid && cmd_in_ready_r) begin
                    request_seq_r <= request_seq_r + 16'h0001;
                    request_issued_pulse_r <= 1'b1;
                    response_received_pulse_r <= response_receive_next;
                    watchdog_age_r <= 16'h0000;
                    if (packet_valid_in && resp_accept_next) begin
                        response_candidate_cmd_r <= cmd_in_field;
                        response_seq_r <= seq_in;
                        response_age_r <= age_in;
                        last_accepted_seq_r <= seq_in;
                        response_valid_r <= 1'b1;
                    end else begin
                        if (packet_valid_in && resp_duplicate_next) begin
                            response_duplicate_r <= 1'b1;
                        end
                        if (packet_valid_in && resp_ooo_next) begin
                            response_out_of_order_r <= 1'b1;
                        end
                        if (packet_valid_in && resp_stale_next) begin
                            response_stale_r <= 1'b1;
                        end
                    end
                end else begin
                    if (watchdog_age_r != cfg_timeout_threshold) begin
                        watchdog_age_r <= watchdog_age_r + 16'h0001;
                    end
                end
            end else begin
                watchdog_age_r <= 16'h0000;
            end
            safe_status_hold_r <= status_flags_in;
        end
    end
endmodule

module adaptive_aero_control_safety (
    clk,
    rst_n,
    cfg_enable,
    cfg_safe_cmd,
    cfg_cmd_min,
    cfg_cmd_max,
    cfg_rate_limit_en,
    cfg_rate_limit_step,
    cfg_timeout_threshold,
    cfg_stale_threshold,
    response_candidate_cmd,
    response_seq,
    response_age,
    response_valid,
    response_duplicate,
    response_out_of_order,
    response_stale,
    request_issued_pulse,
    response_received_pulse,
    status_flags_in,
    actuator_cmd,
    actuator_cmd_valid,
    status_flags,
    fault_counters,
    debug_counters
);
    input clk;
    input rst_n;
    input cfg_enable;
    input [15:0] cfg_safe_cmd;
    input [15:0] cfg_cmd_min;
    input [15:0] cfg_cmd_max;
    input cfg_rate_limit_en;
    input [15:0] cfg_rate_limit_step;
    input [15:0] cfg_timeout_threshold;
    input [15:0] cfg_stale_threshold;
    input [15:0] response_candidate_cmd;
    input [15:0] response_seq;
    input [15:0] response_age;
    input response_valid;
    input response_duplicate;
    input response_out_of_order;
    input response_stale;
    input request_issued_pulse;
    input response_received_pulse;
    input [15:0] status_flags_in;
    output [15:0] actuator_cmd;
    output actuator_cmd_valid;
    output [15:0] status_flags;
    output [31:0] fault_counters;
    output [31:0] debug_counters;

    reg [15:0] actuator_cmd_r;
    reg actuator_cmd_valid_r;
    reg [15:0] status_flags_r;
    reg [31:0] fault_counters_r;
    reg [31:0] debug_counters_r;
    reg [15:0] last_cmd_r;
    reg [15:0] safe_cmd_r;

    wire [15:0] clamp_min;
    wire [15:0] clamp_max;
    wire [15:0] unclamped_cmd;
    wire [15:0] clamped_cmd;
    wire [15:0] rate_step_abs;
    wire [15:0] rate_limited_hi;
    wire [15:0] rate_limited_lo;
    wire [15:0] rate_limited_cmd;
    wire clamp_hit;
    wire timeout_hit;
    wire fallback_hit;
    wire accept_hit;

    assign clamp_min = cfg_cmd_min;
    assign clamp_max = cfg_cmd_max;
    assign unclamped_cmd = response_candidate_cmd;
    assign rate_step_abs = (cfg_rate_limit_step == 16'h0000) ? 16'h0001 : cfg_rate_limit_step;
    assign rate_limited_hi = last_cmd_r + rate_step_abs;
    assign rate_limited_lo = last_cmd_r - rate_step_abs;
    assign rate_limited_cmd = (unclamped_cmd > rate_limited_hi) ? rate_limited_hi :
                              (unclamped_cmd < rate_limited_lo) ? rate_limited_lo :
                              unclamped_cmd;
    assign clamp_hit = (response_valid && ((rate_limited_cmd < clamp_min) || (rate_limited_cmd > clamp_max)));
    assign timeout_hit = (cfg_timeout_threshold != 16'h0000) && (response_age > cfg_timeout_threshold);
    assign fallback_hit = (!cfg_enable) || response_duplicate || response_out_of_order || response_stale || timeout_hit;
    assign accept_hit = cfg_enable && response_valid && !response_duplicate && !response_out_of_order && !response_stale && !timeout_hit;

    assign actuator_cmd = actuator_cmd_r;
    assign actuator_cmd_valid = actuator_cmd_valid_r;
    assign status_flags = status_flags_r;
    assign fault_counters = fault_counters_r;
    assign debug_counters = debug_counters_r;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            actuator_cmd_r <= 16'h0000;
            actuator_cmd_valid_r <= 1'b0;
            status_flags_r <= 16'h0000;
            fault_counters_r <= 32'h00000000;
            debug_counters_r <= 32'h00000000;
            last_cmd_r <= 16'h0000;
            safe_cmd_r <= 16'h0000;
        end else begin
            safe_cmd_r <= cfg_safe_cmd;
            status_flags_r[0] <= status_flags_in[0] | request_issued_pulse;
            status_flags_r[1] <= status_flags_in[1] | response_received_pulse;
            status_flags_r[2] <= status_flags_in[2] | clamp_hit;
            status_flags_r[3] <= status_flags_in[3] | timeout_hit;
            status_flags_r[4] <= status_flags_in[4] | response_stale;
            status_flags_r[5] <= status_flags_in[5] | response_duplicate;
            status_flags_r[6] <= status_flags_in[6] | response_out_of_order;
            status_flags_r[7] <= status_flags_in[7] | fallback_hit;
            status_flags_r[15:8] <= 8'h00;
            if (accept_hit) begin
                if (cfg_rate_limit_en) begin
                    if (rate_limited_cmd < clamp_min) begin
                        actuator_cmd_r <= clamp_min;
                        last_cmd_r <= clamp_min;
                        status_flags_r[2] <= 1'b1;
                    end else if (rate_limited_cmd > clamp_max) begin
                        actuator_cmd_r <= clamp_max;
                        last_cmd_r <= clamp_max;
                        status_flags_r[2] <= 1'b1;
                    end else begin
                        actuator_cmd_r <= rate_limited_cmd;
                        last_cmd_r <= rate_limited_cmd;
                    end
                end else begin
                    if (response_candidate_cmd < clamp_min) begin
                        actuator_cmd_r <= clamp_min;
                        last_cmd_r <= clamp_min;
                        status_flags_r[2] <= 1'b1;
                    end else if (response_candidate_cmd > clamp_max) begin
                        actuator_cmd_r <= clamp_max;
                        last_cmd_r <= clamp_max;
                        status_flags_r[2] <= 1'b1;
                    end else begin
                        actuator_cmd_r <= response_candidate_cmd;
                        last_cmd_r <= response_candidate_cmd;
                    end
                end
                actuator_cmd_valid_r <= 1'b1;
            end else begin
                actuator_cmd_valid_r <= 1'b0;
                if (!cfg_enable || timeout_hit || response_stale || response_duplicate || response_out_of_order) begin
                    actuator_cmd_r <= safe_cmd_r;
                    last_cmd_r <= safe_cmd_r;
                    status_flags_r[7] <= 1'b1;
                end
            end
            if (clamp_hit) begin
                fault_counters_r <= fault_counters_r + 32'h00000001;
            end
            if (timeout_hit) begin
                fault_counters_r <= fault_counters_r + 32'h00000002;
            end
            if (response_stale) begin
                fault_counters_r <= fault_counters_r + 32'h00000004;
            end
            if (response_duplicate) begin
                fault_counters_r <= fault_counters_r + 32'h00000008;
            end
            if (response_out_of_order) begin
                fault_counters_r <= fault_counters_r + 32'h00000010;
            end
            if (request_issued_pulse) begin
                debug_counters_r <= debug_counters_r + 32'h00000001;
            end
            if (response_received_pulse) begin
                debug_counters_r <= debug_counters_r + 32'h00000002;
            end
            if (accept_hit) begin
                debug_counters_r <= debug_counters_r + 32'h00000004;
            end
        end
    end
endmodule

module adaptive_aero_control_registers (
    clk,
    rst_n,
    cfg_addr,
    cfg_wdata,
    cfg_rdata,
    cfg_we,
    cfg_re,
    cfg_enable,
    cfg_safe_cmd,
    cfg_cmd_min,
    cfg_cmd_max,
    cfg_rate_limit_en,
    cfg_rate_limit_step,
    cfg_timeout_threshold,
    cfg_stale_threshold,
    status_flags_in,
    fault_counters_in,
    debug_counters_in
);
    input clk;
    input rst_n;
    input [7:0] cfg_addr;
    input [31:0] cfg_wdata;
    output [31:0] cfg_rdata;
    input cfg_we;
    input cfg_re;
    output cfg_enable;
    output [15:0] cfg_safe_cmd;
    output [15:0] cfg_cmd_min;
    output [15:0] cfg_cmd_max;
    output cfg_rate_limit_en;
    output [15:0] cfg_rate_limit_step;
    output [15:0] cfg_timeout_threshold;
    output [15:0] cfg_stale_threshold;
    input [15:0] status_flags_in;
    input [31:0] fault_counters_in;
    input [31:0] debug_counters_in;

    reg cfg_enable_r;
    reg [15:0] cfg_safe_cmd_r;
    reg [15:0] cfg_cmd_min_r;
    reg [15:0] cfg_cmd_max_r;
    reg cfg_rate_limit_en_r;
    reg [15:0] cfg_rate_limit_step_r;
    reg [15:0] cfg_timeout_threshold_r;
    reg [15:0] cfg_stale_threshold_r;
    reg [31:0] cfg_rdata_r;

    assign cfg_enable = cfg_enable_r;
    assign cfg_safe_cmd = cfg_safe_cmd_r;
    assign cfg_cmd_min = cfg_cmd_min_r;
    assign cfg_cmd_max = cfg_cmd_max_r;
    assign cfg_rate_limit_en = cfg_rate_limit_en_r;
    assign cfg_rate_limit_step = cfg_rate_limit_step_r;
    assign cfg_timeout_threshold = cfg_timeout_threshold_r;
    assign cfg_stale_threshold = cfg_stale_threshold_r;
    assign cfg_rdata = cfg_rdata_r;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cfg_enable_r <= 1'b0;
            cfg_safe_cmd_r <= 16'h0000;
            cfg_cmd_min_r <= 16'h0000;
            cfg_cmd_max_r <= 16'h7FFF;
            cfg_rate_limit_en_r <= 1'b0;
            cfg_rate_limit_step_r <= 16'h0001;
            cfg_timeout_threshold_r <= 16'h00FF;
            cfg_stale_threshold_r <= 16'h0008;
            cfg_rdata_r <= 32'h00000000;
        end else begin
            if (cfg_we) begin
                case (cfg_addr)
                    8'h00: cfg_enable_r <= cfg_wdata[0];
                    8'h04: cfg_safe_cmd_r[7:0] <= cfg_wdata[7:0];
                    8'h05: cfg_safe_cmd_r[15:8] <= cfg_wdata[7:0];
                    8'h08: cfg_cmd_min_r[7:0] <= cfg_wdata[7:0];
                    8'h09: cfg_cmd_min_r[15:8] <= cfg_wdata[7:0];
                    8'h0C: cfg_cmd_max_r[7:0] <= cfg_wdata[7:0];
                    8'h0D: cfg_cmd_max_r[15:8] <= cfg_wdata[7:0];
                    8'h10: cfg_rate_limit_en_r <= cfg_wdata[0];
                    8'h14: cfg_rate_limit_step_r[7:0] <= cfg_wdata[7:0];
                    8'h15: cfg_rate_limit_step_r[15:8] <= cfg_wdata[7:0];
                    8'h18: cfg_timeout_threshold_r[7:0] <= cfg_wdata[7:0];
                    8'h19: cfg_timeout_threshold_r[15:8] <= cfg_wdata[7:0];
                    8'h1C: cfg_stale_threshold_r[7:0] <= cfg_wdata[7:0];
                    8'h1D: cfg_stale_threshold_r[15:8] <= cfg_wdata[7:0];
                    default: begin
                    end
                endcase
            end
            if (cfg_re) begin
                case (cfg_addr)
                    8'h00: cfg_rdata_r <= {31'h0, cfg_enable_r};
                    8'h04: cfg_rdata_r <= {24'h0, cfg_safe_cmd_r[7:0]};
                    8'h05: cfg_rdata_r <= {24'h0, cfg_safe_cmd_r[15:8]};
                    8'h08: cfg_rdata_r <= {24'h0, cfg_cmd_min_r[7:0]};
                    8'h09: cfg_rdata_r <= {24'h0, cfg_cmd_min_r[15:8]};
                    8'h0C: cfg_rdata_r <= {24'h0, cfg_cmd_max_r[7:0]};
                    8'h0D: cfg_rdata_r <= {24'h0, cfg_cmd_max_r[15:8]};
                    8'h10: cfg_rdata_r <= {31'h0, cfg_rate_limit_en_r};
                    8'h14: cfg_rdata_r <= {24'h0, cfg_rate_limit_step_r[7:0]};
                    8'h15: cfg_rdata_r <= {24'h0, cfg_rate_limit_step_r[15:8]};
                    8'h18: cfg_rdata_r <= {24'h0, cfg_timeout_threshold_r[7:0]};
                    8'h19: cfg_rdata_r <= {24'h0, cfg_timeout_threshold_r[15:8]};
                    8'h1C: cfg_rdata_r <= {24'h0, cfg_stale_threshold_r[7:0]};
                    8'h1D: cfg_rdata_r <= {24'h0, cfg_stale_threshold_r[15:8]};
                    8'h20: cfg_rdata_r <= {16'h0000, status_flags_in};
                    8'h21: cfg_rdata_r <= 32'h00000000;
                    8'h24: cfg_rdata_r <= fault_counters_in;
                    8'h25: cfg_rdata_r <= fault_counters_in;
                    8'h26: cfg_rdata_r <= fault_counters_in;
                    8'h27: cfg_rdata_r <= fault_counters_in;
                    8'h28: cfg_rdata_r <= debug_counters_in;
                    8'h29: cfg_rdata_r <= debug_counters_in;
                    8'h2A: cfg_rdata_r <= debug_counters_in;
                    8'h2B: cfg_rdata_r <= debug_counters_in;
                    default: cfg_rdata_r <= 32'h00000000;
                endcase
            end
        end
    end
endmodule

module adaptive_aero_control_top (
    clk,
    rst_n,
    cmd_in_valid,
    cmd_in_ready,
    cmd_in_data,
    cmd_out_valid,
    cmd_out_ready,
    cmd_out_data,
    cfg_addr,
    cfg_wdata,
    cfg_rdata,
    cfg_we,
    cfg_re,
    actuator_cmd,
    actuator_cmd_valid,
    status_flags
);
    input clk;
    input rst_n;
    input cmd_in_valid;
    output cmd_in_ready;
    input [127:0] cmd_in_data;
    output cmd_out_valid;
    input cmd_out_ready;
    output [127:0] cmd_out_data;
    input [7:0] cfg_addr;
    input [31:0] cfg_wdata;
    output [31:0] cfg_rdata;
    input cfg_we;
    input cfg_re;
    output [15:0] actuator_cmd;
    output actuator_cmd_valid;
    output [15:0] status_flags;

    wire cfg_enable;
    wire [15:0] cfg_safe_cmd;
    wire [15:0] cfg_cmd_min;
    wire [15:0] cfg_cmd_max;
    wire cfg_rate_limit_en;
    wire [15:0] cfg_rate_limit_step;
    wire [15:0] cfg_timeout_threshold;
    wire [15:0] cfg_stale_threshold;

    wire [15:0] request_seq_out;
    wire [15:0] last_accepted_seq_out;
    wire [15:0] response_candidate_cmd;
    wire [15:0] response_seq;
    wire [15:0] response_age;
    wire response_valid;
    wire response_duplicate;
    wire response_out_of_order;
    wire response_stale;
    wire request_issued_pulse;
    wire response_received_pulse;
    wire [15:0] transport_status_flags;

    wire [15:0] safety_status_flags;
    wire [31:0] fault_counters;
    wire [31:0] debug_counters;

    adaptive_aero_control_registers u_regs (
        .clk(clk),
        .rst_n(rst_n),
        .cfg_addr(cfg_addr),
        .cfg_wdata(cfg_wdata),
        .cfg_rdata(cfg_rdata),
        .cfg_we(cfg_we),
        .cfg_re(cfg_re),
        .cfg_enable(cfg_enable),
        .cfg_safe_cmd(cfg_safe_cmd),
        .cfg_cmd_min(cfg_cmd_min),
        .cfg_cmd_max(cfg_cmd_max),
        .cfg_rate_limit_en(cfg_rate_limit_en),
        .cfg_rate_limit_step(cfg_rate_limit_step),
        .cfg_timeout_threshold(cfg_timeout_threshold),
        .cfg_stale_threshold(cfg_stale_threshold),
        .status_flags_in(status_flags),
        .fault_counters_in(fault_counters),
        .debug_counters_in(debug_counters)
    );

    adaptive_aero_control_transport u_transport (
        .clk(clk),
        .rst_n(rst_n),
        .cmd_in_valid(cmd_in_valid),
        .cmd_in_ready(cmd_in_ready),
        .cmd_in_data(cmd_in_data),
        .cmd_out_valid(cmd_out_valid),
        .cmd_out_ready(cmd_out_ready),
        .cmd_out_data(cmd_out_data),
        .cfg_enable(cfg_enable),
        .cfg_timeout_threshold(cfg_timeout_threshold),
        .cfg_stale_threshold(cfg_stale_threshold),
        .request_seq_out(request_seq_out),
        .last_accepted_seq_out(last_accepted_seq_out),
        .response_candidate_cmd(response_candidate_cmd),
        .response_seq(response_seq),
        .response_age(response_age),
