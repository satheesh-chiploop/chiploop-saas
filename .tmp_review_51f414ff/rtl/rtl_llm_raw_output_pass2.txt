        .csb(history_csb),
        .we(history_we),
        .addr(history_addr),
        .din(history_din),
        .dout(history_dout)

        .cfg_addr(cfg_addr),

        .cfg_rdata(cfg_rdata),

        .cfg_enable(cfg_enable),
        .cfg_safe_cmd(cfg_safe_cmd),
        .cfg_cmd_min(cfg_cmd_min),

        .fault_counters_in(fault_counters),
        .debug_counters_in(debug_counters)

        .clk(clk),
        .rst_n(rst_n),
        .cmd_in_valid(cmd_in_valid),
        .cmd_in_ready(cmd_in_ready),
        .cmd_in_data(cmd_in_data),
        .cmd_out_valid(cmd_out_valid),
        .cmd_out_ready(cmd_out_ready),
        .cmd_out_data(cmd_out_data),
        .cfg_enable(cfg_enable),
        .cfg_timeout_threshold(cfg_timeout_threshold),
        .cfg_stale_threshold(cfg_stale_threshold),
        .request_seq_out(request_seq_out),
        .last_accepted_seq_out(last_accepted_seq_out),
        .response_candidate_cmd(response_candidate_cmd),
        .response_seq(response_seq),
        .response_age(response_age),
        .response_valid(response_valid),
        .response_duplicate(response_duplicate),
        .response_out_of_order(response_out_of_order),
        .response_stale(response_stale),
        .request_issued_pulse(request_issued_pulse),
        .response_received_pulse(response_received_pulse),
        .status_flags_in(transport_status_flags),
        .history_csb(history_csb),
        .history_we(history_we),
        .history_addr(history_addr),
        .history_din(history_din),
        .history_dout(history_dout)
    );

    adaptive_aero_control_safety u_safety (
        .clk(clk),
        .rst_n(rst_n),
        .cfg_enable(cfg_enable),
        .cfg_safe_cmd(cfg_safe_cmd),

        .cfg_stale_threshold(cfg_stale_threshold),

        .response_age(response_age),
        .response_valid(response_valid),
        .response_duplicate(response_duplicate),
        .response_out_of_order(response_out_of_order),
        .response_stale(response_stale),
        .request_issued_pulse(request_issued_pulse),
        .response_received_pulse(response_received_pulse),
        .status_flags_in(transport_status_flags),
        .actuator_cmd(actuator_cmd),
        .actuator_cmd_valid(actuator_cmd_valid),
        .status_flags(safety_status_flags),
        .fault_counters(fault_counters),
        .debug_counters(debug_counters)

    assign status_flags = safety_status_flags;
