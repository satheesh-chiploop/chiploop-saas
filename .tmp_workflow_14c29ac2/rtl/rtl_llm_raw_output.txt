---BEGIN adaptive_aero_control_top.v---
module adaptive_aero_control_top (
    clk,
    reset_n,
    mmio_addr,
    mmio_wdata,
    mmio_valid,
    mmio_we,
    mmio_rdata,
    mmio_ready,
    req_valid,
    req_ready,
    req_data,
    rsp_valid,
    rsp_ready,
    rsp_data,
    actuator_cmd,
    actuator_valid,
    fault_flags
);
    input clk;
    input reset_n;
    input [7:0] mmio_addr;
    input [31:0] mmio_wdata;
    input mmio_valid;
    input mmio_we;
    output [31:0] mmio_rdata;
    output mmio_ready;
    output req_valid;
    input req_ready;
    output [95:0] req_data;
    input rsp_valid;
    output rsp_ready;
    input [95:0] rsp_data;
    output [31:0] actuator_cmd;
    output actuator_valid;
    output [7:0] fault_flags;

    reg [31:0] mmio_rdata_r;
    reg mmio_ready_r;
    reg req_valid_r;
    reg [95:0] req_data_r;
    reg rsp_ready_r;
    reg [31:0] actuator_cmd_r;
    reg actuator_valid_r;
    reg [7:0] fault_flags_r;

    reg [3:0] state_q;
    reg [3:0] state_d;

    reg control_enable_q;
    reg control_arm_q;
    reg [3:0] control_mode_q;
    reg [15:0] timeout_cycles_q;
    reg [31:0] clamp_min_q;
    reg [31:0] clamp_max_q;
    reg [15:0] seq_seed_q;
    reg [15:0] velocity_setpoint_q;
    reg [7:0] geom_ref_id_q;
    reg [3:0] request_opcode_q;

    reg pending_request_q;
    reg response_valid_seen_q;
    reg fault_latched_q;
    reg stale_reject_q;
    reg timeout_fault_q;
    reg malformed_response_q;
    reg clamp_saturation_q;
    reg sequence_error_q;
    reg configuration_error_q;

    reg [15:0] seq_current_q;
    reg [15:0] outstanding_seq_q;
    reg [15:0] timeout_count_q;
    reg [31:0] accepted_cmd_q;
    reg accepted_cmd_valid_q;

    reg [95:0] req_word;
    reg [31:0] readback_status;
    reg [31:0] readback_faults;
    reg [31:0] readback_control;
    reg [31:0] readback_timeout;
    reg [31:0] readback_clamp_min;
    reg [31:0] readback_clamp_max;
    reg [31:0] readback_seq_seed;
    reg [31:0] readback_velocity;
    reg [31:0] readback_geom;
    reg [31:0] readback_default;

    reg [31:0] proposed_cmd;
    reg [31:0] clamped_cmd;
    reg clamp_hit;
    reg response_ok;
    reg pending_next;
    reg [15:0] timeout_limit;
    reg [15:0] timeout_next;

    wire armed_live;
    assign armed_live = control_enable_q & control_arm_q;

    assign mmio_rdata = mmio_rdata_r;
    assign mmio_ready = mmio_ready_r;
    assign req_valid = req_valid_r;
    assign req_data = req_data_r;
    assign rsp_ready = rsp_ready_r;
    assign actuator_cmd = actuator_cmd_r;
    assign actuator_valid = actuator_valid_r;
    assign fault_flags = fault_flags_r;

    localparam [3:0] S_IDLE = 4'd0;
    localparam [3:0] S_REQUEST = 4'd1;
    localparam [3:0] S_WAIT = 4'd2;
    localparam [3:0] S_VALIDATE = 4'd3;
    localparam [3:0] S_APPLY = 4'd4;
    localparam [3:0] S_FAULT = 4'd5;

    always @(*) begin
        state_d = state_q;
        mmio_rdata_r = 32'h00000000;
        mmio_ready_r = 1'b0;
        req_valid_r = 1'b0;
        req_data_r = 96'h000000000000000000000000;
        rsp_ready_r = 1'b0;
        actuator_cmd_r = 32'h00000000;
        actuator_valid_r = 1'b0;
        fault_flags_r = 8'h00;
        readback_default = 32'h00000000;
        readback_control = {24'h000000, control_mode_q, 2'b00, control_arm_q, control_enable_q};
        readback_timeout = {16'h0000, timeout_cycles_q};
        readback_clamp_min = clamp_min_q;
        readback_clamp_max = clamp_max_q;
        readback_seq_seed = {16'h0000, seq_seed_q};
        readback_velocity = {16'h0000, velocity_setpoint_q};
        readback_geom = {12'h000, request_opcode_q, geom_ref_id_q};
        readback_status = 32'h00000000;
        readback_faults = 32'h00000000;
        proposed_cmd = {16'h0000, velocity_setpoint_q};
        clamped_cmd = proposed_cmd;
        clamp_hit = 1'b0;
        response_ok = 1'b0;
        pending_next = pending_request_q;
        timeout_limit = (timeout_cycles_q == 16'h0000) ? 16'h0001 : timeout_cycles_q;
        timeout_next = timeout_count_q;

        readback_status[0] = armed_live;
        readback_status[1] = pending_request_q;
        readback_status[2] = response_valid_seen_q;
        readback_status[3] = fault_latched_q;
        readback_status[8] = stale_reject_q;
        readback_status[9] = timeout_fault_q;
        readback_status[10] = malformed_response_q;
        readback_status[11] = clamp_saturation_q;
        readback_status[13] = sequence_error_q;
        readback_status[14] = configuration_error_q;

        readback_faults[0] = stale_reject_q;
        readback_faults[1] = timeout_fault_q;
        readback_faults[2] = malformed_response_q;
        readback_faults[3] = clamp_saturation_q;
        readback_faults[4] = fault_latched_q;
        readback_faults[5] = sequence_error_q;
        readback_faults[6] = configuration_error_q;
        readback_faults[7] = pending_request_q & ~armed_live;

        if (proposed_cmd < clamp_min_q) begin
            clamped_cmd = clamp_min_q;
            clamp_hit = 1'b1;
        end else if (proposed_cmd > clamp_max_q) begin
            clamped_cmd = clamp_max_q;
            clamp_hit = 1'b1;
        end

        if (rsp_valid && pending_request_q) begin
            if (rsp_data[95:80] == outstanding_seq_q) begin
                if (rsp_data[79:72] == 8'h01) begin
                    response_ok = 1'b1;
                end
            end
        end

        mmio_ready_r = mmio_valid;
        rsp_ready_r = pending_request_q & armed_live;

        case (mmio_addr)
            8'h00: begin
                readback_control = {24'h000000, control_mode_q, 2'b00, control_arm_q, control_enable_q};
                if (mmio_valid && mmio_we) begin
                    readback_control[0] = mmio_wdata[0];
                    readback_control[1] = mmio_wdata[1];
                    readback_control[2] = mmio_wdata[2];
                    readback_control[7:4] = mmio_wdata[7:4];
                end
                mmio_rdata_r = readback_control;
            end
            8'h04: begin
                mmio_rdata_r = readback_timeout;
            end
            8'h08: begin
                mmio_rdata_r = readback_clamp_min;
            end
            8'h0C: begin
                mmio_rdata_r = readback_clamp_max;
            end
            8'h14: begin
                mmio_rdata_r = readback_seq_seed;
            end
            8'h18: begin
                mmio_rdata_r = readback_velocity;
            end
            8'h1C: begin
                mmio_rdata_r = readback_geom;
            end
            8'h20: begin
                mmio_rdata_r = readback_status;
            end
            8'h24: begin
                mmio_rdata_r = readback_faults;
            end
            default: begin
                mmio_rdata_r = readback_default;
            end
        endcase

        if (state_q == S_IDLE) begin
            if (armed_live && !pending_request_q) begin
                state_d = S_REQUEST;
            end
        end else if (state_q == S_REQUEST) begin
            state_d = S_WAIT;
        end else if (state_q == S_WAIT) begin
            if (rsp_valid) begin
                state_d = S_VALIDATE;
            end else if (pending_request_q && (timeout_count_q >= timeout_limit)) begin
                state_d = S_FAULT;
            end
        end else if (state_q == S_VALIDATE) begin
            if (response_ok) begin
                state_d = S_APPLY;
            end else begin
                state_d = S_FAULT;
            end
        end else if (state_q == S_APPLY) begin
            state_d = S_IDLE;
        end else begin
            state_d = S_FAULT;
        end

        if (armed_live && !pending_request_q && !fault_latched_q) begin
            req_valid_r = 1'b1;
            req_word[95:80] = seq_current_q;
            req_word[79:72] = request_opcode_q;
            req_word[71:64] = geom_ref_id_q;
            req_word[63:48] = velocity_setpoint_q;
            req_word[47:44] = control_mode_q;
            req_word[43:0] = 44'h00000000000;
            req_data_r = req_word;
        end

        if (state_q == S_APPLY) begin
            actuator_cmd_r = clamped_cmd;
            actuator_valid_r = 1'b1;
        end else if (fault_latched_q || configuration_error_q || timeout_fault_q || stale_reject_q || malformed_response_q || sequence_error_q) begin
            actuator_valid_r = 1'b0;
        end

        if (fault_latched_q) begin
            fault_flags_r = readback_faults[7:0];
        end else begin
            fault_flags_r = readback_faults[7:0];
        end
    end

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            state_q <= S_IDLE;
            control_enable_q <= 1'b0;
            control_arm_q <= 1'b0;
            control_mode_q <= 4'h0;
            timeout_cycles_q <= 16'd32;
            clamp_min_q <= 32'd0;
            clamp_max_q <= 32'd4095;
            seq_seed_q <= 16'd0;
            velocity_setpoint_q <= 16'd0;
            geom_ref_id_q <= 8'd0;
            request_opcode_q <= 4'd1;
            pending_request_q <= 1'b0;
            response_valid_seen_q <= 1'b0;
            fault_latched_q <= 1'b0;
            stale_reject_q <= 1'b0;
            timeout_fault_q <= 1'b0;
            malformed_response_q <= 1'b0;
            clamp_saturation_q <= 1'b0;
            sequence_error_q <= 1'b0;
            configuration_error_q <= 1'b0;
            seq_current_q <= 16'd0;
            outstanding_seq_q <= 16'd0;
            timeout_count_q <= 16'd0;
            accepted_cmd_q <= 32'd0;
            accepted_cmd_valid_q <= 1'b0;
        end else begin
            state_q <= state_d;
            if (mmio_valid && mmio_we) begin
                case (mmio_addr)
                    8'h00: begin
                        control_enable_q <= mmio_wdata[0];
                        control_arm_q <= mmio_wdata[1];
                        if (mmio_wdata[2]) begin
                            fault_latched_q <= 1'b0;
                            stale_reject_q <= 1'b0;
                            timeout_fault_q <= 1'b0;
                            malformed_response_q <= 1'b0;
                            clamp_saturation_q <= 1'b0;
                            sequence_error_q <= 1'b0;
                            configuration_error_q <= 1'b0;
                        end
                        control_mode_q <= mmio_wdata[7:4];
                    end
                    8'h04: begin
                        timeout_cycles_q <= mmio_wdata[15:0];
                    end
                    8'h08: begin
                        clamp_min_q <= mmio_wdata;
                    end
                    8'h0C: begin
                        clamp_max_q <= mmio_wdata;
                    end
                    8'h14: begin
                        seq_seed_q <= mmio_wdata[15:0];
                        seq_current_q <= mmio_wdata[15:0];
                    end
                    8'h18: begin
                        velocity_setpoint_q <= mmio_wdata[15:0];
                    end
                    8'h1C: begin
                        geom_ref_id_q <= mmio_wdata[7:0];
                        request_opcode_q <= mmio_wdata[11:8];
                    end
                    default: begin
                    end
                endcase
            end

            if (mmio_valid && mmio_we && (mmio_addr == 8'h00) && mmio_wdata[2]) begin
                fault_latched_q <= 1'b0;
                stale_reject_q <= 1'b0;
                timeout_fault_q <= 1'b0;
                malformed_response_q <= 1'b0;
                clamp_saturation_q <= 1'b0;
                sequence_error_q <= 1'b0;
                configuration_error_q <= 1'b0;
            end

            if ((clamp_min_q > clamp_max_q) && (mmio_addr != 8'h08) && (mmio_addr != 8'h0C)) begin
                configuration_error_q <= 1'b1;
                fault_latched_q <= 1'b1;
            end

            if (state_q == S_REQUEST) begin
                pending_request_q <= 1'b1;
                outstanding_seq_q <= seq_current_q;
                seq_current_q <= seq_current_q + 16'd1;
                timeout_count_q <= 16'd0;
                response_valid_seen_q <= 1'b0;
            end else if (state_q == S_WAIT) begin
                if (pending_request_q) begin
                    timeout_count_q <= timeout_count_q + 16'd1;
                end
                if (rsp_valid) begin
                    response_valid_seen_q <= 1'b1;
                end
            end else if (state_q == S_VALIDATE) begin
                if (response_ok) begin
                    pending_request_q <= 1'b0;
                    accepted_cmd_q <= clamped_cmd;
                    accepted_cmd_valid_q <= 1'b1;
                    if (clamp_hit) begin
                        clamp_saturation_q <= 1'b1;
                    end
                    fault_latched_q <= fault_latched_q;
                end else begin
                    pending_request_q <= 1'b0;
                    fault_latched_q <= 1'b1;
                    if (rsp_valid && pending_request_q && (rsp_data[95:80] != outstanding_seq_q)) begin
                        stale_reject_q <= 1'b1;
                        sequence_error_q <= 1'b1;
                    end else begin
                        malformed_response_q <= 1'b1;
                    end
                end
            end else if (state_q == S_FAULT) begin
                pending_request_q <= 1'b0;
                fault_latched_q <= 1'b1;
                timeout_fault_q <= 1'b1;
            end else if (state_q == S_APPLY) begin
                pending_request_q <= 1'b0;
                response_valid_seen_q <= 1'b1;
            end
        end
    end
endmodule
---END adaptive_aero_control_top.v---
