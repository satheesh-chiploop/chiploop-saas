    wire [15:0] cfg_freshness_threshold_w;
    wire [31:0] cfg_min_cmd_w;
    wire [31:0] cfg_max_cmd_w;
    wire [31:0] cfg_fallback_cmd_w;
    wire [15:0] cfg_step_limit_w;
    wire [2:0] cfg_outstanding_limit_w;
    wire [7:0] read_byte;

    assign cfg_timeout_threshold_w = {reg_timeout_1, reg_timeout_0};
    assign cfg_freshness_threshold_w = {reg_freshness_1, reg_freshness_0};
    assign cfg_min_cmd_w = {reg_limits_3, reg_limits_2, reg_limits_1, reg_limits_0};
    assign cfg_max_cmd_w = {reg_limits_7, reg_limits_6, reg_limits_5, reg_limits_4};
    assign cfg_fallback_cmd_w = {reg_limits_11, reg_limits_10, reg_limits_9, reg_limits_8};
    assign cfg_step_limit_w = {reg_limits_13, reg_limits_12};
    assign cfg_outstanding_limit_w = reg_pipeline_0[2:0];

    function [63:0] read_mux;
        input [63:0] addr;
        begin
            read_mux = 64'h0000000000000000;
            case (addr[7:0])
                8'h00: read_mux = {56'h00000000000000, reg_ctrl};
                8'h01: read_mux = {56'h00000000000000, reg_request_0};
                8'h02: read_mux = {56'h00000000000000, reg_request_1};
                8'h03: read_mux = {56'h00000000000000, reg_request_2};
                8'h04: read_mux = {56'h00000000000000, reg_request_3};
                8'h05: read_mux = {56'h00000000000000, reg_request_4};
                8'h06: read_mux = {56'h00000000000000, reg_request_5};
                8'h07: read_mux = {56'h00000000000000, reg_request_6};
                8'h08: read_mux = {56'h00000000000000, reg_request_7};
                8'h09: read_mux = {56'h00000000000000, reg_request_8};
                8'h0A: read_mux = {56'h00000000000000, reg_request_9};
                8'h0B: read_mux = {56'h00000000000000, reg_request_10};
                8'h0C: read_mux = {56'h00000000000000, reg_request_11};
                8'h10: read_mux = {56'h00000000000000, reg_limits_0};
                8'h11: read_mux = {56'h00000000000000, reg_limits_1};
                8'h12: read_mux = {56'h00000000000000, reg_limits_2};
                8'h13: read_mux = {56'h00000000000000, reg_limits_3};
                8'h14: read_mux = {56'h00000000000000, reg_limits_4};
                8'h15: read_mux = {56'h00000000000000, reg_limits_5};
                8'h16: read_mux = {56'h00000000000000, reg_limits_6};
                8'h17: read_mux = {56'h00000000000000, reg_limits_7};
                8'h18: read_mux = {56'h00000000000000, reg_limits_8};
                8'h19: read_mux = {56'h00000000000000, reg_limits_9};
                8'h1A: read_mux = {56'h00000000000000, reg_limits_10};
                8'h1B: read_mux = {56'h00000000000000, reg_limits_11};
                8'h1C: read_mux = {56'h00000000000000, reg_limits_12};
                8'h1D: read_mux = {56'h00000000000000, reg_limits_13};
                8'h20: read_mux = {56'h00000000000000, reg_timeout_0};
                8'h21: read_mux = {56'h00000000000000, reg_timeout_1};
                8'h22: read_mux = {56'h00000000000000, reg_freshness_0};
                8'h23: read_mux = {56'h00000000000000, reg_freshness_1};
                8'h24: read_mux = {56'h00000000000000, reg_pipeline_0};
                8'h25: read_mux = {56'h00000000000000, reg_status_policy_0};
                8'h30: read_mux = {56'h00000000000000, telemetry_state, telemetry_busy, telemetry_timeout_fault, telemetry_stale_fault, telemetry_protocol_fault, 51'h0000000000000};
                8'h31: read_mux = {56'h00000000000000, telemetry_clamp_active, telemetry_fallback_active, telemetry_irq_latched, 5'h00};
                8'h32: read_mux = {56'h00000000000000, telemetry_outstanding_count, telemetry_response_count[4:0]};
                8'h33: read_mux = {56'h00000000000000, telemetry_last_seq[7:0]};
                8'h34: read_mux = {56'h00000000000000, telemetry_last_seq[15:8]};
                8'h35: read_mux = {56'h00000000000000, telemetry_response_count[7:0]};
                8'h36: read_mux = {56'h00000000000000, telemetry_response_count[15:8]};
                8'h37: read_mux = {56'h00000000000000, telemetry_timeout_fault, cfg_sticky_fault, 6'h00};
                8'h3F: read_mux = {63'h000000000000000, mmio_error};
                default: read_mux = 64'h0000000000000000;
            endcase
        end
    endfunction

    always @(*) begin
        mmio_ready = 1'b1;
        mmio_rdata = read_mux(mmio_addr);
        mmio_rvalid = mmio_valid;
        mmio_error = 1'b0;

        cfg_enable = reg_ctrl[0];
        cfg_soft_reset = reg_ctrl[1];
        cfg_launch = reg_ctrl[2];
        cfg_fault_clear = reg_ctrl[3];
        cfg_sticky_fault = reg_ctrl[4];
        cfg_pipelined_mode = reg_ctrl[5];
        cfg_session_rollover = reg_ctrl[6];
        cfg_status_clear = reg_ctrl[7];

        cfg_request_type = reg_request_0;
        cfg_sequence_seed = {reg_request_2, reg_request_1};
        cfg_operating_velocity = {reg_request_4, reg_request_3};
        cfg_velocity_fraction = {reg_request_6, reg_request_5};

        cfg_max_cmd = cfg_max_cmd_w;

        cfg_irq_on_fault = reg_status_policy_0[1];
        cfg_irq_on_timeout = reg_status_policy_0[2];
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            reg_ctrl <= 8'h00;
            reg_request_0 <= 8'h00;
            reg_request_1 <= 8'h00;
            reg_request_2 <= 8'h00;
            reg_request_3 <= 8'h00;
            reg_request_4 <= 8'h00;
            reg_request_5 <= 8'h00;

            reg_limits_7 <= 8'h00;
            reg_limits_8 <= 8'h00;
            reg_limits_9 <= 8'h00;
            reg_limits_10 <= 8'h00;
            reg_limits_11 <= 8'h00;
            reg_limits_12 <= 8'h10;
            reg_limits_13 <= 8'h00;
            reg_timeout_0 <= 8'h10;
            reg_timeout_1 <= 8'h00;
            reg_freshness_0 <= 8'h10;
            reg_freshness_1 <= 8'h00;
            reg_pipeline_0 <= 8'h01;
            reg_status_policy_0 <= 8'h07;
        end else begin
            if (mmio_valid && mmio_write) begin
                case (mmio_addr[7:0])
                    8'h00: begin
                        if (mmio_wstrb[0]) reg_ctrl <= mmio_wdata[7:0];
                    end
                    8'h01: begin
                        if (mmio_wstrb[0]) reg_request_0 <= mmio_wdata[7:0];
                    end
                    8'h02: begin
                        if (mmio_wstrb[0]) reg_request_1 <= mmio_wdata[7:0];
                    end
                    8'h03: begin
                        if (mmio_wstrb[0]) reg_request_2 <= mmio_wdata[7:0];
                    end
                    8'h04: begin
                        if (mmio_wstrb[0]) reg_request_3 <= mmio_wdata[7:0];
                    end
                    8'h05: begin
                        if (mmio_wstrb[0]) reg_request_4 <= mmio_wdata[7:0];
                    end
                    8'h06: begin
                        if (mmio_wstrb[0]) reg_request_5 <= mmio_wdata[7:0];
                    end
                    8'h07: begin
                        if (mmio_wstrb[0]) reg_request_6 <= mmio_wdata[7:0];
                    end
                    8'h08: begin
                        if (mmio_wstrb[0]) reg_request_7 <= mmio_wdata[7:0];
                    end
                    8'h09: begin
                        if (mmio_wstrb[0]) reg_request_8 <= mmio_wdata[7:0];
                    end
                    8'h0A: begin
                        if (mmio_wstrb[0]) reg_request_9 <= mmio_wdata[7:0];
                    end
                    8'h0B: begin
                        if (mmio_wstrb[0]) reg_request_10 <= mmio_wdata[7:0];

module adaptive_aero_request_packager (
    input clk,
    input rst_n,
    input cfg_enable,
    input cfg_launch,
    input cfg_pipelined_mode,
    input [2:0] cfg_outstanding_limit,
    input [7:0] cfg_request_type,
    input [15:0] cfg_sequence_seed,
    input [15:0] cfg_operating_velocity,
    input [15:0] cfg_velocity_fraction,
    input [15:0] cfg_reference_id,
    input [15:0] cfg_geometry_id,
    input cfg_session_rollover,
    input req_issue_ack,
    input req_issue_ready,
    input [2:0] req_outstanding_count,
    output reg surrogate_req_valid,
    output reg [63:0] surrogate_req_data,
    input surrogate_req_ready,
    output reg req_launch_accepted,
    output reg [15:0] req_seq_active,
    output reg req_outstanding_inc
);
    reg [15:0] current_seq;
    reg holding_req;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            current_seq <= cfg_sequence_seed;
            surrogate_req_valid <= 1'b0;
            surrogate_req_data <= 64'h0000000000000000;
            req_launch_accepted <= 1'b0;
            req_seq_active <= cfg_sequence_seed;
            req_outstanding_inc <= 1'b0;
            holding_req <= 1'b0;
        end else begin
            req_launch_accepted <= 1'b0;
            req_outstanding_inc <= 1'b0;
            if (!cfg_enable) begin
                surrogate_req_valid <= 1'b0;
                holding_req <= 1'b0;
                current_seq <= cfg_sequence_seed;
                req_seq_active <= cfg_sequence_seed;
            end else begin
                if (cfg_launch && (!surrogate_req_valid || surrogate_req_ready) && (req_outstanding_count < cfg_outstanding_limit)) begin
                    req_launch_accepted <= 1'b1;
                    req_outstanding_inc <= 1'b1;
                    current_seq <= current_seq + 16'h0001;
                    req_seq_active <= current_seq + 16'h0001;
                    surrogate_req_valid <= 1'b1;
                    surrogate_req_data <= {cfg_request_type, current_seq + 16'h0001, cfg_operating_velocity, cfg_velocity_fraction, cfg_reference_id, cfg_geometry_id};
                    holding_req <= 1'b1;
                end else if (surrogate_req_valid && surrogate_req_ready) begin
                    surrogate_req_valid <= 1'b0;
                    holding_req <= 1'b0;
                end else if (!cfg_launch) begin
                    req_outstanding_inc <= 1'b0;
                end
                if (cfg_session_rollover && (cfg_sequence_seed > current_seq)) begin
                    current_seq <= cfg_sequence_seed;
                    req_seq_active <= cfg_sequence_seed;
                end
                if (cfg_pipelined_mode && req_issue_ready && req_issue_ack) begin
                    surrogate_req_valid <= 1'b1;
                end
            end
        end
    end
endmodule

module adaptive_aero_response_validator (
    input clk,
    input rst_n,
    input cfg_enable,
    input [15:0] cfg_timeout_threshold,
    input [15:0] cfg_freshness_threshold,
    input [31:0] cfg_min_cmd,
    input [31:0] cfg_max_cmd,
    input [15:0] cfg_step_limit,
    input [31:0] cfg_fallback_cmd,
    input cfg_session_rollover,
    input [15:0] active_req_seq,
    input [15:0] request_age,
    input timeout_expired,
    input stale_session_event,
    input surrogate_rsp_valid,
    input [63:0] surrogate_rsp_data,
    output reg surrogate_rsp_ready,
    output reg rsp_accept,
    output reg rsp_seq_match,
    output reg rsp_timeout_fault,
    output reg rsp_stale_fault,
    output reg rsp_protocol_fault,
    output reg rsp_clamp_active,
    output reg [31:0] rsp_accepted_cmd,
    output reg [15:0] rsp_response_count
);
    wire [31:0] rsp_cmd_raw;

    assign rsp_cmd_raw = surrogate_rsp_data[31:0];

    always @(*) begin
        surrogate_rsp_ready = 1'b1;
        rsp_accept = 1'b0;
        rsp_seq_match = 1'b0;
        rsp_timeout_fault = 1'b0;
        rsp_stale_fault = 1'b0;
        rsp_protocol_fault = 1'b0;
        rsp_clamp_active = 1'b0;
        rsp_accepted_cmd = 32'h00000000;
        rsp_response_count = 16'h0000;

        if (cfg_enable && surrogate_rsp_valid && !timeout_expired && !stale_session_event) begin
            rsp_seq_match = 1'b1;
            rsp_accept = 1'b1;
            rsp_accepted_cmd = rsp_cmd_raw;
            rsp_response_count = 16'h0001;
            if (rsp_cmd_raw < cfg_min_cmd) begin
                rsp_accepted_cmd = cfg_min_cmd;
                rsp_clamp_active = 1'b1;
            end else if (rsp_cmd_raw > cfg_max_cmd) begin
                rsp_accepted_cmd = cfg_max_cmd;
                rsp_clamp_active = 1'b1;
            end
        end
        if (timeout_expired) rsp_timeout_fault = 1'b1;
        if (stale_session_event) rsp_stale_fault = 1'b1;
        if (surrogate_rsp_valid && (surrogate_rsp_data[63:48] != active_req_seq)) rsp_protocol_fault = 1'b1;
    end
endmodule

module adaptive_aero_timeout_monitor (
    input clk,
    input rst_n,
    input cfg_enable,
    input [15:0] cfg_timeout_threshold,
    input [15:0] cfg_freshness_threshold,
    input [2:0] cfg_outstanding_limit,
    input cfg_session_rollover,
    input req_launch_accepted,
    input rsp_accept,
    input [15:0] active_req_seq,
    input [15:0] request_age,
    input [2:0] outstanding_count,
    input busy,
    output reg timeout_expired,
    output reg stale_session_event,
    output reg timeout_fault_latched
);
    reg [15:0] age_count;
    reg [15:0] last_seq;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            age_count <= 16'h0000;
            last_seq <= 16'h0000;
            timeout_expired <= 1'b0;
            stale_session_event <= 1'b0;
            timeout_fault_latched <= 1'b0;
        end else begin
            if (!cfg_enable) begin
                age_count <= 16'h0000;
                timeout_expired <= 1'b0;
                stale_session_event <= 1'b0;
                timeout_fault_latched <= 1'b0;
            end else begin
                if (req_launch_accepted) begin
                    age_count <= 16'h0000;
                    last_seq <= active_req_seq;
                end else if (busy) begin
                    age_count <= age_count + 16'h0001;
                end
                if (rsp_accept) begin
                    timeout_expired <= 1'b0;
                    stale_session_event <= 1'b0;
                end else if (age_count >= cfg_timeout_threshold) begin
                    timeout_expired <= 1'b1;
                    timeout_fault_latched <= 1'b1;
                end
                if (cfg_session_rollover && (active_req_seq < last_seq)) begin
                    stale_session_event <= 1'b1;
                end
                if (request_age > cfg_freshness_threshold) begin
                    stale_session_event <= 1'b1;
                end
            end
        end
    end
endmodule

module adaptive_aero_fallback_controller (
    input clk,
    input rst_n,
    input cfg_enable,
    input cfg_fault_clear,
    input cfg_sticky_fault,
    input [31:0] cfg_min_cmd,
    input [31:0] cfg_max_cmd,
    input [31:0] cfg_fallback_cmd,
    input [15:0] cfg_step_limit,
    input [31:0] rsp_accepted_cmd,
    input rsp_accept,
    input rsp_timeout_fault,
    input rsp_stale_fault,
    input rsp_protocol_fault,
    input timeout_fault_latched,
    output reg [31:0] safe_cmd,
    output reg fallback_active,
    output reg fault_active,
    output reg clamp_active,
    output reg fault_timeout,
    output reg fault_stale,
    output reg fault_protocol
);
    reg [31:0] prev_safe;
    reg [31:0] bounded_cmd;
    reg [31:0] step_cmd;
    reg any_fault;

    always @(*) begin
        bounded_cmd = cfg_fallback_cmd;
        step_cmd = rsp_accepted_cmd;
        clamp_active = 1'b0;
        fault_timeout = 1'b0;
        fault_stale = 1'b0;
        fault_protocol = 1'b0;
        any_fault = rsp_timeout_fault | rsp_stale_fault | rsp_protocol_fault | timeout_fault_latched;
        if (step_cmd < cfg_min_cmd) step_cmd = cfg_min_cmd;
        if (step_cmd > cfg_max_cmd) step_cmd = cfg_max_cmd;
        bounded_cmd = step_cmd;
        if (cfg_step_limit != 16'h0000) begin
            if (bounded_cmd > prev_safe) begin
                if ((bounded_cmd - prev_safe) > {16'h0000, cfg_step_limit}) bounded_cmd = prev_safe + {16'h0000, cfg_step_limit};
            end else begin
                if ((prev_safe - bounded_cmd) > {16'h0000, cfg_step_limit}) bounded_cmd = prev_safe - {16'h0000, cfg_step_limit};
            end
        end
        clamp_active = (bounded_cmd != rsp_accepted_cmd);
        any_fault = rsp_timeout_fault | rsp_stale_fault | rsp_protocol_fault | timeout_fault_latched;
        fault_timeout = rsp_timeout_fault | timeout_fault_latched;
        fault_stale = rsp_stale_fault;
        fault_protocol = rsp_protocol_fault;
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            safe_cmd <= cfg_fallback_cmd;
            fallback_active <= 1'b1;
            fault_active <= 1'b0;
            prev_safe <= cfg_fallback_cmd;
        end else begin
            if (!cfg_enable) begin
                safe_cmd <= cfg_fallback_cmd;
                fallback_active <= 1'b1;
                fault_active <= 1'b0;
                prev_safe <= cfg_fallback_cmd;
            end else begin
                if (any_fault) begin
                    fallback_active <= 1'b1;
                    fault_active <= 1'b1;
                    safe_cmd <= cfg_fallback_cmd;
                    prev_safe <= cfg_fallback_cmd;
                end else if (cfg_fault_clear && !cfg_sticky_fault) begin
                    fault_active <= 1'b0;
                    fallback_active <= 1'b0;
                    safe_cmd <= bounded_cmd;
                    prev_safe <= bounded_cmd;
                end else if (rsp_accept) begin
                    fallback_active <= 1'b0;
                    safe_cmd <= bounded_cmd;
                    prev_safe <= bounded_cmd;
                end
                if (cfg_sticky_fault && (any_fault || fault_active)) fault_active <= 1'b1;
            end
        end
    end
endmodule

    input fault_active,

    output reg [1:0] actuator_mode,
    output reg actuator_update_pulse

    reg [31:0] last_cmd;

    reg last_fault;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            actuator_cmd <= safe_cmd;
            actuator_valid <= 1'b0;
            actuator_mode <= 2'b01;
            actuator_update_pulse <= 1'b0;
            last_cmd <= safe_cmd;
            last_fallback <= 1'b1;
            last_fault <= 1'b0;
        end else begin
            actuator_update_pulse <= 1'b0;
            if (fault_active) begin
                actuator_cmd <= safe_cmd;
                actuator_valid <= 1'b1;
                actuator_mode <= 2'b10;
            end else if (fallback_active) begin
                actuator_cmd <= safe_cmd;
                actuator_valid <= 1'b1;
                actuator_mode <= 2'b01;

                actuator_cmd <= safe_cmd;
                actuator_valid <= 1'b1;
                actuator_mode <= 2'b00;
            end
            if ((fallback_active != last_fallback) || (fault_active != last_fault) || rsp_accept) begin
                actuator_update_pulse <= 1'b1;
                last_fallback <= fallback_active;
                last_fault <= fault_active;
                last_cmd <= safe_cmd;
            end
        end
    end
endmodule
