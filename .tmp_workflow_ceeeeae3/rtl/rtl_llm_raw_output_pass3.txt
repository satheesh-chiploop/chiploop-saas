module motor_control_register_map (
    input clk,
    input reset_n,
    input csr_valid,
    input csr_we,
    input csr_re,
    input [7:0] csr_be,
    input [7:0] csr_addr,
    input [63:0] csr_wdata,
    output reg [63:0] csr_rdata,
    output csr_ready,
    output reg [15:0] cfg_seq_id,
    output reg [15:0] cfg_timeout_ticks,
    output reg [15:0] cfg_freshness_limit,
    output reg [15:0] cfg_velocity_mps,
    output reg [31:0] cfg_geometry_base_addr,
    output reg [15:0] cfg_geometry_length,
    output reg [7:0] cfg_geometry_format,
    output reg [15:0] cfg_actuator_min,
    output reg [15:0] cfg_actuator_max,
    output reg [15:0] cfg_slew_limit,
    output reg [31:0] cfg_checkpoint_id,
    output reg [15:0] cfg_control_flags,
    output reg launch_pulse,
    output reg clear_fault_pulse,
    input status_busy,
    input status_done,
    input status_fault,
    input status_timeout,
    input status_stale,
    input status_clamped,
    input status_fallback_active,
    input status_transport_error,
    input status_response_valid,
    input [63:0] response_summary
);
assign csr_ready = 1'b1;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        cfg_seq_id <= 16'h0000;
        cfg_timeout_ticks <= 16'h0020;
        cfg_freshness_limit <= 16'h0010;
        cfg_velocity_mps <= 16'h0000;
        cfg_geometry_base_addr <= 32'h00000000;
        cfg_geometry_length <= 16'h0000;
        cfg_geometry_format <= 8'h00;
        cfg_actuator_min <= 16'h0000;
        cfg_actuator_max <= 16'h00FF;
        cfg_slew_limit <= 16'h0000;
        cfg_checkpoint_id <= 32'h00000000;
        cfg_control_flags <= 16'h0000;
        launch_pulse <= 1'b0;
        clear_fault_pulse <= 1'b0;
    end else begin
        launch_pulse <= 1'b0;
        clear_fault_pulse <= 1'b0;
        if (csr_valid && csr_we) begin
            case (csr_addr)
                8'h03: cfg_seq_id[7:0] <= csr_wdata[7:0];
                8'h04: cfg_seq_id[15:8] <= csr_wdata[7:0];
                8'h05: cfg_timeout_ticks[7:0] <= csr_wdata[7:0];
                8'h06: cfg_timeout_ticks[15:8] <= csr_wdata[7:0];
                8'h07: cfg_freshness_limit[7:0] <= csr_wdata[7:0];
                8'h08: cfg_freshness_limit[15:8] <= csr_wdata[7:0];
                8'h09: cfg_velocity_mps[7:0] <= csr_wdata[7:0];
                8'h0A: cfg_velocity_mps[15:8] <= csr_wdata[7:0];
                8'h0B: cfg_geometry_base_addr[7:0] <= csr_wdata[7:0];
                8'h0C: cfg_geometry_base_addr[15:8] <= csr_wdata[7:0];
                8'h0D: cfg_geometry_base_addr[23:16] <= csr_wdata[7:0];
                8'h0E: cfg_geometry_base_addr[31:24] <= csr_wdata[7:0];
                8'h0F: cfg_geometry_length[7:0] <= csr_wdata[7:0];
                8'h10: cfg_geometry_length[15:8] <= csr_wdata[7:0];
                8'h11: cfg_geometry_format <= csr_wdata[7:0];
                8'h12: cfg_actuator_min[7:0] <= csr_wdata[7:0];
                8'h13: cfg_actuator_min[15:8] <= csr_wdata[7:0];
                8'h14: cfg_actuator_max[7:0] <= csr_wdata[7:0];
                8'h15: cfg_actuator_max[15:8] <= csr_wdata[7:0];
                8'h16: cfg_slew_limit[7:0] <= csr_wdata[7:0];
                8'h17: cfg_slew_limit[15:8] <= csr_wdata[7:0];
                8'h18: cfg_checkpoint_id[7:0] <= csr_wdata[7:0];
                8'h19: cfg_checkpoint_id[15:8] <= csr_wdata[7:0];
                8'h1A: cfg_checkpoint_id[23:16] <= csr_wdata[7:0];
                8'h1B: cfg_checkpoint_id[31:24] <= csr_wdata[7:0];
                8'h1C: cfg_control_flags[7:0] <= csr_wdata[7:0];
                8'h1D: cfg_control_flags[15:8] <= csr_wdata[7:0];
                8'h02: clear_fault_pulse <= csr_wdata[0];
                8'h01: launch_pulse <= csr_wdata[0];
                default: begin
                end
            endcase
        end
    end
end

always @(*) begin
    csr_rdata = 64'h0000000000000000;
    case (csr_addr)
        8'h00: csr_rdata = {56'h0, 56'h0, status_busy};
        8'h01: csr_rdata = {56'h0, 56'h0, status_done};
        8'h02: csr_rdata = {56'h0, 56'h0, status_fault, status_timeout, status_stale, status_clamped, status_fallback_active, status_transport_error, status_response_valid};
        8'h03: csr_rdata = {56'h0, 56'h0, cfg_seq_id[7:0]};
        8'h04: csr_rdata = {56'h0, 56'h0, cfg_seq_id[15:8]};
        8'h05: csr_rdata = {56'h0, 56'h0, cfg_timeout_ticks[7:0]};
        8'h06: csr_rdata = {56'h0, 56'h0, cfg_timeout_ticks[15:8]};
        8'h07: csr_rdata = {56'h0, 56'h0, cfg_freshness_limit[7:0]};
        8'h08: csr_rdata = {56'h0, 56'h0, cfg_freshness_limit[15:8]};
        8'h09: csr_rdata = {56'h0, 56'h0, cfg_velocity_mps[7:0]};
        8'h0A: csr_rdata = {56'h0, 56'h0, cfg_velocity_mps[15:8]};
        8'h0B: csr_rdata = {56'h0, 56'h0, cfg_geometry_base_addr[7:0]};
        8'h0C: csr_rdata = {56'h0, 56'h0, cfg_geometry_base_addr[15:8]};
        8'h0D: csr_rdata = {56'h0, 56'h0, cfg_geometry_base_addr[23:16]};
        8'h0E: csr_rdata = {56'h0, 56'h0, cfg_geometry_base_addr[31:24]};
        8'h0F: csr_rdata = {56'h0, 56'h0, cfg_geometry_length[7:0]};
        8'h10: csr_rdata = {56'h0, 56'h0, cfg_geometry_length[15:8]};
        8'h11: csr_rdata = {56'h0, 56'h0, cfg_geometry_format};
        8'h12: csr_rdata = {56'h0, 56'h0, cfg_actuator_min[7:0]};
        8'h13: csr_rdata = {56'h0, 56'h0, cfg_actuator_min[15:8]};
        8'h14: csr_rdata = {56'h0, 56'h0, cfg_actuator_max[7:0]};
        8'h15: csr_rdata = {56'h0, 56'h0, cfg_actuator_max[15:8]};
        8'h16: csr_rdata = {56'h0, 56'h0, cfg_slew_limit[7:0]};
        8'h17: csr_rdata = {56'h0, 56'h0, cfg_slew_limit[15:8]};
        8'h18: csr_rdata = {56'h0, 56'h0, cfg_checkpoint_id[7:0]};
        8'h19: csr_rdata = {56'h0, 56'h0, cfg_checkpoint_id[15:8]};
        8'h1A: csr_rdata = {56'h0, 56'h0, cfg_checkpoint_id[23:16]};
        8'h1B: csr_rdata = {56'h0, 56'h0, cfg_checkpoint_id[31:24]};
        8'h1C: csr_rdata = {56'h0, 56'h0, cfg_control_flags[7:0]};
        8'h1D: csr_rdata = {56'h0, 56'h0, cfg_control_flags[15:8]};
        8'h20: csr_rdata = {56'h0, response_summary[7:0]};
        8'h21: csr_rdata = {56'h0, response_summary[15:8]};
        8'h22: csr_rdata = {56'h0, response_summary[23:16]};
        8'h23: csr_rdata = {56'h0, response_summary[31:24]};
        8'h24: csr_rdata = {56'h0, response_summary[39:32]};
        8'h25: csr_rdata = {56'h0, response_summary[47:40]};
        8'h26: csr_rdata = {56'h0, response_summary[55:48]};
        8'h27: csr_rdata = {56'h0, response_summary[63:56]};
        default: csr_rdata = 64'h0000000000000000;
    endcase
end

module payload_pointer_manager (
    input clk,
    input reset_n,
    input [31:0] cfg_geometry_base_addr,
    input [15:0] cfg_geometry_length,
    input [7:0] cfg_geometry_format,
    input [15:0] cfg_control_flags,
    output reg [31:0] geometry_base_addr_q,
    output reg [15:0] geometry_length_q,
    output reg [7:0] geometry_format_q,
    output reg geometry_valid,
    output reg alignment_error
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        geometry_base_addr_q <= 32'h00000000;
        geometry_length_q <= 16'h0000;
        geometry_format_q <= 8'h00;
        geometry_valid <= 1'b0;
        alignment_error <= 1'b0;
    end else begin
        geometry_base_addr_q <= cfg_geometry_base_addr;
        geometry_length_q <= cfg_geometry_length;
        geometry_format_q <= cfg_geometry_format;
        geometry_valid <= (cfg_geometry_length != 16'h0000) && (cfg_geometry_format != 8'h00) && (cfg_control_flags[0] | ~cfg_control_flags[0]);
        alignment_error <= (cfg_geometry_base_addr[1:0] != 2'b00) | (cfg_geometry_length == 16'h0000);
    end
end
endmodule

module request_packager (
    input clk,
    input reset_n,
    input start,
    input [15:0] seq_id_in,
    input [15:0] cfg_velocity_mps,
    input [31:0] cfg_checkpoint_id,
    input [15:0] cfg_control_flags,
    input [31:0] geometry_base_addr_q,
    input [15:0] geometry_length_q,
    input [7:0] geometry_format_q,
    input freshness_ok,
    input descriptor_valid,
    output reg [127:0] req_payload,
    output reg req_valid,
    input req_ready,
    output reg request_accepted,
    output reg request_error
);
reg pending;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        req_payload <= 128'h00000000000000000000000000000000;
        req_valid <= 1'b0;
        request_accepted <= 1'b0;
        request_error <= 1'b0;
        pending <= 1'b0;
    end else begin
        request_accepted <= 1'b0;
        request_error <= 1'b0;
        if (start && descriptor_valid && freshness_ok && !req_valid && !pending) begin
            req_payload <= {seq_id_in, geometry_base_addr_q, geometry_length_q, geometry_format_q, cfg_velocity_mps, 8'hA5, cfg_checkpoint_id[31:0], cfg_control_flags, 32'h00000000};
            req_valid <= 1'b1;
            pending <= 1'b1;
        end
        if (req_valid && req_ready) begin
            req_valid <= 1'b0;
            pending <= 1'b0;
            request_accepted <= 1'b1;
        end
        if (start && (!descriptor_valid || !freshness_ok)) begin
            request_error <= 1'b1;
        end
    end
end
endmodule

module transport_controller (
    input clk,
    input reset_n,
    input req_valid,
    output reg req_ready,
    input [127:0] req_payload,
    input rsp_valid,
    output reg rsp_ready,
    input [127:0] rsp_payload,
    input [15:0] timeout_ticks,
    output reg busy,
    output reg done,
    output reg timeout_fault,
    output reg transport_error,
    output reg response_valid,
    output reg [63:0] response_summary
);
reg [15:0] timer;
reg inflight;
reg [127:0] req_hold;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        req_ready <= 1'b1;
        rsp_ready <= 1'b0;
        busy <= 1'b0;
        done <= 1'b0;
        timeout_fault <= 1'b0;
        transport_error <= 1'b0;
        response_valid <= 1'b0;
        response_summary <= 64'h0000000000000000;
        timer <= 16'h0000;
        inflight <= 1'b0;
        req_hold <= 128'h0;
    end else begin
        done <= 1'b0;
        timeout_fault <= 1'b0;
        transport_error <= 1'b0;
        response_valid <= 1'b0;
        if (!inflight && req_valid) begin
            req_ready <= 1'b1;
            req_hold <= req_payload;
            inflight <= 1'b1;
            busy <= 1'b1;
            timer <= timeout_ticks;
            rsp_ready <= 1'b1;
        end else begin
            req_ready <= 1'b0;
        end
        if (inflight) begin
            if (rsp_valid && rsp_ready) begin
                response_valid <= 1'b1;
                response_summary <= rsp_payload[63:0] ^ req_hold[63:0];
                done <= 1'b1;
                busy <= 1'b0;
                inflight <= 1'b0;
                rsp_ready <= 1'b0;
            end else if (timer == 16'h0000) begin
                timeout_fault <= 1'b1;
                transport_error <= 1'b1;
                busy <= 1'b0;
                inflight <= 1'b0;
                rsp_ready <= 1'b0;
            end else begin
                timer <= timer - 16'h0001;
            end
        end
    end
end
endmodule

        last_accepted_seq <= 16'h0000;

        stale_fault <= 1'b0;
        timeout_config_valid <= 1'b0;
    end else begin
        timeout_config_valid <= (timeout_ticks != 16'h0000);
        stale_fault <= 1'b0;

            freshness_ok <= 1'b0;
            stale_fault <= 1'b1;
        end else begin
            freshness_ok <= 1'b1;
            last_accepted_seq <= seq_id_in;
        end
    end
end

module response_parser (
    input clk,
    input reset_n,
    input rsp_valid,
    input [127:0] rsp_payload,
    input [15:0] expected_seq_id,
    output reg response_ok,
    output reg [15:0] parsed_cmd_value,
    output reg [15:0] parsed_cmd_rate,
    output reg [15:0] parsed_flags,
    output reg response_valid,
    output reg malformed_response
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        response_ok <= 1'b0;
        parsed_cmd_value <= 16'h0000;
        parsed_cmd_rate <= 16'h0000;
        parsed_flags <= 16'h0000;
        response_valid <= 1'b0;
        malformed_response <= 1'b0;
    end else begin
        response_valid <= 1'b0;
        response_ok <= 1'b0;
        malformed_response <= 1'b0;
        if (rsp_valid && (rsp_payload[127:112] == expected_seq_id) && (rsp_payload[15:0] != 16'hFFFF)) begin
            parsed_cmd_value <= rsp_payload[31:16];
            parsed_cmd_rate <= rsp_payload[47:32];
            parsed_flags <= rsp_payload[63:48];
            response_ok <= 1'b1;
            response_valid <= 1'b1;
        end else if (rsp_valid) begin
            malformed_response <= 1'b1;
            response_valid <= 1'b1;
        end
    end
end
endmodule

module command_clamper (
    input clk,
    input reset_n,
    input response_ok,
    input [15:0] parsed_cmd_value,
    input [15:0] parsed_cmd_rate,
    input [15:0] actuator_min,
    input [15:0] actuator_max,
    input [15:0] slew_limit,
    output reg [15:0] cmd_value_out,
    output reg [15:0] cmd_rate_out,
    output reg clamped
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        cmd_value_out <= 16'h0000;
        cmd_rate_out <= 16'h0000;
        clamped <= 1'b0;
    end else begin
        clamped <= 1'b0;
        if (response_ok) begin
            cmd_value_out <= (parsed_cmd_value < actuator_min) ? actuator_min :
                             (parsed_cmd_value > actuator_max) ? actuator_max : parsed_cmd_value;
            cmd_rate_out <= (parsed_cmd_rate > slew_limit) ? slew_limit : parsed_cmd_rate;
            clamped <= (parsed_cmd_value < actuator_min) | (parsed_cmd_value > actuator_max) | (parsed_cmd_rate > slew_limit);
        end
    end
end
endmodule

---BEGIN motor_control_fsm.v---
module motor_control_fsm (
    input clk,
    input reset_n,
    input launch_pulse,
    input clear_fault_pulse,
    input freshness_ok,
    input descriptor_valid,
    input request_accepted,
    input request_error,
    input response_ok,
    input malformed_response,
    input timeout_fault,
    input stale_fault,
    input transport_error,
    input clamped,
    output reg busy,
    output reg done,
    output reg fault,
    output reg state_idle,
    output reg state_issue_req,
    output reg state_wait_rsp,
    output reg state_validate,
    output reg state_clamp,
    output reg state_apply,
    output reg state_fallback,
    output reg state_fault
);
localparam S_IDLE = 3'd0;
localparam S_ISSUE = 3'd1;
localparam S_WAIT = 3'd2;
localparam S_VALIDATE = 3'd3;
localparam S_CLAMP = 3'd4;
localparam S_APPLY = 3'd5;
localparam S_FALLBACK = 3'd6;
localparam S_FAULT = 3'd7;
reg [2:0] state;
reg [2:0] next_state;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) state <= S_IDLE;
    else state <= next_state;
end

always @(*) begin
    next_state = state;
    busy = 1'b0;
    done = 1'b0;
    fault = 1'b0;
    state_idle = 1'b0;
    state_issue_req = 1'b0;
    state_wait_rsp = 1'b0;
    state_validate = 1'b0;
    state_clamp = 1'b0;
    state_apply = 1'b0;
    state_fallback = 1'b0;
    state_fault = 1'b0;
    case (state)
        S_IDLE: begin
            state_idle = 1'b1;
            if (launch_pulse && freshness_ok && descriptor_valid) next_state = S_ISSUE;
            else if (launch_pulse && (!freshness_ok || !descriptor_valid)) next_state = S_FALLBACK;
        end
        S_ISSUE: begin
            busy = 1'b1;
            state_issue_req = 1'b1;
            if (request_error) next_state = S_FALLBACK;
            else if (request_accepted) next_state = S_WAIT;
        end
        S_WAIT: begin
            busy = 1'b1;
            state_wait_rsp = 1'b1;
            if (timeout_fault || stale_fault || transport_error) next_state = S_FAULT;
            else if (response_ok) next_state = S_VALIDATE;
            else if (malformed_response) next_state = S_FALLBACK;
        end
        S_VALIDATE: begin
            state_validate = 1'b1;
            if (response_ok) next_state = S_CLAMP;
            else next_state = S_FALLBACK;
        end
        S_CLAMP: begin
            state_clamp = 1'b1;
            if (clamped) next_state = S_FALLBACK;
            else next_state = S_APPLY;
        end
        S_APPLY: begin
            done = 1'b1;
            state_apply = 1'b1;
            next_state = S_IDLE;
        end
        S_FALLBACK: begin
            fault = 1'b1;
            state_fallback = 1'b1;
            if (clear_fault_pulse) next_state = S_IDLE;
        end
        S_FAULT: begin
            fault = 1'b1;
            state_fault = 1'b1;
            if (clear_fault_pulse) next_state = S_IDLE;
        end
        default: begin
            next_state = S_IDLE;
        end
    endcase
end
endmodule
---END motor_control_fsm.v---
