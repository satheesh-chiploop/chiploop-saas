---BEGIN control_status_regfile.v---
module control_status_regfile(
    input clk,
    input rst_n,
    input [7:0] reg_addr,
    input [31:0] reg_wdata,
    input reg_write,
    input reg_read,
    output reg [31:0] reg_rdata,
    output reg reg_ready,
    output reg control_enable,
    output reg [31:0] stream_timeout_cycles,
    output reg [31:0] min_actuator_cmd,
    output reg [31:0] max_actuator_cmd,
    output reg [31:0] safe_fallback_cmd,
    output reg [15:0] expected_seq,
    output reg model_ready,
    input fault_latched,
    output reg [15:0] last_seq_seen,
    output reg [7:0] last_error_code,
    output reg fault_clear,
    input fault_set,
    input [7:0] fault_code_in,
    input seq_update_valid,
    input [15:0] seq_update_value,
    input model_ready_in
);
    reg fault_set_d;
    reg seq_update_valid_d;
    reg model_ready_in_d;
    reg [7:0] fault_code_in_d;
    reg [15:0] seq_update_value_d;

    localparam [7:0] ADDR_CONTROL_ENABLE       = 8'h00;
    localparam [7:0] ADDR_STREAM_TIMEOUT_CYCLES = 8'h04;
    localparam [7:0] ADDR_MIN_ACTUATOR_CMD     = 8'h08;
    localparam [7:0] ADDR_MAX_ACTUATOR_CMD     = 8'h0C;
    localparam [7:0] ADDR_SAFE_FALLBACK_CMD    = 8'h10;
    localparam [7:0] ADDR_EXPECTED_SEQ         = 8'h14;
    localparam [7:0] ADDR_MODEL_READY          = 8'h18;
    localparam [7:0] ADDR_FAULT_LATCHED         = 8'h1C;
    localparam [7:0] ADDR_LAST_SEQ_SEEN         = 8'h20;
    localparam [7:0] ADDR_LAST_ERROR_CODE       = 8'h24;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            control_enable <= 1'b0;
            stream_timeout_cycles <= 32'h0000C350;
            min_actuator_cmd <= 32'h00000000;
            max_actuator_cmd <= 32'hFFFFFFFF;
            safe_fallback_cmd <= 32'h00000000;
            expected_seq <= 16'h0001;
            model_ready <= 1'b0;
            last_seq_seen <= 16'h0000;
            last_error_code <= 8'h00;
            fault_clear <= 1'b0;
            reg_rdata <= 32'h00000000;
            reg_ready <= 1'b0;
            fault_set_d <= 1'b0;
            seq_update_valid_d <= 1'b0;
            model_ready_in_d <= 1'b0;
            fault_code_in_d <= 8'h00;
            seq_update_value_d <= 16'h0000;
        end else begin
            fault_set_d <= fault_set;
            seq_update_valid_d <= seq_update_valid;
            model_ready_in_d <= model_ready_in;
            fault_code_in_d <= fault_code_in;
            seq_update_value_d <= seq_update_value;
            fault_clear <= 1'b0;
            reg_ready <= reg_write | reg_read;
            if (seq_update_valid_d) begin
                last_seq_seen <= seq_update_value_d;
            end
            model_ready <= model_ready_in_d;
            if (fault_set_d) begin
                last_error_code <= fault_code_in_d;
            end
            if (reg_write) begin
                case (reg_addr)
                    ADDR_CONTROL_ENABLE: control_enable <= reg_wdata[0];
                    ADDR_STREAM_TIMEOUT_CYCLES: stream_timeout_cycles <= reg_wdata;
                    ADDR_MIN_ACTUATOR_CMD: min_actuator_cmd <= reg_wdata;
                    ADDR_MAX_ACTUATOR_CMD: max_actuator_cmd <= reg_wdata;
                    ADDR_SAFE_FALLBACK_CMD: safe_fallback_cmd <= reg_wdata;
                    ADDR_EXPECTED_SEQ: expected_seq <= reg_wdata[15:0];
                    ADDR_LAST_ERROR_CODE: fault_clear <= reg_wdata[0];
                    default: begin
                    end
                endcase
            end
            if (reg_read) begin
                case (reg_addr)
                    ADDR_CONTROL_ENABLE: reg_rdata <= {31'b0, control_enable};
                    ADDR_STREAM_TIMEOUT_CYCLES: reg_rdata <= stream_timeout_cycles;
                    ADDR_MIN_ACTUATOR_CMD: reg_rdata <= min_actuator_cmd;
                    ADDR_MAX_ACTUATOR_CMD: reg_rdata <= max_actuator_cmd;
                    ADDR_SAFE_FALLBACK_CMD: reg_rdata <= safe_fallback_cmd;
                    ADDR_EXPECTED_SEQ: reg_rdata <= {16'b0, expected_seq};
                    ADDR_MODEL_READY: reg_rdata <= {31'b0, model_ready};
                    ADDR_FAULT_LATCHED: reg_rdata <= {31'b0, fault_latched};
                    ADDR_LAST_SEQ_SEEN: reg_rdata <= {16'b0, last_seq_seen};
                    ADDR_LAST_ERROR_CODE: reg_rdata <= {24'b0, last_error_code};
                    default: reg_rdata <= 32'h00000000;
                endcase
            end else begin
                reg_rdata <= 32'h00000000;
            end
        end
    end
endmodule
---END control_status_regfile.v---
---BEGIN request_packer.v---
module request_packer(
    input clk,
    input rst_n,
    input control_enable,
    input launch_req,
    input [7:0] protocol_version,
    input [7:0] command_type,
    input [15:0] seq_number,
    input [31:0] geometry_handle,
    input [15:0] stream_velocity_mps,
    input envelope_valid,
    input req_ready,
    output reg req_valid,
    output reg [127:0] req_packet,
    output reg packet_valid
);
    reg [127:0] req_packet_next;
    reg req_valid_next;
    reg packet_valid_next;
    reg [7:0] checksum;
    reg [127:0] packet_base;
    wire launch_ok;

    assign launch_ok = control_enable & launch_req & envelope_valid;

    always @(*) begin
        req_packet_next = req_packet;
        req_valid_next = 1'b0;
        packet_valid_next = packet_valid;
        packet_base = 128'h0;
        checksum = 8'h00;
        packet_base[127:120] = protocol_version;
        packet_base[119:112] = command_type;
        packet_base[111:96] = seq_number;
        packet_base[95:64] = geometry_handle;
        packet_base[63:48] = stream_velocity_mps;
        packet_base[47:8] = 40'h0000000000;
        checksum = packet_base[127:120] ^ packet_base[119:112] ^ packet_base[111:104] ^ packet_base[103:96] ^ packet_base[95:88] ^ packet_base[87:80] ^ packet_base[79:72] ^ packet_base[71:64] ^ packet_base[63:56] ^ packet_base[55:48] ^ packet_base[47:40] ^ packet_base[39:32] ^ packet_base[31:24] ^ packet_base[23:16] ^ packet_base[15:8];
        packet_base[7:0] = checksum;
        if (launch_ok && req_ready) begin
            req_packet_next = packet_base;
            req_valid_next = 1'b1;
            packet_valid_next = 1'b1;
        end else if (!control_enable || !envelope_valid) begin
            req_packet_next = 128'h00000000000000000000000000000000;
            req_valid_next = 1'b0;
            packet_valid_next = 1'b0;
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            req_valid <= 1'b0;
            req_packet <= 128'h00000000000000000000000000000000;
            packet_valid <= 1'b0;
        end else begin
            req_valid <= req_valid_next;
            req_packet <= req_packet_next;
            packet_valid <= packet_valid_next;
        end
    end
endmodule
---END request_packer.v---
---BEGIN response_unpacker.v---
module response_unpacker(
    input clk,
    input rst_n,
    input resp_valid,
    input [127:0] resp_packet,
    input [15:0] expected_seq,
    input timeout_active,
    input resp_ready,
    output reg resp_accepted,
    output reg resp_stale,
    output reg resp_malformed,
    output reg resp_seq_mismatch,
    output reg resp_integrity_ok,
    output reg [7:0] response_status,
    output reg [15:0] response_seq,
    output reg [31:0] command_intent
);
    reg [7:0] checksum_calc;
    reg [7:0] checksum_rx;
    reg header_ok;
    reg seq_ok;
    reg integrity_ok;
    reg valid_window;

    always @(*) begin
        resp_accepted = 1'b0;
        resp_stale = 1'b0;
        resp_malformed = 1'b0;
        resp_seq_mismatch = 1'b0;
        resp_integrity_ok = 1'b0;
        response_status = 8'h00;
        response_seq = 16'h0000;
        command_intent = 32'h00000000;
        checksum_calc = 8'h00;
        checksum_rx = resp_packet[7:0];
        header_ok = (resp_packet[127:120] == 8'hA5);
        seq_ok = (resp_packet[111:96] == expected_seq);
        valid_window = resp_valid & resp_ready;
        checksum_calc = resp_packet[127:120] ^ resp_packet[119:112] ^ resp_packet[111:104] ^ resp_packet[103:96] ^ resp_packet[95:88] ^ resp_packet[87:80] ^ resp_packet[79:72] ^ resp_packet[71:64] ^ resp_packet[63:56] ^ resp_packet[55:48] ^ resp_packet[47:40] ^ resp_packet[39:32] ^ resp_packet[31:24] ^ resp_packet[23:16] ^ resp_packet[15:8];
        integrity_ok = (checksum_calc == checksum_rx);
        response_seq = resp_packet[111:96];
        response_status = resp_packet[119:112];
        command_intent = resp_packet[95:64];
        resp_integrity_ok = integrity_ok;
        if (valid_window) begin
            if (!header_ok || !integrity_ok) begin
                resp_malformed = 1'b1;
            end else if (timeout_active) begin
                resp_stale = 1'b1;
            end else if (!seq_ok) begin
                resp_seq_mismatch = 1'b1;
            end else begin
                resp_accepted = 1'b1;
            end
        end
    end
endmodule
---END response_unpacker.v---
---BEGIN freshness_timeout_monitor.v---
module freshness_timeout_monitor(
    input clk,
    input rst_n,
    input control_enable,
    input [31:0] stream_timeout_cycles,
    input [15:0] expected_seq,
    input launch_req,
    input resp_accepted,
    input resp_stale,
    input resp_malformed,
    input resp_seq_mismatch,
    input envelope_valid,
    output reg safe_fallback_active,
    output reg timeout_active,
    output reg fault_set,
    input [7:0] fault_code,
    output reg seq_update_valid,
    output reg [15:0] seq_update_value
);
    reg [31:0] timeout_counter;
    reg request_outstanding;
    reg [7:0] fault_code_latched;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            timeout_counter <= 32'h00000000;
            request_outstanding <= 1'b0;
            safe_fallback_active <= 1'b1;
            timeout_active <= 1'b0;
            fault_set <= 1'b0;
            seq_update_valid <= 1'b0;
            seq_update_value <= 16'h0000;
            fault_code_latched <= 8'h00;
        end else begin
            fault_set <= 1'b0;
            seq_update_valid <= 1'b0;
            if (control_enable && launch_req && envelope_valid) begin
                request_outstanding <= 1'b1;
                timeout_counter <= 32'h00000000;
            end else if (request_outstanding) begin
                if (timeout_counter < stream_timeout_cycles) begin
                    timeout_counter <= timeout_counter + 32'h00000001;
                end else begin
                    timeout_active <= 1'b1;
                    fault_set <= 1'b1;
                    fault_code_latched <= 8'h01;
                    safe_fallback_active <= 1'b1;
                    request_outstanding <= 1'b0;
                end
            end
            if (resp_accepted) begin
                request_outstanding <= 1'b0;
                timeout_counter <= 32'h00000000;
                timeout_active <= 1'b0;
                seq_update_valid <= 1'b1;
                seq_update_value <= expected_seq;
                safe_fallback_active <= 1'b0;
            end
            if (resp_stale || resp_malformed || resp_seq_mismatch) begin
                fault_set <= 1'b1;
                fault_code_latched <= (resp_stale) ? 8'h02 : (resp_malformed ? 8'h03 : 8'h04);
                safe_fallback_active <= 1'b1;
                timeout_active <= 1'b1;
                request_outstanding <= 1'b0;
            end
            if (!control_enable || !envelope_valid) begin
                safe_fallback_active <= 1'b1;
            end
        end
    end
endmodule
---END freshness_timeout_monitor.v---
---BEGIN command_shaper.v---
module command_shaper(
    input clk,
    input rst_n,
    input [31:0] command_intent,
    input intent_valid,
    input [31:0] min_actuator_cmd,
    input [31:0] max_actuator_cmd,
    input [31:0] safe_fallback_cmd,
    input fault_latched,
    input safe_fallback_active,
    output reg [31:0] actuator_cmd,
    output reg actuator_cmd_valid,
    output reg clamp_event
);
    reg [31:0] clipped_cmd;
    reg use_fallback;
    reg clamp_needed;

    always @(*) begin
        clipped_cmd = command_intent;
        use_fallback = 1'b0;
        clamp_needed = 1'b0;
        actuator_cmd = safe_fallback_cmd;
        actuator_cmd_valid = 1'b0;
        clamp_event = 1'b0;
        if (fault_latched || safe_fallback_active || !intent_valid) begin
            use_fallback = 1'b1;
        end else begin
            if (command_intent < min_actuator_cmd) begin
                clipped_cmd = min_actuator_cmd;
                clamp_needed = 1'b1;
            end else if (command_intent > max_actuator_cmd) begin
                clipped_cmd = max_actuator_cmd;
                clamp_needed = 1'b1;
            end
        end
        if (use_fallback) begin
            actuator_cmd = safe_fallback_cmd;
            actuator_cmd_valid = 1'b1;
        end else begin
            actuator_cmd = clipped_cmd;
            actuator_cmd_valid = 1'b1;
            clamp_event = clamp_needed;
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            actuator_cmd <= 32'h00000000;
            actuator_cmd_valid <= 1'b0;
            clamp_event <= 1'b0;
        end
    end
endmodule
---END command_shaper.v---
---BEGIN fault_manager.v---
module fault_manager(
    input clk,
    input rst_n,
    input fault_set,
    input fault_clear,
    input [7:0] fault_code_in,
    input request_active,
    input request_permitted,
    input response_permitted,
    input timeout_active,
    input resp_malformed,
    input resp_seq_mismatch,
    output reg fault_latched,
    output reg [7:0] fault_code_out,
    output reg [15:0] fault_count
);
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            fault_latched <= 1'b0;
            fault_code_out <= 8'h00;
            fault_count <= 16'h0000;
        end else begin
            if (fault_clear) begin
                fault_latched <= 1'b0;
                fault_code_out <= 8'h00;
            end else if (fault_set) begin
                fault_latched <= 1'b1;
                fault_code_out <= fault_code_in;
                fault_count <= fault_count + 16'h0001;
            end
        end
    end
endmodule
---END fault_manager.v---
---BEGIN transaction_controller.v---
module transaction_controller(
    input clk,
    input rst_n,
    input control_enable,
    input safe_fallback_active,
    input fault_latched,
    input req_ready,
    input resp_valid,
    input resp_accepted,
    input resp_stale,
    input resp_malformed,
    input resp_seq_mismatch,
    output reg launch_req,
    output reg request_active,
    output reg request_permitted,
    output reg response_permitted,
    output reg resp_ready,
    output reg sequence_progress
);
    always @(*) begin
        launch_req = 1'b0;
        request_active = 1'b0;
        request_permitted = 1'b0;
        response_permitted = 1'b0;
        resp_ready = 1'b0;
        sequence_progress = 1'b0;
        if (control_enable && !safe_fallback_active && !fault_latched) begin
            request_permitted = 1'b1;
            resp_ready = 1'b1;
            if (req_ready) begin
                launch_req = 1'b1;
                request_active = 1'b1;
            end
            if (resp_valid && (resp_accepted || resp_stale || resp_malformed || resp_seq_mismatch)) begin
                response_permitted = 1'b1;
                sequence_progress = resp_accepted;
            end
        end
    end
endmodule
---END transaction_controller.v---
