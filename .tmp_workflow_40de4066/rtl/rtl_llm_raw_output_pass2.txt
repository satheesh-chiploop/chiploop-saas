);

input reset_n;

input cfg_bus_write;

input [63:0] cfg_bus_wdata;
output [63:0] cfg_bus_rdata;

input status_timeout_fault_in;

output [1:0] cfg_mode_select;
output [15:0] cfg_stream_velocity_setpoint;
output [15:0] cfg_actuator_min_clamp;
output [15:0] cfg_actuator_max_clamp;
output [15:0] cfg_request_timeout_cycles;
output [15:0] cfg_stale_threshold_cycles;
output [15:0] cfg_sequence_id;

reg [63:0] cfg_bus_rdata_r;
reg cfg_bus_ready_r;

reg [15:0] cfg_actuator_min_clamp_r;
reg [15:0] cfg_actuator_max_clamp_r;

reg [15:0] cfg_sequence_id_r;
reg [15:0] cfg_geometry_ref_id_r;
reg cfg_fault_ack_r;
reg [15:0] cfg_fallback_actuator_pos_r;
reg [15:0] status_shadow_r;
wire [15:0] status_bits_w;

assign status_bits_w = {
    status_timeout_fault_in,
    status_stale_fault_in,
    status_comm_fault_in,
    status_clamp_active_in,
    status_fallback_active_in,
    status_request_inflight_in,
    status_last_sequence_in[10:0]
};

assign cfg_bus_rdata = cfg_bus_rdata_r;
assign cfg_bus_ready = cfg_bus_ready_r;
assign cfg_enable = cfg_control_reg[0];
assign cfg_mode_select = cfg_control_reg[2:1];
assign cfg_stream_velocity_setpoint = cfg_stream_velocity_setpoint_r;
assign cfg_actuator_min_clamp = cfg_actuator_min_clamp_r;
assign cfg_actuator_max_clamp = cfg_actuator_max_clamp_r;
assign cfg_request_timeout_cycles = cfg_request_timeout_cycles_r;
assign cfg_stale_threshold_cycles = cfg_stale_threshold_cycles_r;
assign cfg_sequence_id = cfg_sequence_id_r;
assign cfg_geometry_ref_id = cfg_geometry_ref_id_r;
assign cfg_fault_ack = cfg_fault_ack_r;
assign cfg_fallback_actuator_pos = cfg_fallback_actuator_pos_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        cfg_control_reg <= 16'h0000;
        cfg_stream_velocity_setpoint_r <= 16'h0000;
        cfg_actuator_min_clamp_r <= 16'h0000;
        cfg_actuator_max_clamp_r <= 16'h00ff;
        cfg_request_timeout_cycles_r <= 16'd32;
        cfg_stale_threshold_cycles_r <= 16'd8;
        cfg_sequence_id_r <= 16'h0001;
        cfg_geometry_ref_id_r <= 16'h0000;
        cfg_fault_ack_r <= 1'b0;
        cfg_fallback_actuator_pos_r <= 16'h0000;
        status_shadow_r <= 16'h0000;
    end else begin
        cfg_fault_ack_r <= 1'b0;
        status_shadow_r <= status_bits_w;
        if (cfg_bus_valid && cfg_bus_write) begin
            case (cfg_bus_addr)
                8'h00: begin
                    cfg_control_reg <= cfg_bus_wdata[15:0];
                    cfg_fault_ack_r <= cfg_bus_wdata[0];
                end
                8'h08: cfg_stream_velocity_setpoint_r <= cfg_bus_wdata[15:0];
                8'h10: cfg_actuator_min_clamp_r <= cfg_bus_wdata[15:0];
                8'h18: cfg_actuator_max_clamp_r <= cfg_bus_wdata[15:0];
                8'h20: cfg_request_timeout_cycles_r <= cfg_bus_wdata[15:0];
                8'h28: cfg_stale_threshold_cycles_r <= cfg_bus_wdata[15:0];
                8'h30: cfg_sequence_id_r <= cfg_bus_wdata[15:0];
                8'h38: cfg_geometry_ref_id_r <= cfg_bus_wdata[15:0];
                8'h40: cfg_fallback_actuator_pos_r <= cfg_bus_wdata[15:0];
                default: begin end
            endcase
        end
    end
end

always @(*) begin
    cfg_bus_rdata_r = 64'h0000000000000000;
    cfg_bus_ready_r = 1'b1;
    case (cfg_bus_addr)
        8'h00: cfg_bus_rdata_r = {48'h000000000000, cfg_control_reg};
        8'h08: cfg_bus_rdata_r = {48'h000000000000, cfg_stream_velocity_setpoint_r};
        8'h10: cfg_bus_rdata_r = {48'h000000000000, cfg_actuator_min_clamp_r};
        8'h18: cfg_bus_rdata_r = {48'h000000000000, cfg_actuator_max_clamp_r};
        8'h20: cfg_bus_rdata_r = {48'h000000000000, cfg_request_timeout_cycles_r};
        8'h28: cfg_bus_rdata_r = {48'h000000000000, cfg_stale_threshold_cycles_r};
        8'h30: cfg_bus_rdata_r = {48'h000000000000, cfg_sequence_id_r};
        8'h38: cfg_bus_rdata_r = {48'h000000000000, cfg_geometry_ref_id_r};
        8'h40: cfg_bus_rdata_r = {48'h000000000000, cfg_fallback_actuator_pos_r};
        8'h48: cfg_bus_rdata_r = {48'h000000000000, status_shadow_r};
        default: cfg_bus_rdata_r = 64'h0000000000000000;
    endcase
end
endmodule

module adaptive_aero_request_packager (
    clk,
    reset_n,
    cfg_enable,
    cfg_mode_select,
    cfg_stream_velocity_setpoint,
    cfg_sequence_id,
    cfg_geometry_ref_id,
    request_valid,
    request_ready,
    request_data,
    request_payload
);
input clk;
input reset_n;
input cfg_enable;
input [1:0] cfg_mode_select;
input [15:0] cfg_stream_velocity_setpoint;
input [15:0] cfg_sequence_id;
input [15:0] cfg_geometry_ref_id;
output request_valid;
input request_ready;
output [127:0] request_data;
output [127:0] request_payload;

reg request_valid_r;
reg [127:0] request_data_r;
reg [127:0] request_payload_r;
reg inflight_r;

assign request_valid = request_valid_r;
assign request_data = request_data_r;
assign request_payload = request_payload_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        request_valid_r <= 1'b0;
        request_data_r <= 128'h00000000000000000000000000000000;
        request_payload_r <= 128'h00000000000000000000000000000000;
        inflight_r <= 1'b0;
    end else begin
        if (cfg_enable && !inflight_r) begin
            request_valid_r <= 1'b1;
            request_data_r <= {cfg_sequence_id, cfg_geometry_ref_id, cfg_mode_select, cfg_stream_velocity_setpoint, 96'h000000000000000000000000};
            request_payload_r <= {cfg_sequence_id, cfg_geometry_ref_id, cfg_mode_select, cfg_stream_velocity_setpoint, 96'h000000000000000000000000};
        end else if (request_valid_r && request_ready) begin
            request_valid_r <= 1'b0;
        end
        if (request_valid_r && request_ready) begin
            inflight_r <= 1'b1;
        end else if (!cfg_enable) begin
            inflight_r <= 1'b0;
        end
    end
end
endmodule

module adaptive_aero_response_monitor (
    clk,
    reset_n,
    cfg_fault_ack,
    cfg_request_timeout_cycles,
    cfg_stale_threshold_cycles,
    model_req_valid,
    model_req_ready,
    model_rsp_valid,
    model_rsp_ready,
    model_rsp_data,
    response_recommendation,
    response_valid_sample,
    status_timeout_fault,
    status_stale_fault,
    status_comm_fault,
    status_request_inflight,
    status_last_sequence,
    response_payload
);
input clk;
input reset_n;
input cfg_fault_ack;
input [15:0] cfg_request_timeout_cycles;
input [15:0] cfg_stale_threshold_cycles;
input model_req_valid;
input model_req_ready;
input model_rsp_valid;
output model_rsp_ready;
input [127:0] model_rsp_data;
output [1:0] response_recommendation;
output response_valid_sample;
output status_timeout_fault;
output status_stale_fault;
output status_comm_fault;
output status_request_inflight;
output [15:0] status_last_sequence;
output [127:0] response_payload;

reg model_rsp_ready_r;
reg [1:0] response_recommendation_r;
reg response_valid_sample_r;
reg status_timeout_fault_r;
reg status_stale_fault_r;
reg status_comm_fault_r;
reg status_request_inflight_r;
reg [15:0] status_last_sequence_r;
reg [127:0] response_payload_r;
reg [15:0] timeout_count_r;
reg [15:0] stale_count_r;

assign model_rsp_ready = model_rsp_ready_r;
assign response_recommendation = response_recommendation_r;
assign response_valid_sample = response_valid_sample_r;
assign status_timeout_fault = status_timeout_fault_r;
assign status_stale_fault = status_stale_fault_r;
assign status_comm_fault = status_comm_fault_r;
assign status_request_inflight = status_request_inflight_r;
assign status_last_sequence = status_last_sequence_r;
assign response_payload = response_payload_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        model_rsp_ready_r <= 1'b1;
        response_recommendation_r <= 2'b00;
        response_valid_sample_r <= 1'b0;
        status_timeout_fault_r <= 1'b0;
        status_stale_fault_r <= 1'b0;
        status_comm_fault_r <= 1'b0;
        status_request_inflight_r <= 1'b0;
        status_last_sequence_r <= 16'h0000;
        response_payload_r <= 128'h00000000000000000000000000000000;
        timeout_count_r <= 16'h0000;
        stale_count_r <= 16'h0000;
    end else begin
        response_valid_sample_r <= 1'b0;
        if (cfg_fault_ack) begin
            status_timeout_fault_r <= 1'b0;
            status_stale_fault_r <= 1'b0;
            status_comm_fault_r <= 1'b0;
        end
        if (model_req_valid && model_req_ready) begin
            status_request_inflight_r <= 1'b1;
            timeout_count_r <= cfg_request_timeout_cycles;
            stale_count_r <= cfg_stale_threshold_cycles;
        end
        if (status_request_inflight_r) begin
            if (model_rsp_valid) begin
                response_valid_sample_r <= 1'b1;
                response_payload_r <= model_rsp_data;
                status_last_sequence_r <= model_rsp_data[15:0];
                response_recommendation_r <= model_rsp_data[17:16];
                status_request_inflight_r <= 1'b0;
                timeout_count_r <= 16'h0000;
                stale_count_r <= 16'h0000;
            end else begin
                if (timeout_count_r != 16'h0000) timeout_count_r <= timeout_count_r - 16'h0001;
                if (stale_count_r != 16'h0000) stale_count_r <= stale_count_r - 16'h0001;
                if (timeout_count_r == 16'h0001) status_timeout_fault_r <= 1'b1;
                if (stale_count_r == 16'h0001) status_stale_fault_r <= 1'b1;
            end
        end
        status_comm_fault_r <= status_timeout_fault_r | status_stale_fault_r;
    end
end
endmodule

module adaptive_aero_safety_guard (
    clk,
    reset_n,
    cfg_enable,
    cfg_mode_select,
    cfg_actuator_min_clamp,
    cfg_actuator_max_clamp,
    cfg_fallback_actuator_pos,
    response_recommendation,
    response_valid_sample,
    status_timeout_fault,
    status_stale_fault,
    status_comm_fault,
    status_clamp_active,
    status_fallback_active,
    actuator_cmd_valid,
    actuator_cmd_data
);
input clk;
input reset_n;
input cfg_enable;
input [1:0] cfg_mode_select;
input [15:0] cfg_actuator_min_clamp;
input [15:0] cfg_actuator_max_clamp;
input [15:0] cfg_fallback_actuator_pos;
input [1:0] response_recommendation;
input response_valid_sample;
input status_timeout_fault;
input status_stale_fault;
input status_comm_fault;
output status_clamp_active;
output status_fallback_active;
output actuator_cmd_valid;
output [15:0] actuator_cmd_data;

reg status_clamp_active_r;
reg status_fallback_active_r;
reg actuator_cmd_valid_r;
reg [15:0] actuator_cmd_data_r;
reg [15:0] selected_cmd_r;

assign status_clamp_active = status_clamp_active_r;
assign status_fallback_active = status_fallback_active_r;
assign actuator_cmd_valid = actuator_cmd_valid_r;
assign actuator_cmd_data = actuator_cmd_data_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        status_clamp_active_r <= 1'b0;
        status_fallback_active_r <= 1'b0;
        actuator_cmd_valid_r <= 1'b0;
        actuator_cmd_data_r <= 16'h0000;
        selected_cmd_r <= 16'h0000;
    end else begin
        if (response_valid_sample && cfg_enable) begin
            case (response_recommendation)
                2'b00: selected_cmd_r <= cfg_fallback_actuator_pos;
                2'b01: selected_cmd_r <= cfg_actuator_min_clamp;
                2'b10: selected_cmd_r <= cfg_actuator_max_clamp;
                default: selected_cmd_r <= cfg_fallback_actuator_pos;
            endcase
            actuator_cmd_valid_r <= 1'b1;
            status_fallback_active_r <= (response_recommendation == 2'b00);
            status_clamp_active_r <= (response_recommendation == 2'b01) || (response_recommendation == 2'b10);
        end else begin
            actuator_cmd_valid_r <= 1'b0;
        end
        if (status_timeout_fault || status_stale_fault || status_comm_fault || !cfg_enable) begin
            selected_cmd_r <= cfg_fallback_actuator_pos;
            status_fallback_active_r <= 1'b1;
            status_clamp_active_r <= 1'b0;
        end
        if (status_timeout_fault || status_stale_fault || status_comm_fault) begin
            actuator_cmd_data_r <= cfg_fallback_actuator_pos;
        end else begin
            if (response_valid_sample && cfg_enable) begin
                if (response_recommendation == 2'b01) begin
                    if (cfg_fallback_actuator_pos < cfg_actuator_min_clamp) actuator_cmd_data_r <= cfg_actuator_min_clamp;
                    else if (cfg_fallback_actuator_pos > cfg_actuator_max_clamp) actuator_cmd_data_r <= cfg_actuator_max_clamp;
                    else actuator_cmd_data_r <= cfg_fallback_actuator_pos;
                end else if (response_recommendation == 2'b10) begin
                    if (cfg_fallback_actuator_pos < cfg_actuator_min_clamp) actuator_cmd_data_r <= cfg_actuator_min_clamp;
                    else if (cfg_fallback_actuator_pos > cfg_actuator_max_clamp) actuator_cmd_data_r <= cfg_actuator_max_clamp;
                    else actuator_cmd_data_r <= cfg_fallback_actuator_pos;
                end else begin
                    actuator_cmd_data_r <= cfg_fallback_actuator_pos;
                end
            end
        end
    end
end
endmodule
