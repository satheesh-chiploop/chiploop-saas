---BEGIN adaptive_aero_control_top.v---
module adaptive_aero_control_top (
    clk,
    rst_n,
    cfg_wr_en,
    cfg_rd_en,
    cfg_addr,
    cfg_wdata,
    cfg_rdata,
    cfg_ready,
    req_stream_valid,
    req_stream_ready,
    req_stream_data,
    rsp_stream_valid,
    rsp_stream_ready,
    rsp_stream_data,
    actuator_cmd_valid,
    actuator_cmd_ready,
    actuator_cmd_data,
    telemetry_out_valid,
    telemetry_out_ready,
    telemetry_out_data
);

input clk;
input rst_n;
input cfg_wr_en;
input cfg_rd_en;
input [7:0] cfg_addr;
input [31:0] cfg_wdata;
output [31:0] cfg_rdata;
output cfg_ready;
output req_stream_valid;
input req_stream_ready;
output [127:0] req_stream_data;
input rsp_stream_valid;
output rsp_stream_ready;
input [127:0] rsp_stream_data;
output actuator_cmd_valid;
input actuator_cmd_ready;
output [63:0] actuator_cmd_data;
output telemetry_out_valid;
input telemetry_out_ready;
output [31:0] telemetry_out_data;

reg [31:0] ctrl_reg;
reg [31:0] seq_reg;
reg [31:0] timeout_reg;
reg [31:0] setpoint_reg;
reg [31:0] cmd_min_reg;
reg [31:0] cmd_max_reg;
reg [31:0] slew_reg;

reg [31:0] current_cmd_reg;
reg [31:0] outstanding_seq_reg;
reg [31:0] age_counter_reg;
reg [31:0] fault_code_reg;
reg stale_flag_reg;
reg timeout_flag_reg;
reg clamp_flag_reg;
reg fallback_flag_reg;
reg request_pending_reg;
reg response_pending_reg;
reg configured_reg;
reg [31:0] telemetry_reg;
reg [127:0] req_payload_reg;
reg [127:0] rsp_payload_reg;
reg [63:0] actuator_cmd_reg;
reg [31:0] cfg_rdata_reg;
reg cfg_ready_reg;
reg req_stream_valid_reg;
reg rsp_stream_ready_reg;
reg actuator_cmd_valid_reg;
reg telemetry_out_valid_reg;

reg [31:0] clamped_cmd_reg;
reg [31:0] slew_cmd_reg;
reg [31:0] next_cmd_calc_reg;
reg [31:0] rsp_seq_reg;
reg [31:0] rsp_cmd_reg;
reg [31:0] rsp_age_reg;
reg [31:0] rsp_valid_tag_reg;
reg [31:0] fallback_cmd_reg;

wire [127:0] u_resp_stage_sram_dout;
reg [3:0] u_resp_stage_sram_addr;
reg [127:0] u_resp_stage_sram_din;
reg u_resp_stage_sram_csb;
reg u_resp_stage_sram_web;

adaptive_aero_resp_stage_sram u_resp_stage_sram (
    .clk(clk),
    .csb(u_resp_stage_sram_csb),
    .web(u_resp_stage_sram_web),
    .addr(u_resp_stage_sram_addr),
    .din(u_resp_stage_sram_din),
    .dout(u_resp_stage_sram_dout)
);

assign cfg_rdata = cfg_rdata_reg;
assign cfg_ready = cfg_ready_reg;
assign req_stream_valid = req_stream_valid_reg;
assign req_stream_data = req_payload_reg;
assign rsp_stream_ready = rsp_stream_ready_reg;
assign actuator_cmd_valid = actuator_cmd_valid_reg;
assign actuator_cmd_data = actuator_cmd_reg;
assign telemetry_out_valid = telemetry_out_valid_reg;
assign telemetry_out_data = telemetry_reg;

always @(*) begin
    cfg_rdata_reg = 32'h00000000;
    cfg_ready_reg = 1'b1;
    req_stream_valid_reg = 1'b0;
    req_payload_reg = 128'h00000000000000000000000000000000;
    rsp_stream_ready_reg = 1'b0;
    actuator_cmd_valid_reg = 1'b0;
    actuator_cmd_reg = 64'h0000000000000000;
    telemetry_out_valid_reg = 1'b0;
    telemetry_reg = 32'h00000000;
    clamped_cmd_reg = 32'h00000000;
    slew_cmd_reg = 32'h00000000;
    next_cmd_calc_reg = 32'h00000000;
    rsp_seq_reg = 32'h00000000;
    rsp_cmd_reg = 32'h00000000;
    rsp_age_reg = 32'h00000000;
    rsp_valid_tag_reg = 32'h00000000;
    fallback_cmd_reg = 32'h00000000;
    u_resp_stage_sram_addr = 4'h0;
    u_resp_stage_sram_din = 128'h00000000000000000000000000000000;
    u_resp_stage_sram_csb = 1'b1;
    u_resp_stage_sram_web = 1'b1;

    rsp_seq_reg = rsp_stream_data[31:0];
    rsp_cmd_reg = rsp_stream_data[63:32];
    rsp_age_reg = rsp_stream_data[95:64];
    rsp_valid_tag_reg = rsp_stream_data[127:96];

    if (ctrl_reg[0] && configured_reg) begin
        req_stream_valid_reg = request_pending_reg;
        req_payload_reg = {ctrl_reg[7:0], seq_reg, setpoint_reg, current_cmd_reg};
        rsp_stream_ready_reg = request_pending_reg;
    end

    if (cfg_rd_en) begin
        case (cfg_addr)
            8'h00: cfg_rdata_reg = ctrl_reg;
            8'h04: cfg_rdata_reg = telemetry_reg;
            8'h08: cfg_rdata_reg = seq_reg;
            8'h0C: cfg_rdata_reg = timeout_reg;
            8'h10: cfg_rdata_reg = setpoint_reg;
            8'h14: cfg_rdata_reg = cmd_min_reg;
            8'h18: cfg_rdata_reg = cmd_max_reg;
            8'h1C: cfg_rdata_reg = slew_reg;
            default: cfg_rdata_reg = 32'h00000000;
        endcase
    end

    if (configured_reg && ctrl_reg[0] && request_pending_reg && req_stream_ready) begin
        req_stream_valid_reg = 1'b1;
    end

    if (configured_reg && ctrl_reg[0] && rsp_stream_valid && rsp_stream_ready_reg && rsp_valid_tag_reg[0]) begin
        if (rsp_seq_reg == outstanding_seq_reg && rsp_age_reg <= timeout_reg) begin
            next_cmd_calc_reg = rsp_cmd_reg;
        end else begin
            next_cmd_calc_reg = current_cmd_reg;
        end
    end else begin
        next_cmd_calc_reg = current_cmd_reg;
    end

    if (next_cmd_calc_reg < cmd_min_reg) begin
        clamped_cmd_reg = cmd_min_reg;
    end else if (next_cmd_calc_reg > cmd_max_reg) begin
        clamped_cmd_reg = cmd_max_reg;
    end else begin
        clamped_cmd_reg = next_cmd_calc_reg;
    end

    if (ctrl_reg[1] && slew_reg[31]) begin
        if (clamped_cmd_reg > current_cmd_reg) begin
            if ((clamped_cmd_reg - current_cmd_reg) > slew_reg[15:0]) begin
                slew_cmd_reg = current_cmd_reg + slew_reg[15:0];
            end else begin
                slew_cmd_reg = clamped_cmd_reg;
            end
        end else begin
            if ((current_cmd_reg - clamped_cmd_reg) > slew_reg[15:0]) begin
                slew_cmd_reg = current_cmd_reg - slew_reg[15:0];
            end else begin
                slew_cmd_reg = clamped_cmd_reg;
            end
        end
    end else begin
        slew_cmd_reg = clamped_cmd_reg;
    end

    if (!ctrl_reg[0] || !configured_reg) begin
        fallback_cmd_reg = cmd_min_reg;
    end else if (timeout_flag_reg || stale_flag_reg || fault_code_reg != 32'h00000000) begin
        fallback_cmd_reg = cmd_min_reg;
    end else begin
        fallback_cmd_reg = slew_cmd_reg;
    end

    actuator_cmd_valid_reg = configured_reg && ctrl_reg[0];
    actuator_cmd_reg = {32'h00000000, fallback_cmd_reg};
    telemetry_reg = {fallback_flag_reg, clamp_flag_reg, timeout_flag_reg, stale_flag_reg, ctrl_reg[3:0], fault_code_reg[22:0]};
    telemetry_out_valid_reg = configured_reg;
end

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        ctrl_reg <= 32'h00000000;
        seq_reg <= 32'h00000000;
        timeout_reg <= 32'h00000100;
        setpoint_reg <= 32'h00000000;
        cmd_min_reg <= 32'h00000000;
        cmd_max_reg <= 32'h0000FFFF;
        slew_reg <= 32'h00000000;
        current_cmd_reg <= 32'h00000000;
        outstanding_seq_reg <= 32'h00000000;
        age_counter_reg <= 32'h00000000;
        fault_code_reg <= 32'h00000000;
        stale_flag_reg <= 1'b0;
        timeout_flag_reg <= 1'b0;
        clamp_flag_reg <= 1'b0;
        fallback_flag_reg <= 1'b0;
        request_pending_reg <= 1'b0;
        response_pending_reg <= 1'b0;
        configured_reg <= 1'b0;
        req_payload_reg <= 128'h00000000000000000000000000000000;
        rsp_payload_reg <= 128'h00000000000000000000000000000000;
        actuator_cmd_reg <= 64'h0000000000000000;
        telemetry_reg <= 32'h00000000;
        cfg_rdata_reg <= 32'h00000000;
        cfg_ready_reg <= 1'b1;
        req_stream_valid_reg <= 1'b0;
        rsp_stream_ready_reg <= 1'b0;
        actuator_cmd_valid_reg <= 1'b0;
        telemetry_out_valid_reg <= 1'b0;
        u_resp_stage_sram_addr <= 4'h0;
        u_resp_stage_sram_din <= 128'h00000000000000000000000000000000;
        u_resp_stage_sram_csb <= 1'b1;
        u_resp_stage_sram_web <= 1'b1;
    end else begin
        if (cfg_wr_en) begin
            case (cfg_addr)
                8'h00: begin
                    ctrl_reg <= cfg_wdata;
                    configured_reg <= cfg_wdata[0];
                    if (cfg_wdata[2]) begin
                        fault_code_reg <= 32'h00000000;
                        stale_flag_reg <= 1'b0;
                        timeout_flag_reg <= 1'b0;
                        clamp_flag_reg <= 1'b0;
                        fallback_flag_reg <= 1'b0;
                    end
                end
                8'h08: seq_reg <= cfg_wdata;
                8'h0C: timeout_reg <= cfg_wdata;
                8'h10: setpoint_reg <= cfg_wdata;
                8'h14: cmd_min_reg <= cfg_wdata;
                8'h18: cmd_max_reg <= cfg_wdata;
                8'h1C: slew_reg <= cfg_wdata;
                default: begin
                end
            endcase
        end

        if (ctrl_reg[0] && configured_reg) begin
            if (request_pending_reg && req_stream_ready) begin
                outstanding_seq_reg <= seq_reg;
                age_counter_reg <= 32'h00000000;
                request_pending_reg <= 1'b0;
            end else if (!request_pending_reg) begin
                request_pending_reg <= 1'b1;
                req_payload_reg <= {ctrl_reg[7:0], seq_reg, setpoint_reg, current_cmd_reg};
                outstanding_seq_reg <= seq_reg;
                age_counter_reg <= 32'h00000000;
            end

            if (rsp_stream_valid) begin
                rsp_payload_reg <= rsp_stream_data;
                response_pending_reg <= 1'b1;
                u_resp_stage_sram_csb <= 1'b0;
                u_resp_stage_sram_web <= 1'b0;
                u_resp_stage_sram_addr <= seq_reg[3:0];
                u_resp_stage_sram_din <= rsp_stream_data;
            end

            if (response_pending_reg) begin
                if (rsp_stream_valid && rsp_payload_reg[127:96][0]) begin
                    if (rsp_payload_reg[31:0] != outstanding_seq_reg) begin
                        stale_flag_reg <= 1'b1;
                        fault_code_reg <= 32'h00000001;
                    end else if (age_counter_reg > timeout_reg) begin
                        timeout_flag_reg <= 1'b1;
                        fault_code_reg <= 32'h00000002;
                    end else if ((rsp_payload_reg[63:32] < cmd_min_reg) || (rsp_payload_reg[63:32] > cmd_max_reg)) begin
                        clamp_flag_reg <= 1'b1;
                        fault_code_reg <= 32'h00000004;
                    end else begin
                        current_cmd_reg <= rsp_payload_reg[63:32];
                        clamp_flag_reg <= 1'b0;
                        fallback_flag_reg <= 1'b0;
                    end
                    response_pending_reg <= 1'b0;
                end
            end

            if (age_counter_reg != 32'hFFFFFFFF) begin
                age_counter_reg <= age_counter_reg + 32'h00000001;
            end

            if (ctrl_reg[1] && slew_reg[31] && (current_cmd_reg != 32'h00000000)) begin
                if (current_cmd_reg > cmd_max_reg)
                    current_cmd_reg <= cmd_max_reg;
                else if (current_cmd_reg < cmd_min_reg)
                    current_cmd_reg <= cmd_min_reg;
            end

            if (timeout_reg != 32'h00000000 && age_counter_reg > timeout_reg) begin
                timeout_flag_reg <= 1'b1;
                fault_code_reg <= 32'h00000002;
                fallback_flag_reg <= 1'b1;
            end
        end else begin
            request_pending_reg <= 1'b0;
            response_pending_reg <= 1'b0;
            age_counter_reg <= 32'h00000000;
            current_cmd_reg <= cmd_min_reg;
            fallback_flag_reg <= 1'b1;
        end

        if (ctrl_reg[2]) begin
            fault_code_reg <= 32'h00000000;
            stale_flag_reg <= 1'b0;
            timeout_flag_reg <= 1'b0;
            clamp_flag_reg <= 1'b0;
            fallback_flag_reg <= 1'b0;
        end
    end
end

endmodule
---END adaptive_aero_control_top.v---
