---BEGIN adaptive_aero_control_bram_if.v---
module adaptive_aero_control_bram_if (
    clk,
    rst_n,
    wr_en,
    wr_addr,
    wr_data,
    rd_en,
    rd_addr,
    rd_data
);
    input clk;
    input rst_n;
    input wr_en;
    input [5:0] wr_addr;
    input [63:0] wr_data;
    input rd_en;
    input [5:0] rd_addr;
    output [63:0] rd_data;

    reg [63:0] mem [0:63];
    reg [63:0] rd_data_r;
    integer i;

    assign rd_data = rd_data_r;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            rd_data_r <= 64'h0000000000000000;
            for (i = 0; i < 64; i = i + 1) begin
                mem[i] <= 64'h0000000000000000;
            end
        end else begin
            if (wr_en) begin
                mem[wr_addr] <= wr_data;
            end
            if (rd_en) begin
                rd_data_r <= mem[rd_addr];
            end
        end
    end
endmodule
---END adaptive_aero_control_bram_if.v---

---BEGIN adaptive_aero_control_top.v---
module adaptive_aero_control_top (
    clk,
    rst_n,
    cmd_valid,
    cmd_ready,
    cmd_data,
    rsp_valid,
    rsp_ready,
    rsp_data,
    actuator_valid,
    actuator_fault,
    actuator_fallback,
    actuator_clamp,
    actuator_mode,
    actuator_seq_echo,
    actuator_command,
    reg_addr,
    reg_wdata,
    reg_rdata,
    reg_valid,
    reg_write,
    reg_ready
);
    input clk;
    input rst_n;
    input cmd_valid;
    output cmd_ready;
    input [63:0] cmd_data;
    output rsp_valid;
    input rsp_ready;
    output [63:0] rsp_data;
    output actuator_valid;
    output actuator_fault;
    output actuator_fallback;
    output actuator_clamp;
    output [2:0] actuator_mode;
    output [15:0] actuator_seq_echo;
    output [15:0] actuator_command;
    input [7:0] reg_addr;
    input [63:0] reg_wdata;
    output [63:0] reg_rdata;
    input reg_valid;
    input reg_write;
    output reg_ready;

    wire [63:0] bram_rd_data;
    wire bram_wr_en;
    wire [5:0] bram_wr_addr;
    wire [63:0] bram_wr_data;
    wire bram_rd_en;
    wire [5:0] bram_rd_addr;

    reg [63:0] control_reg;
    reg [63:0] timeout_cfg_reg;
    reg [63:0] limits_min_reg;
    reg [63:0] limits_max_reg;
    reg [63:0] fallback_cfg_reg;
    reg [7:0] status0_reg;
    reg [63:0] status1_reg;
    reg [63:0] status2_reg;
    reg [63:0] version_reg;

    reg [15:0] last_accepted_seq;
    reg [7:0] last_fault_code;
    reg [15:0] last_rejected_seq;
    reg [15:0] last_sanitized_cmd;
    reg [15:0] last_applied_cmd;
    reg [15:0] seq_echo_reg;
    reg [15:0] current_cmd;
    reg [15:0] current_sanitized;
    reg [15:0] current_applied;
    reg [15:0] safe_last_cmd;
    reg [15:0] last_seq_seen;
    reg [15:0] freshness_age;
    reg [15:0] timeout_age;
    reg [15:0] pending_seq;
    reg [15:0] pending_cmd;
    reg [15:0] pending_velocity;
    reg [3:0] pending_opcode;
    reg [1:0] pending_mode;
    reg pending_valid;
    reg pending_clamp;
    reg fault_latched;
    reg timeout_active;
    reg fallback_active;
    reg clamp_active;
    reg valid_active;
    reg transport_busy;
    reg rsp_valid_r;
    reg [63:0] rsp_data_r;
    reg [15:0] actuator_command_r;
    reg [2:0] actuator_mode_r;
    reg actuator_fault_r;
    reg actuator_fallback_r;
    reg actuator_clamp_r;
    reg actuator_valid_r;
    reg [2:0] state_reg;
    reg [2:0] next_state;
    reg reg_ready_r;
    reg [63:0] reg_rdata_r;

    localparam [2:0] ST_IDLE = 3'd0;
    localparam [2:0] ST_RECEIVE = 3'd1;
    localparam [2:0] ST_VALIDATE = 3'd2;
    localparam [2:0] ST_APPLY = 3'd3;
    localparam [2:0] ST_REPORT = 3'd4;
    localparam [2:0] ST_ERROR = 3'd5;

    localparam [7:0] FAULT_NONE = 8'h00;
    localparam [7:0] FAULT_INVALID_PACKET = 8'h01;
    localparam [7:0] FAULT_VELOCITY_RANGE = 8'h02;
    localparam [7:0] FAULT_STALE = 8'h03;
    localparam [7:0] FAULT_TIMEOUT = 8'h04;
    localparam [7:0] FAULT_TRANSPORT = 8'h05;

    localparam [31:0] CLEAR_KEY = 32'hA5C35A1C;
    localparam [31:0] IP_VERSION = 32'h00000001;
    localparam [31:0] BUILD_ID = 32'h00000000;

    wire [3:0] cmd_opcode;
    wire [15:0] cmd_seq;
    wire [7:0] cmd_vel;
    wire [15:0] cmd_geom;
    wire [15:0] cmd_intent;
    wire [15:0] cmd_checksum;
    wire [15:0] calc_checksum;
    wire checksum_ok;
    wire velocity_ok;
    wire seq_newer;
    wire seq_repeat;
    wire seq_stale;
    wire [15:0] lower_bound;
    wire [15:0] upper_bound;
    wire [15:0] clamp_min;
    wire [15:0] clamp_max;
    wire [15:0] rate_up_max;
    wire [15:0] rate_down_min;
    wire clamp_enable;
    wire rate_limit_enable;
    wire reject_out_of_range;
    wire enable;

    assign cmd_opcode = cmd_data[63:60];
    assign cmd_seq = cmd_data[59:44];
    assign cmd_vel = cmd_data[43:36];
    assign cmd_geom = cmd_data[35:20];
    assign cmd_intent = cmd_data[19:4];
    assign cmd_checksum = cmd_data[15:0];

    assign calc_checksum = {cmd_opcode, cmd_seq[3:0], cmd_vel, cmd_geom[3:0], cmd_intent[3:0], cmd_seq[15:12]};
    assign checksum_ok = (cmd_checksum == calc_checksum);
    assign velocity_ok = (cmd_vel >= 8'd20) && (cmd_vel <= 8'd55);
    assign seq_repeat = (cmd_seq == last_accepted_seq);
    assign seq_newer = (cmd_seq > last_accepted_seq);
    assign seq_stale = (!seq_newer) || seq_repeat;

    assign enable = control_reg[0];
    assign clamp_enable = control_reg[2];
    assign rate_limit_enable = control_reg[3];
    assign reject_out_of_range = control_reg[4];

    assign clamp_min = limits_min_reg[15:0];
    assign rate_down_min = limits_min_reg[31:16];
    assign clamp_max = limits_max_reg[15:0];
    assign rate_up_max = limits_max_reg[31:16];
    assign lower_bound = clamp_min;
    assign upper_bound = clamp_max;

    assign cmd_ready = (state_reg == ST_IDLE) && enable && !fault_latched && !transport_busy;
    assign bram_wr_en = cmd_ready && cmd_valid && checksum_ok;
    assign bram_wr_addr = seq_echo_reg[5:0];
    assign bram_wr_data = rsp_data_r;
    assign bram_rd_en = reg_valid && !reg_write && (reg_addr == 8'h28);
    assign bram_rd_addr = seq_echo_reg[5:0];

    assign rsp_valid = rsp_valid_r;
    assign rsp_data = rsp_data_r;
    assign actuator_valid = actuator_valid_r;
    assign actuator_fault = actuator_fault_r;
    assign actuator_fallback = actuator_fallback_r;
    assign actuator_clamp = actuator_clamp_r;
    assign actuator_mode = actuator_mode_r;
    assign actuator_seq_echo = seq_echo_reg;
    assign actuator_command = actuator_command_r;
    assign reg_rdata = reg_rdata_r;
    assign reg_ready = reg_valid;

    adaptive_aero_control_bram_if u_bram_if (
        .clk(clk),
        .rst_n(rst_n),
        .wr_en(bram_wr_en),
        .wr_addr(bram_wr_addr),
        .wr_data(bram_wr_data),
        .rd_en(bram_rd_en),
        .rd_addr(bram_rd_addr),
        .rd_data(bram_rd_data)
    );

    always @(*) begin
        next_state = state_reg;
        reg_rdata_r = 64'h0000000000000000;
        rsp_data_r = {48'h000000000000, 16'h0000};
        rsp_valid_r = 1'b0;
        actuator_valid_r = 1'b0;
        actuator_fault_r = fault_latched;
        actuator_fallback_r = fallback_active;
        actuator_clamp_r = clamp_active;
        actuator_mode_r = {1'b0, pending_mode};
        reg_ready_r = reg_valid;
        case (state_reg)
            ST_IDLE: begin
                if (cmd_valid && cmd_ready) begin
                    next_state = ST_RECEIVE;
                end
            end
            ST_RECEIVE: begin
                next_state = ST_VALIDATE;
            end
            ST_VALIDATE: begin
                if (!checksum_ok) begin
                    next_state = ST_ERROR;
                end else if (!velocity_ok && reject_out_of_range) begin
                    next_state = ST_ERROR;
                end else if (seq_stale) begin
                    next_state = ST_ERROR;
                end else begin
                    next_state = ST_APPLY;
                end
            end
            ST_APPLY: begin
                next_state = ST_REPORT;
            end
            ST_REPORT: begin
                if (rsp_ready) begin
                    next_state = ST_IDLE;
                end
            end
            ST_ERROR: begin
                next_state = ST_IDLE;
            end
            default: begin
                next_state = ST_IDLE;
            end
        endcase

        case (reg_addr)
            8'h00: reg_rdata_r = control_reg;
            8'h08: reg_rdata_r = timeout_cfg_reg;
            8'h10: reg_rdata_r = limits_min_reg;
            8'h18: reg_rdata_r = limits_max_reg;
            8'h20: reg_rdata_r = fallback_cfg_reg;
            8'h28: reg_rdata_r = {54'h0, status0_reg[9:0]};
            8'h30: reg_rdata_r = status1_reg;
            8'h38: reg_rdata_r = status2_reg;
            8'h40: reg_rdata_r = 64'h0000000000000000;
            8'h48: reg_rdata_r = version_reg;
            default: reg_rdata_r = 64'h0000000000000000;
        endcase

        if (bram_rd_en) begin
            reg_rdata_r = bram_rd_data;
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            control_reg <= 64'h0000000000000005;
            timeout_cfg_reg <= 64'h0000000003E80000;
            limits_min_reg <= 64'h0000000000000000;
            limits_max_reg <= 64'h00000000FFFF0000;
            fallback_cfg_reg <= 64'h0000000000000000;
            status0_reg <= 8'h00;
            status1_reg <= 64'h0000000000000000;
            status2_reg <= 64'h0000000000000000;
            version_reg <= {32'h00000000, IP_VERSION};
            last_accepted_seq <= 16'h0000;
            last_fault_code <= FAULT_NONE;
            last_rejected_seq <= 16'h0000;
            last_sanitized_cmd <= 16'h0000;
            last_applied_cmd <= 16'h0000;
            seq_echo_reg <= 16'h0000;
            current_cmd <= 16'h0000;
            current_sanitized <= 16'h0000;
            current_applied <= 16'h0000;
            safe_last_cmd <= 16'h0000;
            last_seq_seen <= 16'h0000;
            freshness_age <= 16'h0000;
            timeout_age <= 16'h0000;
            pending_seq <= 16'h0000;
            pending_cmd <= 16'h0000;
            pending_velocity <= 16'h0000;
            pending_opcode <= 4'h0;
            pending_mode <= 2'b00;
            pending_valid <= 1'b0;
            pending_clamp <= 1'b0;
            fault_latched <= 1'b0;
            timeout_active <= 1'b0;
            fallback_active <= 1'b0;
            clamp_active <= 1'b0;
            valid_active <= 1'b0;
            transport_busy <= 1'b0;
            rsp_valid_r <= 1'b0;
            rsp_data_r <= 64'h0000000000000000;
            actuator_command_r <= 16'h0000;
            actuator_mode_r <= 3'b000;
            actuator_fault_r <= 1'b0;
            actuator_fallback_r <= 1'b0;
            actuator_clamp_r <= 1'b0;
            actuator_valid_r <= 1'b0;
            state_reg <= ST_IDLE;
        end else begin
            state_reg <= next_state;
            reg_ready_r <= reg_valid;
            status0_reg[0] <= state_reg[0];
            status0_reg[1] <= state_reg[1];
            status0_reg[2] <= state_reg[2];
            status0_reg[4:3] <= 2'b00;
            status0_reg[5] <= fault_latched;
            status0_reg[6] <= timeout_active;
            status0_reg[7] <= fallback_active;
            status0_reg[8] <= clamp_active;
            status0_reg[9] <= valid_active;
            status1_reg[15:0] <= last_accepted_seq;
            status1_reg[23:16] <= last_fault_code;
            status1_reg[39:24] <= last_rejected_seq;
            status1_reg[63:40] <= 24'h000000;
            status2_reg[15:0] <= last_sanitized_cmd;
            status2_reg[31:16] <= last_applied_cmd;
            status2_reg[47:32] <= seq_echo_reg;
            status2_reg[63:48] <= 16'h0000;
            version_reg <= {BUILD_ID, IP_VERSION};

            if (reg_valid && reg_write) begin
                case (reg_addr)
                    8'h00: begin
                        control_reg[0] <= reg_wdata[0];
                        control_reg[1] <= reg_wdata[1];
                        control_reg[2] <= reg_wdata[2];
                        control_reg[3] <= reg_wdata[3];
                        control_reg[4] <= reg_wdata[4];
                    end
                    8'h08: begin
                        timeout_cfg_reg[15:0] <= reg_wdata[15:0];
                        timeout_cfg_reg[31:16] <= reg_wdata[31:16];
                    end
                    8'h10: begin
                        limits_min_reg[15:0] <= reg_wdata[15:0];
                        limits_min_reg[31:16] <= reg_wdata[31:16];
                    end
                    8'h18: begin
                        limits_max_reg[15:0] <= reg_wdata[15:0];
                        limits_max_reg[31:16] <= reg_wdata[31:16];
                    end
                    8'h20: begin
                        fallback_cfg_reg[1:0] <= reg_wdata[1:0];
                        fallback_cfg_reg[2] <= reg_wdata[2];
                    end
                    8'h40: begin
                        if ((reg_wdata[31:0] == CLEAR_KEY) && reg_wdata[32] && !fault_latched && (state_reg == ST_IDLE)) begin
                            fault_latched <= 1'b0;
                            last_fault_code <= FAULT_NONE;
                        end
                    end
                    default: begin
                    end
                endcase
            end

            if (cmd_valid && cmd_ready) begin
                pending_seq <= cmd_seq;
                pending_cmd <= cmd_intent;
                pending_velocity <= {8'h00, cmd_vel};
                pending_opcode <= cmd_opcode;
                pending_mode <= cmd_opcode[1:0];
                pending_valid <= 1'b1;
                pending_clamp <= 1'b0;
                seq_echo_reg <= cmd_seq;
                current_cmd <= cmd_intent;
                transport_busy <= 1'b1;
                freshness_age <= 16'h0000;
                timeout_age <= 16'h0000;
                last_seq_seen <= cmd_seq;
                rsp_valid_r <= 1'b1;
                rsp_data_r <= {16'h0000, cmd_seq, cmd_intent, cmd_vel, cmd_opcode, cmd_checksum[15:0]};
            end else begin
                if (transport_busy && timeout_age != timeout_cfg_reg[15:0]) begin
                    timeout_age <= timeout_age + 16'h0001;
                end
                if (transport_busy && timeout_age == timeout_cfg_reg[15:0]) begin
                    timeout_active <= 1'b1;
                    fault_latched <= 1'b1;
                    last_fault_code <= FAULT_TIMEOUT;
                    transport_busy <= 1'b0;
                    valid_active <= 1'b0;
                    actuator_valid_r <= 1'b0;
                    actuator_fault_r <= 1'b1;
                    actuator_fallback_r <= 1'b1;
                end
                rsp_valid_r <= rsp_valid_r & ~rsp_ready;
            end

            if (state_reg == ST_VALIDATE) begin
                if (!checksum_ok) begin
                    fault_latched <= 1'b1;
                    last_fault_code <= FAULT_INVALID_PACKET;
                    last_rejected_seq <= cmd_seq;
                    valid_active <= 1'b0;
                    actuator_valid_r <= 1'b0;
                    actuator_fault_r <= 1'b1;
                    actuator_fallback_r <= 1'b1;
                end else if (!velocity_ok && reject_out_of_range) begin
                    fault_latched <= 1'b1;
                    last_fault_code <= FAULT_VELOCITY_RANGE;
                    last_rejected_seq <= cmd_seq;
                    valid_active <= 1'b0;
                    actuator_valid_r <= 1'b0;
                    actuator_fault_r <= 1'b1;
                    actuator_fallback_r <= 1'b1;
                end else if (seq_stale) begin
                    fault_latched <= 1'b1;
                    last_fault_code <= FAULT_STALE;
                    last_rejected_seq <= cmd_seq;
                    valid_active <= 1'b0;
                    actuator_valid_r <= 1'b0;
                    actuator_fault_r <= 1'b1;
                    actuator_fallback_r <= 1'b1;
                end else begin
                    current_sanitized <= cmd_intent;
                    current_applied <= cmd_intent;
                    last_sanitized_cmd <= cmd_intent;
                    last_applied_cmd <= cmd_intent;
                    last_accepted_seq <= cmd_seq;
                    valid_active <= 1'b1;
                    actuator_valid_r <= 1'b1;
                    actuator_fault_r <= fault_latched;
                    actuator_fallback_r <= 1'b0;
                    clamp_active <= 1'b0;
                end
            end

            if (state_reg == ST_APPLY) begin
                if (clamp_enable) begin
                    if (current_sanitized < clamp_min) begin
                        current_sanitized <= clamp_min;
                        clamp_active <= 1'b1;
                    end else if (current_sanitized > clamp_max) begin
                        current_sanitized <= clamp_max;
                        clamp_active <= 1'b1;
                    end
                end
                if (rate_limit_enable) begin
                    if (current_sanitized > current_applied) begin
                        if ((current_sanitized - current_applied) > rate_up_max) begin
                            current_sanitized <= current_applied + rate_up_max;
                            clamp_active <= 1'b1;
                        end
                    end else begin
                        if ((current_applied - current_sanitized) > rate_down_min) begin
                            current_sanitized <= current_applied - rate_down_min;
                            clamp_active <= 1'b1;
                        end
                    end
                end
                actuator_command_r <= current_sanitized;
                last_sanitized_cmd <= current_sanitized;
                last_applied_cmd <= current_sanitized;
                safe_last_cmd <= current_sanitized;
                valid_active <= 1'b1;
                actuator_valid_r <= 1'b1;
                actuator_mode_r <= {1'b0, pending_mode};
                actuator_clamp_r <= clamp_active;
            end

            if (state_reg == ST_REPORT) begin
                rsp_valid_r <= 1'b1;
                rsp_data_r <= {16'h0000, pending_seq, current_sanitized, last_fault_code, 16'hA55A};
                if (rsp_ready) begin
                    transport_busy <= 1'b0;
                    valid_active <= 1'b0;
                    actuator_valid_r <= 1'b0;
                    rsp_valid_r <= 1'b0;
                end
            end

            if (state_reg == ST_ERROR) begin
                transport_busy <= 1'b0;
                valid_active <= 1'b0;
                actuator_valid_r <= 1'b0;
                actuator_fallback_r <= 1'b1;
            end

            if (fault_latched) begin
                fallback_active <= 1'b1;
                actuator_fault_r <= 1'b1;
                actuator_valid_r <= 1'b0;
            end else begin
                fallback_active <= 1'b0;
            end
        end
    end
endmodule
---END adaptive_aero_control_top.v---
