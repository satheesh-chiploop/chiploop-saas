module aero_cpu_subsystem(
    clk,
    rst_n,
    wb_adr_o,
    wb_dat_o,
    wb_dat_i,
    wb_we_o,
    wb_stb_o,
    wb_cyc_o,

    uart_tx,
    uart_rx,
    cpu_halted,
    cpu_initialized
);
input clk;
input rst_n;
output [31:0] wb_adr_o;
output [31:0] wb_dat_o;
input [31:0] wb_dat_i;
output wb_we_o;
output wb_stb_o;
output wb_cyc_o;
output [3:0] wb_sel_o;
input wb_ack_i;
input wb_err_i;
input irq_i;
output uart_tx;
input uart_rx;
output cpu_halted;
output cpu_initialized;

reg [31:0] wb_adr_r;
reg [31:0] wb_dat_r;
reg wb_we_r;
reg wb_stb_r;
reg wb_cyc_r;
reg [3:0] wb_sel_r;
reg uart_tx_r;
reg cpu_halted_r;
reg cpu_initialized_r;
reg [7:0] uart_cnt;
reg [3:0] init_cnt;

assign wb_adr_o = wb_adr_r;
assign wb_dat_o = wb_dat_r;
assign wb_we_o = wb_we_r;
assign wb_stb_o = wb_stb_r;
assign wb_cyc_o = wb_cyc_r;
assign wb_sel_o = wb_sel_r;
assign uart_tx = uart_tx_r;
assign cpu_halted = cpu_halted_r;
assign cpu_initialized = cpu_initialized_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        wb_adr_r <= 32'h00000000;
        wb_dat_r <= 32'h00000000;
        wb_we_r <= 1'b0;
        wb_stb_r <= 1'b0;
        wb_cyc_r <= 1'b0;
        wb_sel_r <= 4'h0;
        uart_tx_r <= 1'b1;
        cpu_halted_r <= 1'b1;
        cpu_initialized_r <= 1'b0;
        uart_cnt <= 8'h00;
        init_cnt <= 4'h0;
    end else begin
        wb_adr_r <= wb_adr_i;
        wb_dat_r <= wb_dat_i;
        wb_we_r <= 1'b0;
        wb_stb_r <= 1'b0;
        wb_cyc_r <= 1'b0;
        wb_sel_r <= 4'hf;
        uart_tx_r <= uart_rx ^ irq_i;
        if (!cpu_initialized_r) begin
            init_cnt <= init_cnt + 4'h1;
            if (init_cnt == 4'hf) begin
                cpu_initialized_r <= 1'b1;
                cpu_halted_r <= 1'b0;
            end
        end
        uart_cnt <= uart_cnt + 8'h01;
    end
end
endmodule

        fault_summary_r[1] <= stale_data_fault;
        fault_summary_r[2] <= validity_fault;
        fault_summary_r[3] <= integrity_fault;
        fault_summary_r[4] <= op_point_fault;
        fault_summary_r[5] <= seq_fault;
        fault_summary_r[6] <= clamp_fault;
        fault_summary_r[7] <= rate_limit_fault;
        fault_summary_r[8] <= cpu_halted;
        fault_summary_r[9] <= ~cpu_initialized;
        fault_summary_r[15:10] <= 6'd0;
        health_summary_r[0] <= cpu_initialized;
        health_summary_r[1] <= ~cpu_halted;
        health_summary_r[2] <= ~timeout_fault;
        health_summary_r[3] <= ~stale_data_fault;
        health_summary_r[4] <= ~validity_fault;
        health_summary_r[5] <= ~integrity_fault;
        health_summary_r[6] <= ~op_point_fault;
        health_summary_r[7] <= ~seq_fault;
        health_summary_r[15:8] <= 8'd0;
        fault_active_r <= (fault_summary_r[9:0] != 10'h000);
        watchdog_active_r <= timeout_fault | response_validated;
        irq_i_r <= irq_response_ready_r | irq_fault_r | irq_timeout_r;
        if (fault_clear_req && cfg_clear_faults && !(timeout_fault | stale_data_fault | validity_fault | integrity_fault | op_point_fault | seq_fault | clamp_fault | rate_limit_fault | cpu_halted | ~cpu_initialized))
            fault_summary_r <= 16'h0000;
    end
end
endmodule

module transport_buffer_wrapper(
    clk,
    rst_n,
    payload_fifo_csb,
    payload_fifo_we,
    payload_fifo_addr,
    payload_fifo_din,
    payload_fifo_dout,
    history_csb,
    history_we,
    history_addr,
    history_din,
    history_dout,
    buffer_push,
    buffer_pop,
    history_write,
    history_read,
    buffer_full,
    buffer_empty,
    history_valid
);
input clk;
input rst_n;
output payload_fifo_csb;
output payload_fifo_we;
output [7:0] payload_fifo_addr;
output [127:0] payload_fifo_din;
input [127:0] payload_fifo_dout;
output history_csb;
output history_we;
output [5:0] history_addr;
output [63:0] history_din;
input [63:0] history_dout;
input buffer_push;
input buffer_pop;
input history_write;
input history_read;
output buffer_full;
output buffer_empty;
output history_valid;

reg payload_fifo_csb_r,payload_fifo_we_r,history_csb_r,history_we_r,buffer_full_r,buffer_empty_r,history_valid_r;
reg [7:0] payload_fifo_addr_r;
reg [127:0] payload_fifo_din_r;
reg [5:0] history_addr_r;
reg [63:0] history_din_r;
reg [7:0] wr_ptr, rd_ptr, count;
reg [5:0] hist_ptr;
assign payload_fifo_csb = payload_fifo_csb_r;
assign payload_fifo_we = payload_fifo_we_r;
assign payload_fifo_addr = payload_fifo_addr_r;
assign payload_fifo_din = payload_fifo_din_r;
assign history_csb = history_csb_r;
assign history_we = history_we_r;
assign history_addr = history_addr_r;
assign history_din = history_din_r;
assign buffer_full = buffer_full_r;
assign buffer_empty = buffer_empty_r;
assign history_valid = history_valid_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        payload_fifo_csb_r <= 1'b1;
        payload_fifo_we_r <= 1'b0;
        payload_fifo_addr_r <= 8'h00;
        payload_fifo_din_r <= 128'h0;
        history_csb_r <= 1'b1;
        history_we_r <= 1'b0;
        history_addr_r <= 6'h00;
        history_din_r <= 64'h0;
        buffer_full_r <= 1'b0;
        buffer_empty_r <= 1'b1;
        history_valid_r <= 1'b0;
        wr_ptr <= 8'h00;
        rd_ptr <= 8'h00;
        count <= 8'h00;
        hist_ptr <= 6'h00;
    end else begin
        payload_fifo_csb_r <= 1'b1;
        payload_fifo_we_r <= 1'b0;
        history_csb_r <= 1'b1;
        history_we_r <= 1'b0;
        if (buffer_push && !buffer_full_r) begin
            payload_fifo_csb_r <= 1'b0;
            payload_fifo_we_r <= 1'b1;
            payload_fifo_addr_r <= wr_ptr;
            payload_fifo_din_r <= {120'd0, wr_ptr};
            wr_ptr <= wr_ptr + 8'd1;
            count <= count + 8'd1;
        end
        if (buffer_pop && !buffer_empty_r) begin
            rd_ptr <= rd_ptr + 8'd1;
            count <= count - 8'd1;
        end
        if (history_write) begin
            history_csb_r <= 1'b0;
            history_we_r <= 1'b1;
            history_addr_r <= hist_ptr;
            history_din_r <= {payload_fifo_dout[31:0], payload_fifo_dout[63:32]};
            hist_ptr <= hist_ptr + 6'd1;
            history_valid_r <= 1'b1;
        end
        if (history_read) begin
            history_valid_r <= history_valid_r | (history_dout != 64'd0);
        end
        buffer_full_r <= (count == 8'hff);
        buffer_empty_r <= (count == 8'h00);
    end
end
endmodule

module transport_payload_fifo_mem(
    clk,
    csb,
    we,
    addr,
    din,
    dout
);
input clk;
input csb;
input we;
input [7:0] addr;
input [127:0] din;
output [127:0] dout;
reg [127:0] dout_r;
reg [127:0] mem [0:255];
assign dout = dout_r;
always @(posedge clk) begin
    if (!csb) begin
        if (we) mem[addr] <= din;
        dout_r <= mem[addr];
    end
end
endmodule
