---BEGIN sram_mbist_demo_controller.v---
module sram_mbist_demo_controller (
    clk,
    reset_n,
    wr_en,
    wr_addr,
    wr_data,
    rd_en,
    rd_addr,
    bist_start,
    rd_data,
    ready,
    bist_done,
    bist_fail,
    irq
);
input clk;
input reset_n;
input wr_en;
input [7:0] wr_addr;
input [31:0] wr_data;
input rd_en;
input [7:0] rd_addr;
input bist_start;
output [31:0] rd_data;
output ready;
output bist_done;
output bist_fail;
output irq;

wire [31:0] rd_data_w;
wire ready_w;
wire bist_done_w;
wire bist_fail_w;
wire irq_w;
wire [31:0] mem_dout;
wire [31:0] read_data_mux;
wire [31:0] status_reg_w;
reg [31:0] mem [0:255];
reg [31:0] rd_data_r;
reg ready_r;
reg bist_done_r;
reg bist_fail_r;
reg irq_r;
reg [31:0] control_reg;
reg [31:0] status_reg;
reg [31:0] mem_addr_reg;
reg [31:0] mem_wdata_reg;
reg [31:0] mem_rdata_reg;
reg [31:0] mem_control_reg;
reg [31:0] bist_control_reg;
reg [31:0] bist_status_reg;
reg [31:0] irq_status_reg;
reg [31:0] irq_clear_reg;
reg [7:0] last_fail_addr_reg;
reg busy_r;
reg bist_running_r;
reg bist_request_pending_r;
reg [7:0] mem_addr_mux;
reg [31:0] mem_din_mux;
reg mem_csb;
reg mem_web;
integer i;

assign rd_data = rd_data_r;
assign ready = ready_r;
assign bist_done = bist_done_r;
assign bist_fail = bist_fail_r;
assign irq = irq_r;

assign status_reg_w = {28'h0000000, busy_r, bist_fail_r, bist_done_r, ready_r};
assign read_data_mux = rd_data_r;

demo_sram_32x256_model u_sram (
    .clk(clk),
    .csb(mem_csb),
    .web(mem_web),
    .addr(mem_addr_mux),
    .din(mem_din_mux),
    .dout(mem_dout)
);

always @(*) begin
    mem_addr_mux = 8'h00;
    mem_din_mux = 32'h00000000;
    mem_csb = 1'b1;
    mem_web = 1'b1;
    if (busy_r) begin
        mem_csb = 1'b0;
        mem_web = 1'b1;
        mem_addr_mux = mem_addr_reg[7:0];
        mem_din_mux = mem_wdata_reg;
    end
end

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        rd_data_r <= 32'h00000000;
        ready_r <= 1'b0;
        bist_done_r <= 1'b0;
        bist_fail_r <= 1'b0;
        irq_r <= 1'b0;
        control_reg <= 32'h00000000;
        status_reg <= 32'h00000000;
        mem_addr_reg <= 32'h00000000;
        mem_wdata_reg <= 32'h00000000;
        mem_rdata_reg <= 32'h00000000;
        mem_control_reg <= 32'h00000000;
        bist_control_reg <= 32'h00000000;
        bist_status_reg <= 32'h00000000;
        irq_status_reg <= 32'h00000000;
        irq_clear_reg <= 32'h00000000;
        last_fail_addr_reg <= 8'h00;
        busy_r <= 1'b0;
        bist_running_r <= 1'b0;
        bist_request_pending_r <= 1'b0;
        for (i = 0; i < 256; i = i + 1) begin
            mem[i] <= 32'h00000000;
        end
    end else begin
        ready_r <= control_reg[0] & ~busy_r;
        if (control_reg[1]) begin
            control_reg <= 32'h00000000;
            status_reg <= 32'h00000000;
            mem_addr_reg <= 32'h00000000;
            mem_wdata_reg <= 32'h00000000;
            mem_rdata_reg <= 32'h00000000;
            mem_control_reg <= 32'h00000000;
            bist_control_reg <= 32'h00000000;
            bist_status_reg <= 32'h00000000;
            irq_status_reg <= 32'h00000000;
            irq_clear_reg <= 32'h00000000;
            last_fail_addr_reg <= 8'h00;
            busy_r <= 1'b0;
            bist_running_r <= 1'b0;
            bist_request_pending_r <= 1'b0;
            rd_data_r <= 32'h00000000;
            bist_done_r <= 1'b0;
            bist_fail_r <= 1'b0;
            irq_r <= 1'b0;
        end else begin
            if (wr_en) begin
                mem[wr_addr] <= wr_data;
            end
            if (rd_en) begin
                rd_data_r <= mem[rd_addr];
            end
            if (bist_start || bist_control_reg[0]) begin
                bist_request_pending_r <= 1'b1;
                bist_running_r <= 1'b1;
                busy_r <= 1'b1;
                bist_done_r <= 1'b0;
                bist_fail_r <= 1'b0;
                bist_status_reg[2] <= 1'b1;
                bist_status_reg[0] <= 1'b0;
                bist_status_reg[1] <= 1'b0;
            end else begin
                bist_request_pending_r <= 1'b0;
            end
            if (mem_control_reg[0] && control_reg[0]) begin
                mem[mem_addr_reg[7:0]] <= mem_wdata_reg;
            end
            if (mem_control_reg[1] && control_reg[0]) begin
                mem_rdata_reg <= mem[mem_addr_reg[7:0]];
                rd_data_r <= mem[mem_addr_reg[7:0]];
            end
            if (bist_request_pending_r) begin
                bist_done_r <= 1'b1;
                bist_running_r <= 1'b0;
                busy_r <= 1'b0;
                bist_status_reg[2] <= 1'b0;
                bist_status_reg[0] <= 1'b1;
                irq_status_reg[0] <= 1'b1;
            end
            if (irq_clear_reg[0]) begin
                irq_status_reg[0] <= 1'b0;
            end
            if (irq_clear_reg[1]) begin
                irq_status_reg[1] <= 1'b0;
            end
            if (irq_status_reg[0] || irq_status_reg[1]) begin
                irq_r <= control_reg[2];
            end else begin
                irq_r <= 1'b0;
            end
            if (wr_en) begin
                case (wr_addr)
                    8'h00: control_reg <= wr_data;
                    8'h08: mem_addr_reg <= {24'h000000, wr_data[7:0]};
                    8'h0C: mem_wdata_reg <= wr_data;
                    8'h14: mem_control_reg <= wr_data;
                    8'h18: bist_control_reg <= wr_data;
                    8'h20: irq_status_reg <= (irq_status_reg | wr_data);
                    8'h24: irq_clear_reg <= wr_data;
                    default: begin
                    end
                endcase
            end
        end
        status_reg[0] <= ready_r;
        status_reg[1] <= bist_done_r;
        status_reg[2] <= bist_fail_r;
        status_reg[3] <= busy_r;
        status_reg[31:4] <= 28'h0000000;
        bist_status_reg[0] <= bist_done_r;
        bist_status_reg[1] <= bist_fail_r;
        bist_status_reg[2] <= bist_running_r;
        bist_status_reg[10:3] <= last_fail_addr_reg;
        bist_status_reg[31:11] <= 21'h000000;
        if (rd_en) begin
            case (rd_addr)
                8'h00: rd_data_r <= control_reg;
                8'h04: rd_data_r <= status_reg;
                8'h08: rd_data_r <= mem_addr_reg;
                8'h0C: rd_data_r <= mem_wdata_reg;
                8'h10: rd_data_r <= mem_rdata_reg;
                8'h14: rd_data_r <= mem_control_reg;
                8'h18: rd_data_r <= bist_control_reg;
                8'h1C: rd_data_r <= bist_status_reg;
                8'h20: rd_data_r <= irq_status_reg;
                8'h24: rd_data_r <= irq_clear_reg;
                default: rd_data_r <= mem[rd_addr];
            endcase
        end
        if (bist_control_reg[1]) begin
            bist_done_r <= 1'b0;
            bist_fail_r <= 1'b0;
            irq_status_reg[0] <= 1'b0;
            irq_status_reg[1] <= 1'b0;
            bist_status_reg[0] <= 1'b0;
            bist_status_reg[1] <= 1'b0;
            bist_control_reg[1] <= 1'b0;
        end
    end
end
endmodule
---END sram_mbist_demo_controller.v---
