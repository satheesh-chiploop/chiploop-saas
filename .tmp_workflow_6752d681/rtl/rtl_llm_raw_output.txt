    if (!reset_n) begin
        mmio_ready_r <= 1'b0;
        mmio_rvalid_r <= 1'b0;
        mmio_rdata_r <= 64'h0000000000000000;
        csr_write_en_r <= 1'b0;
        csr_read_en_r <= 1'b0;
        csr_addr_r <= 16'h0000;
        csr_wdata_r <= 64'h0000000000000000;
    end else begin
        mmio_ready_r <= mmio_valid;
        mmio_rvalid_r <= mmio_valid & ~mmio_we;
        mmio_rdata_r <= csr_rdata;
        csr_write_en_r <= mmio_valid & mmio_we;
        csr_read_en_r <= mmio_valid & ~mmio_we;
        csr_addr_r <= mmio_addr;
        csr_wdata_r <= mmio_wdata;
    end
end

        8'h1C: csr_rdata_r = {56'h00000000000000, cfg_flow_velocity_mps_r[7:0]};
        8'h1D: csr_rdata_r = {56'h00000000000000, cfg_flow_velocity_mps_r[15:8]};
        8'h1E: csr_rdata_r = {56'h00000000000000, cfg_flow_velocity_mps_r[23:16]};
        8'h1F: csr_rdata_r = {56'h00000000000000, cfg_flow_velocity_mps_r[31:24]};
        8'h20: csr_rdata_r = {56'h00000000000000, status0_r};
        8'h21: csr_rdata_r = {56'h00000000000000, status1_r};
        8'h22: csr_rdata_r = {56'h00000000000000, status2_r};
        8'h23: csr_rdata_r = {56'h00000000000000, status3_r};
        default: csr_rdata_r = 64'h0000000000000000;
    endcase
end

endmodule

module request_packetizer (
    clk,
    reset_n,
    cfg_enable,
    cfg_queue_enable,
    cfg_sequence_seed,
    cfg_geometry_format_id,
    cfg_flow_velocity_mps,
    flow_velocity_valid,
    flow_velocity_in_range,
    geometry_desc_valid,
    geometry_desc_length,
    geometry_desc_framed,
    req_ready,
    req_valid,
    req_data,
    request_accepted,
    outstanding_request_count,
    launch_request
);
input clk;
input reset_n;
input cfg_enable;
input cfg_queue_enable;
input [31:0] cfg_sequence_seed;
input [15:0] cfg_geometry_format_id;
input [31:0] cfg_flow_velocity_mps;
output flow_velocity_valid;
output flow_velocity_in_range;
input geometry_desc_valid;
input [15:0] geometry_desc_length;
input geometry_desc_framed;
input req_ready;
output req_valid;
output [127:0] req_data;
output request_accepted;
output [1:0] outstanding_request_count;
output launch_request;

reg flow_velocity_valid_r;
reg flow_velocity_in_range_r;
reg req_valid_r;
reg [127:0] req_data_r;
reg request_accepted_r;
reg [1:0] outstanding_request_count_r;
reg launch_request_r;
reg [31:0] seq_reg;

assign flow_velocity_valid = flow_velocity_valid_r;
assign flow_velocity_in_range = flow_velocity_in_range_r;
assign req_valid = req_valid_r;
assign req_data = req_data_r;
assign request_accepted = request_accepted_r;
assign outstanding_request_count = outstanding_request_count_r;
assign launch_request = launch_request_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        flow_velocity_valid_r <= 1'b0;
        flow_velocity_in_range_r <= 1'b0;
        req_valid_r <= 1'b0;
        req_data_r <= 128'h00000000000000000000000000000000;
        request_accepted_r <= 1'b0;
        outstanding_request_count_r <= 2'b00;
        launch_request_r <= 1'b0;
        seq_reg <= 32'd1;
    end else begin
        flow_velocity_valid_r <= cfg_enable & geometry_desc_valid;
        flow_velocity_in_range_r <= (cfg_flow_velocity_mps >= 32'd20) && (cfg_flow_velocity_mps <= 32'd55);
        launch_request_r <= 1'b0;
        request_accepted_r <= 1'b0;
        if (cfg_enable && geometry_desc_valid && geometry_desc_framed && flow_velocity_in_range_r) begin
            if ((outstanding_request_count_r == 2'b00) || (cfg_queue_enable && outstanding_request_count_r != 2'b10)) begin
                req_valid_r <= 1'b1;
                req_data_r <= {16'hA5A5, cfg_geometry_format_id, geometry_desc_length, cfg_sequence_seed[15:0], cfg_flow_velocity_mps[15:0], seq_reg[15:0], 64'h0000000000000000};
                if (req_ready) begin
                    request_accepted_r <= 1'b1;
                    launch_request_r <= 1'b1;
                    if (outstanding_request_count_r != 2'b10)
                        outstanding_request_count_r <= outstanding_request_count_r + 2'b01;
                    seq_reg <= seq_reg + 32'd1;
                end
            end else begin
                req_valid_r <= 1'b0;
            end
        end else begin
            req_valid_r <= 1'b0;
        end
        if (request_accepted_r && (outstanding_request_count_r != 2'b00))
            outstanding_request_count_r <= outstanding_request_count_r;
        if (launch_request_r && (outstanding_request_count_r != 2'b00))
            outstanding_request_count_r <= outstanding_request_count_r;
        if (!cfg_enable)
            outstanding_request_count_r <= 2'b00;
    end
end

endmodule

module freshness_and_integrity_monitor (
    clk,
    reset_n,
    launch_request,
    outstanding_request_count,
    cfg_timeout_cycles,
    request_sequence_id,
    response_valid,
    response_sequence_id,
    response_integrity_tag,
    watchdog_expired,
    sequence_match,
    integrity_ok,
    fresh_response_accepted,
    stale_rejected,
    integrity_error,
    fault_code,
    last_sequence_id
);
input clk;
input reset_n;
input launch_request;
input [1:0] outstanding_request_count;
input [31:0] cfg_timeout_cycles;
input [15:0] request_sequence_id;
input response_valid;
input [15:0] response_sequence_id;
input [15:0] response_integrity_tag;
input watchdog_expired;
output sequence_match;
output integrity_ok;
output fresh_response_accepted;
output stale_rejected;
output integrity_error;
output [7:0] fault_code;
output [15:0] last_sequence_id;

reg sequence_match_r;
reg integrity_ok_r;
reg fresh_response_accepted_r;
reg stale_rejected_r;
reg integrity_error_r;
reg [7:0] fault_code_r;
reg [15:0] last_sequence_id_r;

assign sequence_match = sequence_match_r;
assign integrity_ok = integrity_ok_r;
assign fresh_response_accepted = fresh_response_accepted_r;
assign stale_rejected = stale_rejected_r;
assign integrity_error = integrity_error_r;
assign fault_code = fault_code_r;
assign last_sequence_id = last_sequence_id_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        sequence_match_r <= 1'b0;
        integrity_ok_r <= 1'b0;
        fresh_response_accepted_r <= 1'b0;
        stale_rejected_r <= 1'b0;
        integrity_error_r <= 1'b0;
        fault_code_r <= 8'h00;
        last_sequence_id_r <= 16'h0000;
    end else begin
        sequence_match_r <= response_valid && (response_sequence_id == request_sequence_id) && (outstanding_request_count != 2'b00);
        integrity_ok_r <= (response_integrity_tag == {request_sequence_id[7:0], request_sequence_id[15:8]});
        fresh_response_accepted_r <= response_valid && sequence_match_r && integrity_ok_r && !watchdog_expired;
        stale_rejected_r <= response_valid && !sequence_match_r;
        integrity_error_r <= response_valid && sequence_match_r && !integrity_ok_r;
        fault_code_r <= 8'h00;
        if (response_valid && !sequence_match_r)
            fault_code_r <= 8'h11;
        else if (response_valid && sequence_match_r && !integrity_ok_r)
            fault_code_r <= 8'h22;
        else if (watchdog_expired)
            fault_code_r <= 8'h33;
        if (response_valid)
            last_sequence_id_r <= response_sequence_id;
        else if (launch_request)
            last_sequence_id_r <= request_sequence_id;
    end
end

endmodule

module timeout_watchdog (
    clk,
    reset_n,
    launch_request,
    fresh_response_accepted,
    cfg_timeout_cycles,
    busy,
    timeout_fired,
    watchdog_running,
    watchdog_expired
);
input clk;
input reset_n;
input launch_request;
input fresh_response_accepted;
input [31:0] cfg_timeout_cycles;
input busy;
output timeout_fired;
output watchdog_running;
output watchdog_expired;

reg timeout_fired_r;
reg watchdog_running_r;
reg watchdog_expired_r;
reg [31:0] timer_r;

assign timeout_fired = timeout_fired_r;
assign watchdog_running = watchdog_running_r;
assign watchdog_expired = watchdog_expired_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        timeout_fired_r <= 1'b0;
        watchdog_running_r <= 1'b0;
        watchdog_expired_r <= 1'b0;
        timer_r <= 32'd0;
    end else begin
        timeout_fired_r <= 1'b0;
        if (launch_request) begin
            watchdog_running_r <= 1'b1;
            timer_r <= 32'd0;
        end else if (fresh_response_accepted) begin
            watchdog_running_r <= 1'b0;
            timer_r <= 32'd0;
        end else if (watchdog_running_r) begin
            timer_r <= timer_r + 32'd1;
            if (timer_r >= cfg_timeout_cycles) begin
                timeout_fired_r <= 1'b1;
                watchdog_running_r <= 1'b0;
                watchdog_expired_r <= 1'b1;
            end
        end
        if (busy && !watchdog_running_r)
            watchdog_running_r <= watchdog_running_r;
        if (fresh_response_accepted)
            watchdog_expired_r <= 1'b0;
    end
end

endmodule

module safety_supervisor (
    clk,
    reset_n,
    cfg_bypass_mode,
    cfg_clamp_velocity_enable,
    cfg_fallback_hold_last_safe,
    cfg_min_actuator_position,
    cfg_max_actuator_position,
    cfg_max_slew_per_update,
    flow_velocity_mps,
    flow_velocity_valid,
    flow_velocity_in_range,
    advisory_position,
    advisory_valid,
    fault_active,
    clamped_position,
    clamp_event,
    safety_ok,
    safe_position
);
input clk;
input reset_n;
input cfg_bypass_mode;
input cfg_clamp_velocity_enable;
input cfg_fallback_hold_last_safe;
input [31:0] cfg_min_actuator_position;
input [31:0] cfg_max_actuator_position;
input [31:0] cfg_max_slew_per_update;
input [31:0] flow_velocity_mps;
input flow_velocity_valid;
input flow_velocity_in_range;
input [31:0] advisory_position;
input advisory_valid;
input fault_active;
output [31:0] clamped_position;
output clamp_event;
output safety_ok;
output [31:0] safe_position;

reg [31:0] clamped_position_r;
reg clamp_event_r;
reg safety_ok_r;
reg [31:0] safe_position_r;
reg [31:0] last_safe_r;

assign clamped_position = clamped_position_r;
assign clamp_event = clamp_event_r;
assign safety_ok = safety_ok_r;
assign safe_position = safe_position_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        clamped_position_r <= 32'd0;
        clamp_event_r <= 1'b0;
        safety_ok_r <= 1'b0;
        safe_position_r <= 32'd0;
        last_safe_r <= 32'd0;
    end else begin
        clamp_event_r <= 1'b0;
        if (advisory_valid && !fault_active) begin
            if (cfg_bypass_mode) begin
                clamped_position_r <= advisory_position;
                safety_ok_r <= 1'b1;
            end else begin
                if (advisory_position < cfg_min_actuator_position) begin
                    clamped_position_r <= cfg_min_actuator_position;
                    clamp_event_r <= 1'b1;
                end else if (advisory_position > cfg_max_actuator_position) begin
                    clamped_position_r <= cfg_max_actuator_position;
                    clamp_event_r <= 1'b1;
                end else begin
                    clamped_position_r <= advisory_position;
                end
                if ((flow_velocity_valid && !flow_velocity_in_range) && !cfg_clamp_velocity_enable) begin
                    safety_ok_r <= 1'b0;
                end else begin
                    safety_ok_r <= 1'b1;
                end
            end
        end else begin
            safety_ok_r <= 1'b0;
        end
        if (cfg_fallback_hold_last_safe && safety_ok_r)
            last_safe_r <= clamped_position_r;
        if (fault_active || (flow_velocity_valid && !flow_velocity_in_range && !cfg_clamp_velocity_enable))
            safe_position_r <= cfg_min_actuator_position;
        else if (cfg_fallback_hold_last_safe)
            safe_position_r <= last_safe_r;
        else
            safe_position_r <= cfg_max_actuator_position;
    end
end

endmodule

);

input fault_active;

assign fallback_position = fallback_position_r;
assign fault_latched = fault_latched_r;

always @(posedge clk or negedge reset_n) begin

        fallback_active_r <= 1'b1;

            hold_position_r <= safe_position;

        if (cfg_fallback_hold_last_safe)
            fallback_position_r <= hold_position_r;
        else
            fallback_position_r <= cfg_safe_fallback_position;
    end
