module adaptive_aero_control_regfile (
    clk,
    rst_n,
    reg_addr,
    reg_wdata,
    reg_wr_en,
    reg_rd_en,
    reg_rdata,
    cfg_enable,
    cfg_soft_reset,
    cfg_pipelined_mode,
    cfg_seq_seed,
    cfg_request_timeout,
    cfg_response_age_limit,
    cfg_max_act_cmd,
    cfg_min_act_cmd,
    cfg_fallback_cmd,
    cfg_stream_velocity_mps,
    cfg_flow_condition,
    cfg_geometry_format,
    cfg_geometry_source,
    cfg_geom_token,
    cfg_status_clear,
    cfg_issue_request,
    status_in,
    command_in
);
input clk;
input rst_n;
input [7:0] reg_addr;
input [31:0] reg_wdata;
input reg_wr_en;
input reg_rd_en;
output [31:0] reg_rdata;
output cfg_enable;
output cfg_soft_reset;
output cfg_pipelined_mode;
output [15:0] cfg_seq_seed;
output [15:0] cfg_request_timeout;
output [15:0] cfg_response_age_limit;
output [31:0] cfg_max_act_cmd;
output [31:0] cfg_min_act_cmd;
output [31:0] cfg_fallback_cmd;
output [31:0] cfg_stream_velocity_mps;
output [31:0] cfg_flow_condition;
output [7:0] cfg_geometry_format;
output [31:0] cfg_geometry_source;
output [31:0] cfg_geom_token;
output cfg_status_clear;
output cfg_issue_request;
input [31:0] status_in;
input [31:0] command_in;

reg cfg_enable_r;
reg cfg_soft_reset_r;
reg cfg_pipelined_mode_r;
reg [15:0] cfg_seq_seed_r;
reg [15:0] cfg_request_timeout_r;
reg [15:0] cfg_response_age_limit_r;
reg [31:0] cfg_max_act_cmd_r;
reg [31:0] cfg_min_act_cmd_r;
reg [31:0] cfg_fallback_cmd_r;
reg [31:0] cfg_stream_velocity_mps_r;
reg [31:0] cfg_flow_condition_r;
reg [7:0] cfg_geometry_format_r;
reg [31:0] cfg_geometry_source_r;
reg [31:0] cfg_geom_token_r;
reg cfg_status_clear_r;
reg cfg_issue_request_r;
reg [31:0] reg_rdata_r;
reg [31:0] last_status_r;
reg [31:0] last_command_r;

assign cfg_enable = cfg_enable_r;
assign cfg_soft_reset = cfg_soft_reset_r;
assign cfg_pipelined_mode = cfg_pipelined_mode_r;
assign cfg_seq_seed = cfg_seq_seed_r;
assign cfg_request_timeout = cfg_request_timeout_r;
assign cfg_response_age_limit = cfg_response_age_limit_r;
assign cfg_max_act_cmd = cfg_max_act_cmd_r;
assign cfg_min_act_cmd = cfg_min_act_cmd_r;
assign cfg_fallback_cmd = cfg_fallback_cmd_r;
assign cfg_stream_velocity_mps = cfg_stream_velocity_mps_r;
assign cfg_flow_condition = cfg_flow_condition_r;
assign cfg_geometry_format = cfg_geometry_format_r;
assign cfg_geometry_source = cfg_geometry_source_r;
assign cfg_geom_token = cfg_geom_token_r;
assign cfg_status_clear = cfg_status_clear_r;
assign cfg_issue_request = cfg_issue_request_r;
assign reg_rdata = reg_rdata_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        cfg_enable_r <= 1'b0;
        cfg_soft_reset_r <= 1'b0;
        cfg_pipelined_mode_r <= 1'b0;
        cfg_seq_seed_r <= 16'h0000;
        cfg_request_timeout_r <= 16'h0000;
        cfg_response_age_limit_r <= 16'h0000;
        cfg_max_act_cmd_r <= 32'h000000FF;
        cfg_min_act_cmd_r <= 32'h00000000;
        cfg_fallback_cmd_r <= 32'h00000000;
        cfg_stream_velocity_mps_r <= 32'h0000001E;
        cfg_flow_condition_r <= 32'h00000000;
        cfg_geometry_format_r <= 8'h01;
        cfg_geometry_source_r <= 32'h00000001;
        cfg_geom_token_r <= 32'h00000000;
        cfg_status_clear_r <= 1'b0;
        cfg_issue_request_r <= 1'b0;
        reg_rdata_r <= 32'h00000000;
        last_status_r <= 32'h00000000;
        last_command_r <= 32'h00000000;
    end else begin
        cfg_soft_reset_r <= 1'b0;
        cfg_status_clear_r <= 1'b0;
        cfg_issue_request_r <= 1'b0;
        if (reg_wr_en) begin
            case (reg_addr)
                8'h00: begin
                    cfg_enable_r <= reg_wdata[0];
                    cfg_soft_reset_r <= reg_wdata[1];
                    cfg_pipelined_mode_r <= reg_wdata[2];
                    cfg_status_clear_r <= reg_wdata[3];
                    cfg_issue_request_r <= reg_wdata[4];
                end
                8'h01: cfg_seq_seed_r[7:0] <= reg_wdata[7:0];
                8'h02: cfg_seq_seed_r[15:8] <= reg_wdata[7:0];
                8'h03: cfg_request_timeout_r[7:0] <= reg_wdata[7:0];
                8'h04: cfg_request_timeout_r[15:8] <= reg_wdata[7:0];
                8'h05: cfg_response_age_limit_r[7:0] <= reg_wdata[7:0];
                8'h06: cfg_response_age_limit_r[15:8] <= reg_wdata[7:0];
                8'h07: cfg_min_act_cmd_r[7:0] <= reg_wdata[7:0];
                8'h08: cfg_min_act_cmd_r[15:8] <= reg_wdata[7:0];
                8'h09: cfg_min_act_cmd_r[23:16] <= reg_wdata[7:0];
                8'h0A: cfg_min_act_cmd_r[31:24] <= reg_wdata[7:0];
                8'h0B: cfg_max_act_cmd_r[7:0] <= reg_wdata[7:0];
                8'h0C: cfg_max_act_cmd_r[15:8] <= reg_wdata[7:0];
                8'h0D: cfg_max_act_cmd_r[23:16] <= reg_wdata[7:0];
                8'h0E: cfg_max_act_cmd_r[31:24] <= reg_wdata[7:0];
                8'h0F: cfg_fallback_cmd_r[7:0] <= reg_wdata[7:0];
                8'h10: cfg_fallback_cmd_r[15:8] <= reg_wdata[7:0];
                8'h11: cfg_fallback_cmd_r[23:16] <= reg_wdata[7:0];
                8'h12: cfg_fallback_cmd_r[31:24] <= reg_wdata[7:0];
                8'h13: cfg_stream_velocity_mps_r[7:0] <= reg_wdata[7:0];
                8'h14: cfg_stream_velocity_mps_r[15:8] <= reg_wdata[7:0];
                8'h15: cfg_stream_velocity_mps_r[23:16] <= reg_wdata[7:0];
                8'h16: cfg_stream_velocity_mps_r[31:24] <= reg_wdata[7:0];
                8'h17: cfg_flow_condition_r[7:0] <= reg_wdata[7:0];
                8'h18: cfg_flow_condition_r[15:8] <= reg_wdata[7:0];
                8'h19: cfg_flow_condition_r[23:16] <= reg_wdata[7:0];
                8'h1A: cfg_flow_condition_r[31:24] <= reg_wdata[7:0];
                8'h1B: begin
                    cfg_geometry_format_r[3:0] <= reg_wdata[3:0];
                    cfg_geometry_source_r[3:0] <= reg_wdata[7:4];
                end
                8'h1C: cfg_geom_token_r[7:0] <= reg_wdata[7:0];
                8'h1D: cfg_geom_token_r[15:8] <= reg_wdata[7:0];
                8'h1E: cfg_geom_token_r[23:16] <= reg_wdata[7:0];
                8'h1F: cfg_geom_token_r[31:24] <= reg_wdata[7:0];
                8'h2D: ;
                8'h2E: ;
                8'h2F: ;
                8'h30: ;
                default: begin end
            endcase
        end
        if (reg_rd_en) begin
            case (reg_addr)
                8'h00: reg_rdata_r <= {24'h000000, status_in[6], status_in[5], status_in[4], status_in[3], status_in[2], status_in[1], status_in[0]};
                8'h01: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[7:0]};
                8'h02: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[15:8]};
                8'h03: reg_rdata_r <= {24'h000000, cfg_request_timeout_r[7:0]};
                8'h04: reg_rdata_r <= {24'h000000, cfg_request_timeout_r[15:8]};
                8'h05: reg_rdata_r <= {24'h000000, cfg_response_age_limit_r[7:0]};
                8'h06: reg_rdata_r <= {24'h000000, cfg_response_age_limit_r[15:8]};
                8'h07: reg_rdata_r <= {24'h000000, cfg_min_act_cmd_r[7:0]};
                8'h08: reg_rdata_r <= {24'h000000, cfg_min_act_cmd_r[15:8]};
                8'h09: reg_rdata_r <= {24'h000000, cfg_min_act_cmd_r[23:16]};
                8'h0A: reg_rdata_r <= {24'h000000, cfg_min_act_cmd_r[31:24]};
                8'h0B: reg_rdata_r <= {24'h000000, cfg_max_act_cmd_r[7:0]};
                8'h0C: reg_rdata_r <= {24'h000000, cfg_max_act_cmd_r[15:8]};
                8'h0D: reg_rdata_r <= {24'h000000, cfg_max_act_cmd_r[23:16]};
                8'h0E: reg_rdata_r <= {24'h000000, cfg_max_act_cmd_r[31:24]};
                8'h0F: reg_rdata_r <= {24'h000000, cfg_fallback_cmd_r[7:0]};
                8'h10: reg_rdata_r <= {24'h000000, cfg_fallback_cmd_r[15:8]};
                8'h11: reg_rdata_r <= {24'h000000, cfg_fallback_cmd_r[23:16]};
                8'h12: reg_rdata_r <= {24'h000000, cfg_fallback_cmd_r[31:24]};
                8'h13: reg_rdata_r <= {24'h000000, cfg_stream_velocity_mps_r[7:0]};
                8'h14: reg_rdata_r <= {24'h000000, cfg_stream_velocity_mps_r[15:8]};
                8'h15: reg_rdata_r <= {24'h000000, cfg_stream_velocity_mps_r[23:16]};
                8'h16: reg_rdata_r <= {24'h000000, cfg_stream_velocity_mps_r[31:24]};
                8'h17: reg_rdata_r <= {24'h000000, cfg_flow_condition_r[7:0]};
                8'h18: reg_rdata_r <= {24'h000000, cfg_flow_condition_r[15:8]};
                8'h19: reg_rdata_r <= {24'h000000, cfg_flow_condition_r[23:16]};
                8'h1A: reg_rdata_r <= {24'h000000, cfg_flow_condition_r[31:24]};
                8'h1B: reg_rdata_r <= {24'h000000, cfg_geometry_source_r[7:4], cfg_geometry_format_r[3:0]};
                8'h1C: reg_rdata_r <= {24'h000000, cfg_geom_token_r[7:0]};
                8'h1D: reg_rdata_r <= {24'h000000, cfg_geom_token_r[15:8]};
                8'h1E: reg_rdata_r <= {24'h000000, cfg_geom_token_r[23:16]};
                8'h1F: reg_rdata_r <= {24'h000000, cfg_geom_token_r[31:24]};
                8'h20: reg_rdata_r <= status_in;
                8'h21: reg_rdata_r <= {24'h000000, 1'b0, 1'b0, 1'b0, 1'b0, status_in[4], status_in[5], 1'b0};
                8'h22: reg_rdata_r <= {24'h000000, 8'h00};
                8'h23: reg_rdata_r <= {24'h000000, 8'h00};
                8'h24: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[7:0]};
                8'h25: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[15:8]};
                8'h26: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[7:0]};
                8'h27: reg_rdata_r <= {24'h000000, cfg_seq_seed_r[15:8]};
                8'h28: reg_rdata_r <= command_in;
                8'h29: reg_rdata_r <= command_in;
                8'h2A: reg_rdata_r <= command_in;
                8'h2B: reg_rdata_r <= command_in;
                8'h2C: reg_rdata_r <= 32'h00000000;
                8'h2D: reg_rdata_r <= 32'h00000000;
                8'h2E: reg_rdata_r <= 32'h00000000;
                8'h2F: reg_rdata_r <= 32'h00000000;
                8'h30: reg_rdata_r <= 32'h00000000;
                default: reg_rdata_r <= 32'h00000000;
            endcase
        end
        last_status_r <= status_in;
        last_command_r <= command_in;
    end

input [15:0] cfg_seq_seed;
input [31:0] cfg_stream_velocity_mps;
input [31:0] cfg_flow_condition;
input [7:0] cfg_geometry_format;
input [31:0] cfg_geometry_source;
input [31:0] cfg_geom_token;
input req_ready;
input issue_request;
input [15:0] transaction_id_in;
input [15:0] sequence_counter_in;
input request_inflight;
output [127:0] req_data;
output req_valid;
output request_issued;

reg [127:0] req_data_r;
reg req_valid_r;
reg request_issued_r;
reg [127:0] hold_packet_r;
reg hold_valid_r;

assign req_data = hold_valid_r ? hold_packet_r : req_data_r;
assign req_valid = hold_valid_r;
assign request_issued = request_issued_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        req_data_r <= 128'h00000000000000000000000000000000;
        req_valid_r <= 1'b0;
        request_issued_r <= 1'b0;
        hold_packet_r <= 128'h00000000000000000000000000000000;
        hold_valid_r <= 1'b0;
    end else begin
        request_issued_r <= 1'b0;
        if (cfg_enable && issue_request && (cfg_pipelined_mode || !request_inflight)) begin
            hold_packet_r <= {transaction_id_in, sequence_counter_in, cfg_seq_seed, cfg_geometry_format, cfg_geometry_source[23:0], cfg_stream_velocity_mps[15:0], cfg_flow_condition[15:0], cfg_geom_token};
            hold_valid_r <= 1'b1;
            request_issued_r <= 1'b1;
            req_data_r <= {transaction_id_in, sequence_counter_in, cfg_seq_seed, cfg_geometry_format, cfg_geometry_source[23:0], cfg_stream_velocity_mps[15:0], cfg_flow_condition[15:0], cfg_geom_token};
            req_valid_r <= 1'b1;
        end else if (hold_valid_r && req_ready) begin
            hold_valid_r <= 1'b0;
            req_valid_r <= 1'b0;
        end else if (!cfg_enable) begin
            req_valid_r <= 1'b0;
            hold_valid_r <= 1'b0;
        end else begin
            req_valid_r <= hold_valid_r;
        end
    end
end

endmodule

module adaptive_aero_response_validator (
    clk,
    rst_n,
    rsp_data,
    rsp_valid,
    transaction_id_expected,
    sequence_counter_expected,
    watchdog_count,
    cfg_request_timeout,
    cfg_response_age_limit,
    cfg_enable,
    model_present,
    rsp_ready,
    accepted,
    validation_failed,
    stale_rejected,
    timeout_hit,
    accepted_model_data
);
input clk;
input rst_n;
input [127:0] rsp_data;
input rsp_valid;
input [15:0] transaction_id_expected;
input [15:0] sequence_counter_expected;
input [15:0] watchdog_count;
input [15:0] cfg_request_timeout;
input [15:0] cfg_response_age_limit;
input cfg_enable;
input model_present;
output rsp_ready;
output accepted;
output validation_failed;
output stale_rejected;
output timeout_hit;
output [127:0] accepted_model_data;

reg rsp_ready_r;
reg accepted_r;
reg validation_failed_r;
reg stale_rejected_r;
reg timeout_hit_r;
reg [127:0] accepted_model_data_r;

assign rsp_ready = cfg_enable & model_present;
assign accepted = accepted_r;
assign validation_failed = validation_failed_r;
assign stale_rejected = stale_rejected_r;
assign timeout_hit = timeout_hit_r;
assign accepted_model_data = accepted_model_data_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        accepted_r <= 1'b0;
        validation_failed_r <= 1'b0;
        stale_rejected_r <= 1'b0;
        timeout_hit_r <= 1'b0;
        accepted_model_data_r <= 128'h00000000000000000000000000000000;
        rsp_ready_r <= 1'b0;
    end else begin
        accepted_r <= 1'b0;
        validation_failed_r <= 1'b0;
        stale_rejected_r <= 1'b0;
        timeout_hit_r <= 1'b0;
        if (rsp_valid && cfg_enable && model_present) begin
            if ((watchdog_count >= cfg_request_timeout) && (cfg_request_timeout != 16'h0000)) begin
                timeout_hit_r <= 1'b1;
                validation_failed_r <= 1'b1;
            end else if ((watchdog_count >= cfg_response_age_limit) && (cfg_response_age_limit != 16'h0000)) begin
                stale_rejected_r <= 1'b1;
                validation_failed_r <= 1'b1;
            end else if ((rsp_data[127:112] == transaction_id_expected) && (rsp_data[111:96] == sequence_counter_expected)) begin
                accepted_r <= 1'b1;
                accepted_model_data_r <= rsp_data;
            end else begin
                validation_failed_r <= 1'b1;
            end
        end
        rsp_ready_r <= cfg_enable & model_present;
    end
end

endmodule

module adaptive_aero_command_mapper (
    clk,
    rst_n,
    accepted,
    accepted_model_data,
    cfg_max_act_cmd,
    cfg_min_act_cmd,
    cfg_fallback_cmd,
    mapped_cmd,
    mapped_cmd_valid
);
input clk;
input rst_n;
input accepted;
input [127:0] accepted_model_data;
input [31:0] cfg_max_act_cmd;
input [31:0] cfg_min_act_cmd;
input [31:0] cfg_fallback_cmd;
output [31:0] mapped_cmd;
output mapped_cmd_valid;

reg [31:0] mapped_cmd_r;
reg mapped_cmd_valid_r;
reg [31:0] raw_cmd;

assign mapped_cmd = mapped_cmd_r;
assign mapped_cmd_valid = mapped_cmd_valid_r;

always @(*) begin
    raw_cmd = cfg_fallback_cmd;
    if (accepted) begin
        raw_cmd = accepted_model_data[31:0] ^ accepted_model_data[63:32] ^ accepted_model_data[95:64] ^ accepted_model_data[127:96];
    end
end

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        mapped_cmd_r <= 32'h00000000;
        mapped_cmd_valid_r <= 1'b0;
    end else begin
        if (accepted) begin
            mapped_cmd_r <= raw_cmd;
            mapped_cmd_valid_r <= 1'b1;
        end else begin
            mapped_cmd_r <= mapped_cmd_r;
            mapped_cmd_valid_r <= 1'b0;
        end
    end
end

endmodule

module adaptive_aero_safety_gate (
    clk,
    rst_n,
    enable,
    validation_failed,
    stale_rejected,
    timeout_hit,
    accepted,
    mapped_cmd,
    mapped_cmd_valid,
    cfg_min_act_cmd,
    cfg_max_act_cmd,
    cfg_fallback_cmd,
    cfg_soft_reset,
    cfg_enable,
    cfg_status_clear,
    cfg_issue_request,
    busy,
    ready_for_request,
    response_pending,
    response_accepted,
    saturation,
    fallback_active,
    status_word,
    act_cmd_out,
    act_cmd_valid_out
);
input clk;
input rst_n;
input enable;
input validation_failed;
input stale_rejected;
input timeout_hit;
input accepted;
input [31:0] mapped_cmd;
input mapped_cmd_valid;
input [31:0] cfg_min_act_cmd;
input [31:0] cfg_max_act_cmd;
input [31:0] cfg_fallback_cmd;
input cfg_soft_reset;
input cfg_enable;
input cfg_status_clear;
input cfg_issue_request;
input busy;
input ready_for_request;
input response_pending;
input response_accepted;
output saturation;
output fallback_active;
output [31:0] status_word;
output [31:0] act_cmd_out;
output act_cmd_valid_out;

reg saturation_r;
reg fallback_active_r;
reg [31:0] status_word_r;
reg [31:0] act_cmd_out_r;
reg act_cmd_valid_out_r;
reg [31:0] clamped_cmd;
reg clamped_saturation;
reg [31:0] safe_cmd;

assign saturation = saturation_r;
assign fallback_active = fallback_active_r;
assign status_word = status_word_r;
assign act_cmd_out = act_cmd_out_r;
assign act_cmd_valid_out = act_cmd_valid_out_r;

always @(*) begin
    clamped_cmd = mapped_cmd;
    clamped_saturation = 1'b0;
    if (mapped_cmd < cfg_min_act_cmd) begin
        clamped_cmd = cfg_min_act_cmd;
        clamped_saturation = 1'b1;
    end else if (mapped_cmd > cfg_max_act_cmd) begin
        clamped_cmd = cfg_max_act_cmd;
        clamped_saturation = 1'b1;
    end
    safe_cmd = cfg_fallback_cmd;
    if (enable && cfg_enable && mapped_cmd_valid && !validation_failed && !stale_rejected && !timeout_hit) begin
        safe_cmd = clamped_cmd;
    end
end

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        saturation_r <= 1'b0;
        fallback_active_r <= 1'b1;
        status_word_r <= 32'h00000002;
        act_cmd_out_r <= 32'h00000000;
        act_cmd_valid_out_r <= 1'b0;
    end else begin
        if (cfg_status_clear) begin
            saturation_r <= 1'b0;
        end
        if (cfg_soft_reset || !enable || validation_failed || stale_rejected || timeout_hit) begin
            fallback_active_r <= 1'b1;
            act_cmd_out_r <= cfg_fallback_cmd;
            act_cmd_valid_out_r <= 1'b0;
        end else if (accepted && mapped_cmd_valid) begin
            fallback_active_r <= 1'b0;
            act_cmd_out_r <= safe_cmd;
            act_cmd_valid_out_r <= 1'b1;
        end else begin
            act_cmd_out_r <= act_cmd_out_r;
            act_cmd_valid_out_r <= act_cmd_valid_out_r & enable & cfg_enable;
        end
        if (clamped_saturation) begin
            saturation_r <= 1'b1;
        end
        if (cfg_status_clear) begin
            fallback_active_r <= fallback_active_r & ~cfg_issue_request;
        end
        status_word_r[0] <= enable & cfg_enable;
        status_word_r[1] <= busy;
        status_word_r[2] <= ready_for_request;
        status_word_r[3] <= response_pending;
        status_word_r[4] <= response_accepted;
        status_word_r[5] <= validation_failed;
        status_word_r[6] <= stale_rejected;
        status_word_r[7] <= timeout_hit;
        status_word_r[8] <= saturation_r | clamped_saturation;
        status_word_r[9] <= fallback_active_r;
        status_word_r[10] <= cfg_enable;
        status_word_r[11] <= cfg_soft_reset;
        status_word_r[12] <= cfg_status_clear;
        status_word_r[13] <= cfg_issue_request;
        status_word_r[14] <= mapped_cmd_valid;
        status_word_r[15] <= accepted;
        status_word_r[31:16] <= 16'h0000;
    end
end

endmodule
---END adaptive_aero_safety_gate.v---
