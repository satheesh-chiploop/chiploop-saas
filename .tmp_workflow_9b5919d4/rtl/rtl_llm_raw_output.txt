---BEGIN pwm_controller.v---
module pwm_controller (
    clk,
    reset_n,
    enable,
    duty_cycle,
    period,
    pwm_out,
    counter_value
);

input clk;
input reset_n;
input enable;
input [7:0] duty_cycle;
input [7:0] period;
output pwm_out;
output [7:0] counter_value;

reg [7:0] counter_r;
wire [7:0] counter_next;

assign counter_value = counter_r;
assign pwm_out = (counter_r < duty_cycle) ? 1'b1 : 1'b0;
assign counter_next = (!reset_n) ? 8'h00 :
                      (enable ? ((counter_r == period) ? 8'h00 : (counter_r + 8'h01)) :
                                counter_r);

always @(posedge clk) begin
    counter_r <= counter_next;
end

endmodule
---END pwm_controller.v---
