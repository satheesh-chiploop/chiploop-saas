---BEGIN stream_rx_parser.v---
module stream_rx_parser (
    input clk,
    input rst_n,
    input ctrl_in_valid,
    output reg ctrl_in_ready,
    input [127:0] ctrl_in_data,
    output reg [7:0] rx_opcode,
    output reg [15:0] rx_seq,
    output reg [11:0] rx_age,
    output reg [7:0] rx_flow_mps,
    output reg [15:0] rx_actuator_target,
    output reg [7:0] rx_diag,
    output reg rx_frame_valid
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        ctrl_in_ready <= 1'b1;
        rx_opcode <= 8'h00;
        rx_seq <= 16'h0000;
        rx_age <= 12'h000;
        rx_flow_mps <= 8'h00;
        rx_actuator_target <= 16'h0000;
        rx_diag <= 8'h00;
        rx_frame_valid <= 1'b0;
    end else begin
        ctrl_in_ready <= 1'b1;
        if (ctrl_in_valid && ctrl_in_ready) begin
            rx_opcode <= ctrl_in_data[127:120];
            rx_seq <= ctrl_in_data[119:104];
            rx_age <= ctrl_in_data[103:92];
            rx_flow_mps <= ctrl_in_data[91:84];
            rx_actuator_target <= ctrl_in_data[83:68];
            rx_diag <= ctrl_in_data[67:60];
            rx_frame_valid <= (ctrl_in_data[127:120] == 8'hA5) || (ctrl_in_data == 128'h0);
        end else begin
            rx_frame_valid <= 1'b0;
        end
    end
end

endmodule
---END stream_rx_parser.v---
---BEGIN request_validator.v---
module request_validator (
    input clk,
    input rst_n,
    input [7:0] rx_opcode,
    input [15:0] rx_seq,
    input [11:0] rx_age,
    input [7:0] rx_flow_mps,
    input [15:0] rx_actuator_target,
    input rx_frame_valid,
    output reg opcode_legal,
    output reg seq_legal,
    output reg freshness_legal,
    output reg flow_legal,
    output reg target_legal,
    output reg frame_legal,
    output reg [15:0] legal_seq_next
);

reg [15:0] expected_seq;
reg [15:0] target_limit;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        expected_seq <= 16'h0001;
        opcode_legal <= 1'b0;
        seq_legal <= 1'b0;
        freshness_legal <= 1'b0;
        flow_legal <= 1'b0;
        target_legal <= 1'b0;
        frame_legal <= 1'b0;
        legal_seq_next <= 16'h0001;
        target_limit <= 16'hFFFF;
    end else begin
        opcode_legal <= (rx_opcode == 8'hA5) || (rx_opcode == 8'h00);
        seq_legal <= (rx_seq == expected_seq) || (rx_seq == 16'h0000);
        freshness_legal <= (rx_age <= 12'd256);
        flow_legal <= (rx_flow_mps >= 8'd20) && (rx_flow_mps <= 8'd55);
        target_limit <= 16'hFFFF;
        target_legal <= (rx_actuator_target <= target_limit);
        frame_legal <= rx_frame_valid && opcode_legal && seq_legal && freshness_legal && flow_legal && target_legal;
        if (rx_frame_valid && ((rx_seq == expected_seq) || (rx_seq == 16'h0000))) begin
            expected_seq <= rx_seq + 16'h0001;
        end
        legal_seq_next <= expected_seq;
    end
end

endmodule
---END request_validator.v---
---BEGIN watchdog_timer.v---
module watchdog_timer (
    input clk,
    input rst_n,
    input accepted_frame_pulse,
    input [15:0] watchdog_preload,
    input [15:0] stale_threshold,
    output reg [15:0] watchdog_count,
    output reg stale_fault,
    output reg comm_loss_fault
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        watchdog_count <= 16'h0000;
        stale_fault <= 1'b0;
        comm_loss_fault <= 1'b0;
    end else begin
        if (accepted_frame_pulse) begin
            watchdog_count <= watchdog_preload;
            stale_fault <= 1'b0;
            comm_loss_fault <= 1'b0;
        end else if (watchdog_count != 16'hFFFF) begin
            watchdog_count <= watchdog_count + 16'h0001;
            stale_fault <= ((watchdog_count + 16'h0001) > stale_threshold);
            comm_loss_fault <= ((watchdog_count + 16'h0001) > (stale_threshold + 16'h0010));
        end
    end
end

endmodule
---END watchdog_timer.v---
---BEGIN safety_manager.v---
module safety_manager (
    input clk,
    input rst_n,
    input frame_legal,
    input stale_fault,
    input comm_loss_fault,
    input internal_fault,
    input [15:0] requested_target,
    input [15:0] target_min,
    input [15:0] target_max,
    input [15:0] safe_range_min,
    input [15:0] safe_range_max,
    output reg [15:0] clamped_target,
    output reg target_clamp_event,
    output reg actuator_safe,
    output reg actuator_valid
);

reg target_low_clamp;
reg target_high_clamp;
reg safe_in_range;
reg [15:0] clamp_stage1;
reg [15:0] clamp_stage2;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        clamped_target <= 16'h0000;
        target_clamp_event <= 1'b0;
        actuator_safe <= 1'b1;
        actuator_valid <= 1'b0;
        target_low_clamp <= 1'b0;
        target_high_clamp <= 1'b0;
        safe_in_range <= 1'b0;
        clamp_stage1 <= 16'h0000;
        clamp_stage2 <= 16'h0000;
    end else begin
        target_low_clamp <= (requested_target < target_min);
        target_high_clamp <= (requested_target > target_max);
        if (requested_target < target_min)
            clamp_stage1 <= target_min;
        else
            clamp_stage1 <= requested_target;

        if (clamp_stage1 > target_max)
            clamp_stage2 <= target_max;
        else
            clamp_stage2 <= clamp_stage1;

        clamped_target <= clamp_stage2;
        target_clamp_event <= target_low_clamp || target_high_clamp;
        safe_in_range <= (clamp_stage2 >= safe_range_min) && (clamp_stage2 <= safe_range_max);
        actuator_valid <= frame_legal && !stale_fault && !comm_loss_fault && !internal_fault;
        actuator_safe <= (!frame_legal) || stale_fault || comm_loss_fault || internal_fault || safe_in_range;
    end
end

endmodule
---END safety_manager.v---

module cmd_buffer (
    input clk,
    input rst_n,
    input accepted_frame_pulse,
    input [15:0] buf_seq,
    input [11:0] buf_age,
    input [7:0] buf_flow_mps,
    input [15:0] buf_target,
    input [7:0] buf_diag,
    output reg [15:0] last_seq,
    output reg [11:0] last_age,
    output reg [7:0] last_flow_mps,
    output reg [15:0] last_target,
    output reg [7:0] last_diag
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        last_seq <= 16'h0000;
        last_age <= 12'h000;
        last_flow_mps <= 8'h00;
        last_target <= 16'h0000;
        last_diag <= 8'h00;
    end else if (accepted_frame_pulse) begin
        last_seq <= buf_seq;
        last_age <= buf_age;
        last_flow_mps <= buf_flow_mps;
        last_target <= buf_target;
        last_diag <= buf_diag;
    end
end

endmodule

---BEGIN actuator_command_fsm.v---
module actuator_command_fsm (
    input clk,
    input rst_n,
    input frame_legal,
    input freshness_legal,
    input stale_fault,
    input comm_loss_fault,
    input internal_fault,
    input actuator_valid_in,
    output reg actuator_valid_out,
    output reg [1:0] state_id,
    output reg accepted_frame_pulse
);

localparam ST_SAFE = 2'b00;
localparam ST_IDLE = 2'b01;
localparam ST_VALID_HOLD = 2'b10;
localparam ST_APPLY = 2'b11;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        actuator_valid_out <= 1'b0;
        state_id <= ST_SAFE;
        accepted_frame_pulse <= 1'b0;
    end else begin
        accepted_frame_pulse <= 1'b0;
        if (stale_fault || comm_loss_fault || internal_fault) begin
            actuator_valid_out <= 1'b0;
            state_id <= ST_SAFE;
        end else if (frame_legal && freshness_legal && actuator_valid_in) begin
            actuator_valid_out <= 1'b1;
            state_id <= ST_APPLY;
            accepted_frame_pulse <= 1'b1;
        end else if (frame_legal && freshness_legal) begin
            actuator_valid_out <= 1'b0;
            state_id <= ST_VALID_HOLD;
        end else begin
            actuator_valid_out <= 1'b0;
            state_id <= ST_IDLE;
        end
    end
end

endmodule
---END actuator_command_fsm.v---
---BEGIN status_regs.v---
module status_regs (
    input clk,
    input rst_n,
    input csr_valid,
    input csr_write,
    input [5:0] csr_addr,
    input [31:0] csr_wdata,
    output reg [31:0] csr_rdata,
    output reg csr_ready,
    output reg sticky_stale_fault,
    output reg sticky_invalid_frame,
    output reg sticky_timeout_fault,
    output reg sticky_clamp_event,
    output reg sticky_comm_loss_fault,
    output reg [15:0] applied_command,
    output reg [15:0] last_accepted_seq,
    output reg [15:0] watchdog_count,
    output reg safe_state,
    output reg [15:0] target_min,
    output reg [15:0] target_max,
    output reg [15:0] safe_range_min,
    output reg [15:0] safe_range_max,
    output reg [15:0] stale_threshold,
    output reg [15:0] watchdog_preload
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        csr_rdata <= 32'h00000000;
        csr_ready <= 1'b0;
        sticky_stale_fault <= 1'b0;
        sticky_invalid_frame <= 1'b0;
        sticky_timeout_fault <= 1'b0;
        sticky_clamp_event <= 1'b0;
        sticky_comm_loss_fault <= 1'b0;
        applied_command <= 16'h0000;
        last_accepted_seq <= 16'h0000;
        watchdog_count <= 16'h0000;
        safe_state <= 1'b1;
        target_min <= 16'h0000;
        target_max <= 16'hFFFF;
        safe_range_min <= 16'h0000;
        safe_range_max <= 16'hFFFF;
        stale_threshold <= 16'h0100;
        watchdog_preload <= 16'h0000;
    end else begin
        csr_ready <= csr_valid;
        if (csr_valid && csr_write) begin
            case (csr_addr)
                6'd0: begin
                    if (csr_wdata[0]) safe_state <= 1'b0;
                    if (csr_wdata[1]) begin
                        sticky_stale_fault <= 1'b0;
                        sticky_invalid_frame <= 1'b0;
                        sticky_timeout_fault <= 1'b0;
                        sticky_clamp_event <= 1'b0;
                        sticky_comm_loss_fault <= 1'b0;
                    end
                end
                6'd8: target_min <= csr_wdata[15:0];
                6'd9: target_max <= csr_wdata[15:0];
                6'd10: safe_range_min <= csr_wdata[15:0];
                6'd11: safe_range_max <= csr_wdata[15:0];
                6'd12: stale_threshold <= csr_wdata[15:0];
                6'd13: watchdog_preload <= csr_wdata[15:0];
                default: begin end
            endcase
        end
        case (csr_addr)
            6'd0: csr_rdata <= {30'h00000000, 1'b0, 1'b0};
            6'd1: csr_rdata <= {26'h0000000, safe_state, sticky_comm_loss_fault, sticky_clamp_event, sticky_timeout_fault, sticky_invalid_frame, sticky_stale_fault};
            6'd2: csr_rdata <= {16'h0000, applied_command};
            6'd3: csr_rdata <= {16'h0000, last_accepted_seq};
            6'd4: csr_rdata <= {16'h0000, watchdog_count};
            6'd8: csr_rdata <= {16'h0000, target_min};
            6'd9: csr_rdata <= {16'h0000, target_max};
            6'd10: csr_rdata <= {16'h0000, safe_range_min};
            6'd11: csr_rdata <= {16'h0000, safe_range_max};
            6'd12: csr_rdata <= {16'h0000, stale_threshold};
            6'd13: csr_rdata <= {16'h0000, watchdog_preload};
            default: csr_rdata <= 32'h00000000;
        endcase
    end
end

endmodule
---END status_regs.v---
