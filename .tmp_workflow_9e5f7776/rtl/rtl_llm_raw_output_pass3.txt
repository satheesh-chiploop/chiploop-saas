output reg model_online;

reg [31:0] age_calc;

always @(*) begin
    age_calc = 32'd0;
    if (rsp_timestamp >= req_timestamp)
        age_calc = rsp_timestamp[31:0] - req_timestamp[31:0];
end

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        accepted_response_sequence_id <= 32'd0;
        accepted_response_timestamp <= 64'd0;
        accepted_response_age_cycles <= 32'd0;
        accepted_response_drag_force <= 32'd0;
        accepted_response_lift_force <= 32'd0;
        accepted_response_surface_pressure <= 32'd0;
        accepted_response_flow_field_valid <= 1'b0;
        accepted_response_flow_field <= 64'd0;
        response_validated <= 1'b0;
        stale_reject <= 1'b0;
        timeout <= 1'b0;
        out_of_range <= 1'b0;
        model_online <= 1'b0;
    end else begin
        model_online <= model_online_in & enable;
        response_validated <= 1'b0;
        stale_reject <= 1'b0;
        timeout <= 1'b0;
        out_of_range <= 1'b0;
        if (enable && req_valid && rsp_valid && rsp_frame_valid && (rsp_sequence_id == req_sequence_id) && rsp_flow_field_valid && model_online_in) begin
            if ((stream_velocity_mps >= 16'd20) && (stream_velocity_mps <= 16'd55)) begin
                if ((timeout_threshold_cycles == 32'd0) || (age_calc <= timeout_threshold_cycles) || (age_calc <= latency_budget_cycles)) begin
                    accepted_response_sequence_id <= rsp_sequence_id;
                    accepted_response_timestamp <= rsp_timestamp;
                    accepted_response_age_cycles <= age_calc;
                    accepted_response_drag_force <= rsp_drag_force;
                    accepted_response_lift_force <= rsp_lift_force;
                    accepted_response_surface_pressure <= rsp_surface_pressure;
                    accepted_response_flow_field_valid <= rsp_flow_field_valid;
                    accepted_response_flow_field <= rsp_flow_field;
                    response_validated <= 1'b1;
                end else begin
                    timeout <= 1'b1;
                    stale_reject <= 1'b1;
                end
            end else begin
                out_of_range <= 1'b1;
                stale_reject <= 1'b1;
            end
        end else if (enable && rsp_valid && rsp_frame_valid) begin
            stale_reject <= 1'b1;
        end
    end
end

endmodule

module domino_control_safety_fsm (
    clk,
    reset_n,
    enable,
    clear_fault,
    cmd_min,
    cmd_max,
    safe_state,
    accepted_response_sequence_id,
    accepted_response_timestamp,
    accepted_response_age_cycles,
    accepted_response_drag_force,
    accepted_response_lift_force,
    accepted_response_surface_pressure,
    accepted_response_flow_field_valid,
    accepted_response_flow_field,
    response_validated,
    stale_reject,
    timeout,
    out_of_range,
    actuator_command_next,
    actuator_command,
    actuator_command_valid,
    fallback_active,
    fault_latched,
    fsm_state,
    compute_control_valid
);
input clk;
input reset_n;
input enable;
input clear_fault;
input [31:0] cmd_min;
input [31:0] cmd_max;
input [31:0] safe_state;
input [31:0] accepted_response_sequence_id;
input [63:0] accepted_response_timestamp;
input [31:0] accepted_response_age_cycles;
input [31:0] accepted_response_drag_force;
input [31:0] accepted_response_lift_force;
input [31:0] accepted_response_surface_pressure;
input accepted_response_flow_field_valid;
input [63:0] accepted_response_flow_field;
input response_validated;
input stale_reject;
input timeout;
input out_of_range;
output reg [31:0] actuator_command_next;
output reg [31:0] actuator_command;
output reg actuator_command_valid;
output reg fallback_active;
output reg fault_latched;
output reg [2:0] fsm_state;
output reg compute_control_valid;

localparam [2:0] IDLE = 3'd0;
localparam [2:0] REQUEST_SENT = 3'd1;
localparam [2:0] WAIT_RESPONSE = 3'd2;
localparam [2:0] VALIDATE_RESPONSE = 3'd3;
localparam [2:0] COMPUTE_CONTROL = 3'd4;
localparam [2:0] APPLY_CLAMP = 3'd5;
localparam [2:0] FALLBACK = 3'd6;
localparam [2:0] FAULT_LATCH = 3'd7;

reg [2:0] state_next;
reg [31:0] clamp_lo;
reg [31:0] clamp_hi;
reg [31:0] control_candidate;
reg [31:0] clamp_value;

always @(*) begin
    state_next = fsm_state;
    clamp_lo = cmd_min;
    clamp_hi = cmd_max;
    control_candidate = accepted_response_drag_force ^ accepted_response_lift_force ^ accepted_response_surface_pressure ^ {31'b0, accepted_response_flow_field_valid};
    clamp_value = control_candidate;
    actuator_command_next = control_candidate;
    actuator_command_valid = 1'b0;
    fallback_active = 1'b0;
    compute_control_valid = 1'b0;
    if (fault_latched && !clear_fault) begin
        state_next = FAULT_LATCH;
        fallback_active = 1'b1;
        actuator_command_next = safe_state;
    end else if (enable && (out_of_range || timeout || stale_reject)) begin
        state_next = FALLBACK;
        fallback_active = 1'b1;
        actuator_command_next = safe_state;
    end else if (enable && response_validated) begin
        state_next = APPLY_CLAMP;
        compute_control_valid = 1'b1;
        if (control_candidate < clamp_lo)
            clamp_value = clamp_lo;
        else if (control_candidate > clamp_hi)
            clamp_value = clamp_hi;
        else
            clamp_value = control_candidate;
        actuator_command_next = clamp_value;
    end else begin
        state_next = IDLE;
        actuator_command_next = safe_state;
    end
end

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        fsm_state <= IDLE;
        actuator_command <= 32'd0;
        actuator_command_valid <= 1'b0;
        fallback_active <= 1'b1;
        fault_latched <= 1'b0;
    end else begin
        fsm_state <= state_next;
        if (clear_fault)
            fault_latched <= 1'b0;
        else if (timeout || out_of_range || stale_reject)
            fault_latched <= 1'b1;
        actuator_command <= actuator_command_next;
        if (fallback_active || compute_control_valid)
            actuator_command_valid <= 1'b1;
        else
            actuator_command_valid <= 1'b0;
        if (fallback_active)
            actuator_command <= safe_state;
    end
end

endmodule
