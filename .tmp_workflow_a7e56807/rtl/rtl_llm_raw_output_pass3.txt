---BEGIN csr_block.v---
module csr_block (
    input clk,
    input rst_n,
    input reg_req_valid,
    input reg_req_write,
    input [7:0] reg_req_addr,
    input [31:0] reg_req_wdata,
    output reg_req_ready,
    output reg_rsp_valid,
    output [31:0] reg_rsp_rdata,
    output reg_rsp_error,
    output cfg_enable,
    output [1:0] cfg_fallback_select,
    output [15:0] cfg_timeout_threshold,
    output [15:0] cfg_cmd_min,
    output [15:0] cfg_cmd_max,
    output [15:0] cfg_fallback_value,
    output sticky_fault_clear,
    input [2:0] status_fsm_state,
    input [7:0] status_outstanding_seq,
    input [3:0] status_last_fault_reason,
    input status_safe_fallback_active,
    input [7:0] status_fault_sticky
);

reg cfg_enable_r;
reg [1:0] cfg_fallback_select_r;
reg [15:0] cfg_timeout_threshold_r;
reg [15:0] cfg_cmd_min_r;
reg [15:0] cfg_cmd_max_r;
reg [15:0] cfg_fallback_value_r;
reg sticky_fault_clear_r;
reg reg_req_ready_r;
reg reg_rsp_valid_r;
reg [31:0] reg_rsp_rdata_r;
reg reg_rsp_error_r;

assign cfg_enable = cfg_enable_r;
assign cfg_fallback_select = cfg_fallback_select_r;
assign cfg_timeout_threshold = cfg_timeout_threshold_r;
assign cfg_cmd_min = cfg_cmd_min_r;
assign cfg_cmd_max = cfg_cmd_max_r;
assign cfg_fallback_value = cfg_fallback_value_r;
assign sticky_fault_clear = sticky_fault_clear_r;
assign reg_req_ready = reg_req_ready_r;
assign reg_rsp_valid = reg_rsp_valid_r;
assign reg_rsp_rdata = reg_rsp_rdata_r;
assign reg_rsp_error = reg_rsp_error_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        cfg_enable_r <= 1'b0;
        cfg_fallback_select_r <= 2'b00;
        cfg_timeout_threshold_r <= 16'd100;
        cfg_cmd_min_r <= 16'd0;
        cfg_cmd_max_r <= 16'd4095;
        cfg_fallback_value_r <= 16'd0;
        sticky_fault_clear_r <= 1'b0;
        reg_req_ready_r <= 1'b1;
        reg_rsp_valid_r <= 1'b0;
        reg_rsp_rdata_r <= 32'd0;
        reg_rsp_error_r <= 1'b0;
    end else begin
        reg_rsp_valid_r <= reg_req_valid;
        reg_req_ready_r <= 1'b1;
        sticky_fault_clear_r <= 1'b0;
        reg_rsp_error_r <= 1'b0;
        reg_rsp_rdata_r <= 32'd0;
        if (reg_req_valid) begin
            if (reg_req_write) begin
                case (reg_req_addr)
                    8'h00: cfg_enable_r <= reg_req_wdata[0];
                    8'h01: cfg_fallback_select_r <= reg_req_wdata[1:0];
                    8'h02: cfg_timeout_threshold_r <= reg_req_wdata[15:0];
                    8'h03: cfg_cmd_min_r <= reg_req_wdata[15:0];
                    8'h04: cfg_cmd_max_r <= reg_req_wdata[15:0];
                    8'h05: cfg_fallback_value_r <= reg_req_wdata[15:0];
                    8'h15: sticky_fault_clear_r <= reg_req_wdata[0];
                    default: reg_rsp_error_r <= 1'b1;
                endcase
            end else begin
                case (reg_req_addr)
                    8'h00: reg_rsp_rdata_r <= {31'd0, cfg_enable_r};
                    8'h01: reg_rsp_rdata_r <= {30'd0, cfg_fallback_select_r};
                    8'h02: reg_rsp_rdata_r <= {16'd0, cfg_timeout_threshold_r};
                    8'h03: reg_rsp_rdata_r <= {16'd0, cfg_cmd_min_r};
                    8'h04: reg_rsp_rdata_r <= {16'd0, cfg_cmd_max_r};
                    8'h05: reg_rsp_rdata_r <= {16'd0, cfg_fallback_value_r};
                    8'h10: reg_rsp_rdata_r <= {29'd0, status_fsm_state};
                    8'h11: reg_rsp_rdata_r <= {24'd0, status_outstanding_seq};
                    8'h12: reg_rsp_rdata_r <= {28'd0, status_last_fault_reason};
                    8'h13: reg_rsp_rdata_r <= {31'd0, status_safe_fallback_active};
                    8'h14: reg_rsp_rdata_r <= {24'd0, status_fault_sticky};
                    8'h15: reg_rsp_rdata_r <= 32'd0;
                    default: reg_rsp_error_r <= 1'b1;
                endcase
            end
        end
    end
end

endmodule
---END csr_block.v---
---BEGIN request_packager.v---
module request_packager (
    input clk,
    input rst_n,
    input cmd_valid,
    output cmd_ready,
    input [31:0] cmd_payload,
    output capture_request,
    output [7:0] req_sequence,
    output [31:0] req_descriptor,
    output req_descriptor_valid,
    input cfg_enable,
    input [15:0] cfg_cmd_min,
    input [15:0] cfg_cmd_max,
    input outstanding_busy,
    input fault_block
);

reg [7:0] req_sequence_r;
reg capture_request_r;
reg [31:0] req_descriptor_r;
reg req_descriptor_valid_r;
reg cmd_ready_r;

assign req_sequence = req_sequence_r;
assign capture_request = capture_request_r;
assign req_descriptor = req_descriptor_r;
assign req_descriptor_valid = req_descriptor_valid_r;
assign cmd_ready = cmd_ready_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        req_sequence_r <= 8'd0;
        capture_request_r <= 1'b0;
        req_descriptor_r <= 32'd0;
        req_descriptor_valid_r <= 1'b0;
        cmd_ready_r <= 1'b0;
    end else begin
        capture_request_r <= 1'b0;
        req_descriptor_valid_r <= 1'b0;
        cmd_ready_r <= cfg_enable & ~outstanding_busy & ~fault_block;
        if (cmd_valid && cmd_ready_r) begin
            req_sequence_r <= req_sequence_r + 8'd1;
            capture_request_r <= 1'b1;
            req_descriptor_r <= {cfg_cmd_max[7:0], cfg_cmd_min[7:0], cmd_payload[7:0], 8'hA5};
            req_descriptor_valid_r <= 1'b1;
        end
    end
end

endmodule
---END request_packager.v---
---BEGIN response_validator.v---
module response_validator (
    input clk,
    input rst_n,
    input resp_valid,
    output resp_ready,
    input [31:0] resp_payload,
    input [7:0] outstanding_seq,
    input [15:0] timeout_age,
    input [15:0] timeout_threshold,
    input [15:0] cfg_cmd_min,
    input [15:0] cfg_cmd_max,
    input [7:0] expected_seq,
    output [15:0] validated_command,
    output validated_valid,
    output fault_stale,
    output fault_sequence_mismatch,
    output fault_range_violation
);

reg [15:0] validated_command_r;
reg validated_valid_r;
reg fault_stale_r;
reg fault_sequence_mismatch_r;
reg fault_range_violation_r;

assign resp_ready = 1'b1;
assign validated_command = validated_command_r;
assign validated_valid = validated_valid_r;
assign fault_stale = fault_stale_r;
assign fault_sequence_mismatch = fault_sequence_mismatch_r;
assign fault_range_violation = fault_range_violation_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        validated_command_r <= 16'd0;
        validated_valid_r <= 1'b0;
        fault_stale_r <= 1'b0;
        fault_sequence_mismatch_r <= 1'b0;
        fault_range_violation_r <= 1'b0;
    end else begin
        validated_valid_r <= 1'b0;
        fault_stale_r <= 1'b0;
        fault_sequence_mismatch_r <= 1'b0;
        fault_range_violation_r <= 1'b0;
        if (resp_valid) begin
            if (expected_seq != outstanding_seq) begin
                fault_sequence_mismatch_r <= 1'b1;
            end else if (timeout_age > timeout_threshold) begin
                fault_stale_r <= 1'b1;
            end else if ((resp_payload[15:0] < cfg_cmd_min) || (resp_payload[15:0] > cfg_cmd_max)) begin
                fault_range_violation_r <= 1'b1;
            end else begin
                validated_command_r <= resp_payload[15:0];
                validated_valid_r <= 1'b1;
            end
        end
    end
end

endmodule
---END response_validator.v---
---BEGIN safety_supervisor.v---
module safety_supervisor (
    input clk,
    input rst_n,
    input cfg_enable,
    input [1:0] cfg_fallback_select,
    input [15:0] cfg_timeout_threshold,
    input [15:0] cfg_cmd_min,
    input [15:0] cfg_cmd_max,
    input [15:0] cfg_fallback_value,
    input request_captured,
    input [7:0] request_sequence,
    input response_validated,
    input [15:0] validated_command,
    input fault_stale,
    input fault_sequence_mismatch,
    input fault_range_violation,
    input [15:0] timeout_age,
    input [15:0] timeout_threshold,
    output [2:0] fsm_state,
    output [7:0] outstanding_seq,
    output [3:0] last_fault_reason,
    output safe_fallback_active,
    output fault_flag,
    output [15:0] actuator_setpoint_int,
    output actuator_valid_int,
    output cmd_ready_int
);

localparam [2:0] ST_RESET = 3'd0;
localparam [2:0] ST_IDLE = 3'd1;
localparam [2:0] ST_ISSUE_REQUEST = 3'd2;
localparam [2:0] ST_WAIT_RESPONSE = 3'd3;
localparam [2:0] ST_APPLY_VALIDATED_COMMAND = 3'd4;
localparam [2:0] ST_SAFE_FALLBACK = 3'd5;
localparam [2:0] ST_FAULT_HOLD = 3'd6;

reg [2:0] fsm_state_r;
reg [7:0] outstanding_seq_r;
reg [3:0] last_fault_reason_r;
reg safe_fallback_active_r;
reg fault_flag_r;
reg [15:0] actuator_setpoint_int_r;
reg actuator_valid_int_r;
reg cmd_ready_int_r;

assign fsm_state = fsm_state_r;
assign outstanding_seq = outstanding_seq_r;
assign last_fault_reason = last_fault_reason_r;
assign safe_fallback_active = safe_fallback_active_r;
assign fault_flag = fault_flag_r;
assign actuator_setpoint_int = actuator_setpoint_int_r;
assign actuator_valid_int = actuator_valid_int_r;
assign cmd_ready_int = cmd_ready_int_r;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        fsm_state_r <= ST_RESET;
        outstanding_seq_r <= 8'd0;
        last_fault_reason_r <= 4'd0;
        safe_fallback_active_r <= 1'b1;
        fault_flag_r <= 1'b0;
        actuator_setpoint_int_r <= 16'd0;
        actuator_valid_int_r <= 1'b0;
        cmd_ready_int_r <= 1'b0;
    end else begin
        cmd_ready_int_r <= cfg_enable & ~fault_flag_r & ~safe_fallback_active_r;
        actuator_valid_int_r <= 1'b0;
        if (!cfg_enable) begin
            fsm_state_r <= ST_SAFE_FALLBACK;
            safe_fallback_active_r <= 1'b1;
            actuator_setpoint_int_r <= cfg_fallback_value;
        end else if (fault_stale || fault_sequence_mismatch || fault_range_violation || (cfg_cmd_min > cfg_cmd_max) || (cfg_timeout_threshold == 16'd0) || (timeout_age > timeout_threshold)) begin
            fsm_state_r <= ST_FAULT_HOLD;
            fault_flag_r <= 1'b1;
            safe_fallback_active_r <= 1'b1;
            actuator_setpoint_int_r <= cfg_fallback_value;
            last_fault_reason_r <= fault_stale ? 4'd1 : (fault_sequence_mismatch ? 4'd3 : (fault_range_violation ? 4'd4 : ((cfg_cmd_min > cfg_cmd_max) ? 4'd5 : ((cfg_timeout_threshold == 16'd0) ? 4'd5 : 4'd1))));
        end else if (request_captured) begin
            fsm_state_r <= ST_WAIT_RESPONSE;
            outstanding_seq_r <= request_sequence;
            safe_fallback_active_r <= 1'b0;
            actuator_setpoint_int_r <= cfg_fallback_value;
        end else if (response_validated) begin
            fsm_state_r <= ST_APPLY_VALIDATED_COMMAND;
            safe_fallback_active_r <= 1'b0;
            if (validated_command < cfg_cmd_min)
                actuator_setpoint_int_r <= cfg_cmd_min;
            else if (validated_command > cfg_cmd_max)
                actuator_setpoint_int_r <= cfg_cmd_max;
            else
                actuator_setpoint_int_r <= validated_command;
            actuator_valid_int_r <= 1'b1;
        end else begin
            fsm_state_r <= ST_IDLE;
            if (cfg_fallback_select == 2'b00)
                actuator_setpoint_int_r <= cfg_fallback_value;
            else if (cfg_fallback_select == 2'b01)
                actuator_setpoint_int_r <= cfg_cmd_min;
            else if (cfg_fallback_select == 2'b10)
                actuator_setpoint_int_r <= cfg_cmd_max;
            else
                actuator_setpoint_int_r <= {cfg_cmd_min[15:8], cfg_cmd_max[7:0]};
            safe_fallback_active_r <= 1'b1;
        end
        if (request_captured)
            fsm_state_r <= ST_ISSUE_REQUEST;
        if (cfg_enable && !fault_flag_r && !safe_fallback_active_r && !request_captured && !response_validated && (timeout_age <= timeout_threshold))
            cmd_ready_int_r <= 1'b1;
        else if (fault_flag_r)
            cmd_ready_int_r <= 1'b0;
        if (cfg_enable && request_captured)
            safe_fallback_active_r <= 1'b0;
        if (response_validated && !fault_flag_r)
            outstanding_seq_r <= request_sequence;
    end
end

endmodule
---END safety_supervisor.v---
