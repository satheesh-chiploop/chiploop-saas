---BEGIN adaptive_aero_control_csr.v---
module adaptive_aero_control_csr (
    clk,
    rst_n,
    csr_valid,
    csr_write,
    csr_addr,
    csr_wdata,
    csr_rdata,
    csr_ready,
    cfg_enable,
    cfg_enforce_sequence,
    cfg_stale_threshold,
    cfg_watchdog_limit,
    cfg_min_bound,
    cfg_max_bound,
    cfg_default_mode,
    cfg_clear_sticky,
    status_mode,
    status_last_sequence,
    status_watchdog_count,
    status_stale_fault,
    status_timeout_fault,
    status_clamp_fault,
    status_protocol_fault,
    status_safe_active
);
    input clk;
    input rst_n;
    input csr_valid;
    input csr_write;
    input [5:0] csr_addr;
    input [31:0] csr_wdata;
    output reg [31:0] csr_rdata;
    output reg csr_ready;

    output reg cfg_enable;
    output reg cfg_enforce_sequence;
    output reg [7:0] cfg_stale_threshold;
    output reg [15:0] cfg_watchdog_limit;
    output reg [15:0] cfg_min_bound;
    output reg [15:0] cfg_max_bound;
    output reg [3:0] cfg_default_mode;
    output reg cfg_clear_sticky;

    input [3:0] status_mode;
    input [15:0] status_last_sequence;
    input [15:0] status_watchdog_count;
    input status_stale_fault;
    input status_timeout_fault;
    input status_clamp_fault;
    input status_protocol_fault;
    input status_safe_active;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cfg_enable <= 1'b0;
            cfg_enforce_sequence <= 1'b1;
            cfg_stale_threshold <= 8'd8;
            cfg_watchdog_limit <= 16'd100;
            cfg_min_bound <= 16'd0;
            cfg_max_bound <= 16'd32767;
            cfg_default_mode <= 4'd0;
            cfg_clear_sticky <= 1'b0;
            csr_ready <= 1'b0;
            csr_rdata <= 32'd0;
        end else begin
            csr_ready <= csr_valid;
            cfg_clear_sticky <= 1'b0;
            if (csr_valid && csr_write) begin
                case (csr_addr)
                    6'd0: begin
                        cfg_enable <= csr_wdata[0];
                        cfg_enforce_sequence <= csr_wdata[1];
                        cfg_clear_sticky <= csr_wdata[2];
                    end
                    6'd1: begin
                        cfg_stale_threshold <= csr_wdata[7:0];
                    end
                    6'd2: begin
                        cfg_watchdog_limit <= csr_wdata[15:0];
                    end
                    6'd3: begin
                        cfg_min_bound <= csr_wdata[15:0];
                    end
                    6'd4: begin
                        cfg_max_bound <= csr_wdata[15:0];
                    end
                    6'd5: begin
                        cfg_default_mode <= csr_wdata[3:0];
                    end
                    default: begin
                    end
                endcase
            end
            if (csr_valid && !csr_write) begin
                case (csr_addr)
                    6'd0: csr_rdata <= {29'd0, cfg_clear_sticky, cfg_enforce_sequence, cfg_enable};
                    6'd1: csr_rdata <= {24'd0, cfg_stale_threshold};
                    6'd2: csr_rdata <= {16'd0, cfg_watchdog_limit};
                    6'd3: csr_rdata <= {16'd0, cfg_min_bound};
                    6'd4: csr_rdata <= {16'd0, cfg_max_bound};
                    6'd5: csr_rdata <= {28'd0, cfg_default_mode};
                    6'd8: csr_rdata <= {23'd0, status_clamp_fault, status_timeout_fault, status_stale_fault, status_protocol_fault, status_safe_active, status_mode};
                    6'd9: csr_rdata <= {16'd0, status_last_sequence};
                    6'd10: csr_rdata <= {16'd0, status_watchdog_count};
                    default: csr_rdata <= 32'd0;
                endcase
            end else begin
                csr_rdata <= csr_rdata;
            end
        end
    end
endmodule
---END adaptive_aero_control_csr.v---

---BEGIN adaptive_aero_control_transport.v---
module adaptive_aero_control_transport (
    clk,
    rst_n,
    req_valid,
    req_data,
    req_ready,
    rsp_valid,
    rsp_data,
    rsp_ready,
    cfg_enable,
    cfg_enforce_sequence,
    cfg_stale_threshold,
    cfg_watchdog_limit,
    cfg_min_bound,
    cfg_max_bound,
    cfg_default_mode,
    cfg_clear_sticky,
    status_mode,
    status_last_sequence,
    status_watchdog_count,
    status_stale_fault,
    status_timeout_fault,
    status_clamp_fault,
    status_protocol_fault,
    status_safe_active,
    cmd_valid,
    cmd_value,
    cmd_mode,
    cmd_fault,
    cmd_safe,
    payload_mem_csb,
    payload_mem_we,
    payload_mem_addr,
    payload_mem_din,
    payload_mem_dout
);
    input clk;
    input rst_n;
    input req_valid;
    input [127:0] req_data;
    output reg req_ready;
    output reg rsp_valid;
    output reg [127:0] rsp_data;
    input rsp_ready;
    input cfg_enable;
    input cfg_enforce_sequence;
    input [7:0] cfg_stale_threshold;
    input [15:0] cfg_watchdog_limit;
    input [15:0] cfg_min_bound;
    input [15:0] cfg_max_bound;
    input [3:0] cfg_default_mode;
    input cfg_clear_sticky;
    output reg [3:0] status_mode;
    output reg [15:0] status_last_sequence;
    output reg [15:0] status_watchdog_count;
    output reg status_stale_fault;
    output reg status_timeout_fault;
    output reg status_clamp_fault;
    output reg status_protocol_fault;
    output reg status_safe_active;
    output reg cmd_valid;
    output reg [15:0] cmd_value;
    output reg [3:0] cmd_mode;
    output reg cmd_fault;
    output reg cmd_safe;
    output reg payload_mem_csb;
    output reg payload_mem_we;
    output reg [5:0] payload_mem_addr;
    output reg [127:0] payload_mem_din;
    input [127:0] payload_mem_dout;

    reg [15:0] watchdog_counter;
    reg [15:0] seq_shadow;
    reg [127:0] resp_shadow;
    reg [5:0] mem_addr_q;
    reg [127:0] mem_din_q;
    reg mem_we_q;
    reg mem_csb_q;

    wire [3:0] req_opcode;
    wire [15:0] req_sequence;
    wire [7:0] req_age;
    wire [15:0] req_target;
    wire [15:0] req_value_raw;
    wire [3:0] req_mode_raw;
    wire [5:0] req_mem_addr;
    wire req_checksum_ok;
    wire req_length_ok;
    wire req_opcode_ok;
    wire seq_ok;
    wire stale_ok;
    wire watchdog_expired;
    wire clamp_needed;
    wire [15:0] clamp_low;
    wire [15:0] clamp_high;
    wire [15:0] cmd_clamped;

    assign req_opcode = req_data[127:124];
    assign req_sequence = req_data[123:108];
    assign req_age = req_data[107:100];
    assign req_target = req_data[99:84];
    assign req_value_raw = req_data[83:68];
    assign req_mode_raw = req_data[67:64];
    assign req_mem_addr = req_data[63:58];
    assign req_length_ok = (req_data[57:56] == 2'b01);
    assign req_checksum_ok = (req_data[55:48] == (req_data[127:120] ^ req_data[119:112] ^ req_data[111:104] ^ req_data[103:96]));
    assign req_opcode_ok = (req_opcode == 4'h0) || (req_opcode == 4'h1);
    assign seq_ok = (!cfg_enforce_sequence) || (req_sequence == (seq_shadow + 16'd1));
    assign stale_ok = (req_age <= cfg_stale_threshold);
    assign watchdog_expired = (watchdog_counter >= cfg_watchdog_limit) && (cfg_watchdog_limit != 16'd0);
    assign clamp_low = cfg_min_bound;
    assign clamp_high = cfg_max_bound;
    assign clamp_needed = (req_value_raw < clamp_low) || (req_value_raw > clamp_high);
    assign cmd_clamped = (req_value_raw < clamp_low) ? clamp_low : ((req_value_raw > clamp_high) ? clamp_high : req_value_raw);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            req_ready <= 1'b0;
            rsp_valid <= 1'b0;
            rsp_data <= 128'd0;
            status_mode <= 4'd0;
            status_last_sequence <= 16'd0;
            status_watchdog_count <= 16'd0;
            status_stale_fault <= 1'b0;
            status_timeout_fault <= 1'b0;
            status_clamp_fault <= 1'b0;
            status_protocol_fault <= 1'b0;
            status_safe_active <= 1'b1;
            cmd_valid <= 1'b0;
            cmd_value <= 16'd0;
            cmd_mode <= 4'd0;
            cmd_fault <= 1'b0;
            cmd_safe <= 1'b1;
            payload_mem_csb <= 1'b1;
            payload_mem_we <= 1'b0;
            payload_mem_addr <= 6'd0;
            payload_mem_din <= 128'd0;
            watchdog_counter <= 16'd0;
            seq_shadow <= 16'd0;
            resp_shadow <= 128'd0;
            mem_addr_q <= 6'd0;
            mem_din_q <= 128'd0;
            mem_we_q <= 1'b0;
            mem_csb_q <= 1'b1;
        end else begin
            req_ready <= cfg_enable && !status_timeout_fault && !rsp_valid;
            payload_mem_csb <= mem_csb_q;
            payload_mem_we <= mem_we_q;
            payload_mem_addr <= mem_addr_q;
            payload_mem_din <= mem_din_q;

            if (cfg_clear_sticky) begin
                status_stale_fault <= 1'b0;
                status_timeout_fault <= 1'b0;
                status_clamp_fault <= 1'b0;
                status_protocol_fault <= 1'b0;
            end

            if (watchdog_expired) begin
                status_timeout_fault <= 1'b1;
            end

            if (rsp_valid && rsp_ready) begin
                rsp_valid <= 1'b0;
            end

            if (cfg_enable && req_valid && req_ready) begin
                status_protocol_fault <= 1'b0;
                status_safe_active <= 1'b1;
                mem_csb_q <= 1'b0;
                mem_we_q <= (req_opcode == 4'h1);
                mem_addr_q <= req_mem_addr;
                mem_din_q <= req_data;
                seq_shadow <= req_sequence;
                status_last_sequence <= req_sequence;
                watchdog_counter <= 16'd1;
                status_watchdog_count <= 16'd1;
                status_mode <= cfg_default_mode;
                cmd_mode <= cfg_default_mode;
                if (!req_length_ok || !req_checksum_ok || !req_opcode_ok) begin
                    status_protocol_fault <= 1'b1;
                    cmd_valid <= 1'b0;
                    cmd_fault <= 1'b1;
                    cmd_safe <= 1'b1;
                    rsp_valid <= 1'b1;
                    rsp_data <= {112'd0, 16'h8001};
                end else if (!seq_ok) begin
                    status_protocol_fault <= 1'b1;
                    cmd_valid <= 1'b0;
                    cmd_fault <= 1'b1;
                    cmd_safe <= 1'b1;
                    rsp_valid <= 1'b1;
                    rsp_data <= {112'd0, 16'h8002};
                end else if (!stale_ok) begin
                    status_stale_fault <= 1'b1;
                    cmd_valid <= 1'b0;
                    cmd_fault <= 1'b1;
                    cmd_safe <= 1'b1;
                    rsp_valid <= 1'b1;
                    rsp_data <= {112'd0, 16'h8003};
                end else if (!cfg_enable || watchdog_expired) begin
                    status_timeout_fault <= 1'b1;
                    cmd_valid <= 1'b0;
                    cmd_fault <= 1'b1;
                    cmd_safe <= 1'b1;
                    rsp_valid <= 1'b1;
                    rsp_data <= {112'd0, 16'h8004};
                end else begin
                    cmd_value <= cmd_clamped;
                    cmd_valid <= 1'b1;
                    cmd_fault <= 1'b0;
                    cmd_safe <= 1'b0;
                    rsp_valid <= 1'b1;
                    rsp_data <= {96'd0, req_sequence, req_opcode, clamp_needed, 7'd0, cmd_clamped};
                    if (clamp_needed) begin
                        status_clamp_fault <= 1'b1;
                    end
                    if (req_opcode == 4'h0) begin
                        status_mode <= cfg_default_mode;
                        cmd_mode <= cfg_default_mode;
                    end else begin
                        status_mode <= cfg_default_mode ^ 4'h1;
                        cmd_mode <= cfg_default_mode ^ 4'h1;
                    end
                end
            end else begin
                mem_csb_q <= 1'b1;
                mem_we_q <= 1'b0;
                mem_addr_q <= 6'd0;
                mem_din_q <= 128'd0;
                if (watchdog_counter != 16'hffff) begin
                    watchdog_counter <= watchdog_counter + 16'd1;
                end
                status_watchdog_count <= watchdog_counter;
                if (watchdog_counter >= cfg_watchdog_limit && cfg_watchdog_limit != 16'd0) begin
                    status_timeout_fault <= 1'b1;
                    status_safe_active <= 1'b1;
                    cmd_valid <= 1'b0;
                    cmd_fault <= 1'b1;
                    cmd_safe <= 1'b1;
                end else begin
                    status_safe_active <= ~(cfg_enable && !status_timeout_fault);
                    if (!cfg_enable) begin
                        cmd_valid <= 1'b0;
                        cmd_fault <= 1'b0;
                        cmd_safe <= 1'b1;
                        status_mode <= cfg_default_mode;
                    end
                end
            end

            if (status_timeout_fault || status_stale_fault || status_protocol_fault) begin
                cmd_valid <= 1'b0;
                cmd_fault <= 1'b1;
                cmd_safe <= 1'b1;
                status_safe_active <= 1'b1;
            end
        end
    end
endmodule
---END adaptive_aero_control_transport.v---

---BEGIN fpga_bram_generic.v---
module fpga_bram_generic (
    clk,
    payload_mem_csb,
    payload_mem_we,
    payload_mem_addr,
    payload_mem_din,
    payload_mem_dout
);
    input clk;
    input payload_mem_csb;
    input payload_mem_we;
    input [5:0] payload_mem_addr;
    input [127:0] payload_mem_din;
    output reg [127:0] payload_mem_dout;

    reg [127:0] mem [0:63];
    reg [127:0] read_data_q;
    integer i;

    always @(posedge clk) begin
        if (!payload_mem_csb) begin
            if (payload_mem_we) begin
                mem[payload_mem_addr] <= payload_mem_din;
            end
            read_data_q <= mem[payload_mem_addr];
        end
        payload_mem_dout <= read_data_q;
    end
endmodule
