module adaptive_aero_control_csr (
    clk,
    rst_n,
    ctrl_reg_addr,
    ctrl_reg_wdata,
    ctrl_reg_write_en,
    ctrl_reg_read_en,
    ctrl_reg_rdata,
    ctrl_reg_ready,
    ctrl_reg_valid,
    cfg_enable,
    cfg_clear_fault,
    cfg_sequence_seed,
    cfg_request_timeout_threshold,
    cfg_response_age_threshold,
    cfg_actuator_min_clamp,
    cfg_actuator_max_clamp,
    cfg_actuator_rate_limit,
    cfg_fallback_command_value,
    cfg_operating_mode,
    cfg_config_valid,
    status_fsm_state,
    status_last_accepted_sequence,
    status_last_request_sequence,
    status_last_response_sequence,
    status_timeout_counter,
    status_stale_reject_counter,
    status_clamp_event_counter,
    status_fault_cause_code,
    status_fallback_active,
    status_fault_latched
);
    input clk;
    input rst_n;
    input [7:0] ctrl_reg_addr;
    input [63:0] ctrl_reg_wdata;
    input ctrl_reg_write_en;
    input ctrl_reg_read_en;
    output reg [63:0] ctrl_reg_rdata;
    output ctrl_reg_ready;
    output reg ctrl_reg_valid;

    output reg cfg_enable;
    output reg cfg_clear_fault;
    output reg [15:0] cfg_sequence_seed;
    output reg [15:0] cfg_request_timeout_threshold;
    output reg [15:0] cfg_response_age_threshold;
    output reg [15:0] cfg_actuator_min_clamp;
    output reg [15:0] cfg_actuator_max_clamp;
    output reg [15:0] cfg_actuator_rate_limit;
    output reg [31:0] cfg_fallback_command_value;
    output reg [3:0] cfg_operating_mode;
    output reg cfg_config_valid;

    input [2:0] status_fsm_state;
    input [15:0] status_last_accepted_sequence;
    input [15:0] status_last_request_sequence;
    input [15:0] status_last_response_sequence;
    input [15:0] status_timeout_counter;
    input [15:0] status_stale_reject_counter;
    input [15:0] status_clamp_event_counter;
    input [7:0] status_fault_cause_code;
    input status_fallback_active;
    input status_fault_latched;

    assign ctrl_reg_ready = 1'b1;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            cfg_enable <= 1'b0;
            cfg_clear_fault <= 1'b0;
            cfg_sequence_seed <= 16'h0000;
            cfg_request_timeout_threshold <= 16'h0010;
            cfg_response_age_threshold <= 16'h0010;
            cfg_actuator_min_clamp <= 16'h8000;
            cfg_actuator_max_clamp <= 16'h7FFF;
            cfg_actuator_rate_limit <= 16'h0100;
            cfg_fallback_command_value <= 32'h00000000;
            cfg_operating_mode <= 4'h0;
            cfg_config_valid <= 1'b0;
            ctrl_reg_valid <= 1'b0;
            ctrl_reg_rdata <= 64'h0000000000000000;
        end else begin
            cfg_clear_fault <= 1'b0;
            ctrl_reg_valid <= ctrl_reg_read_en | ctrl_reg_write_en;
            if (ctrl_reg_write_en) begin
                case (ctrl_reg_addr)
                    8'h00: begin
                        cfg_enable <= ctrl_reg_wdata[0];
                        cfg_clear_fault <= ctrl_reg_wdata[1];
                        cfg_config_valid <= ctrl_reg_wdata[2];
                        cfg_operating_mode <= ctrl_reg_wdata[7:4];
                    end
                    8'h08: begin
                        cfg_sequence_seed <= ctrl_reg_wdata[15:0];
                    end
                    8'h10: begin
                        cfg_request_timeout_threshold <= ctrl_reg_wdata[15:0];
                        cfg_response_age_threshold <= ctrl_reg_wdata[31:16];
                    end
                    8'h18: begin
                        cfg_actuator_min_clamp <= ctrl_reg_wdata[15:0];
                        cfg_actuator_max_clamp <= ctrl_reg_wdata[31:16];
                        cfg_actuator_rate_limit <= ctrl_reg_wdata[47:32];
                    end
                    8'h20: begin
                        cfg_fallback_command_value <= ctrl_reg_wdata[31:0];
                    end
                    default: begin
                    end
                endcase
            end
            if (ctrl_reg_read_en) begin
                case (ctrl_reg_addr)
                    8'h00: ctrl_reg_rdata <= {56'h00000000000000, cfg_operating_mode, cfg_config_valid, cfg_clear_fault, cfg_enable};
                    8'h08: ctrl_reg_rdata <= {48'h000000000000, cfg_sequence_seed};
                    8'h10: ctrl_reg_rdata <= {32'h00000000, cfg_response_age_threshold, cfg_request_timeout_threshold};
                    8'h18: ctrl_reg_rdata <= {16'h0000, cfg_actuator_rate_limit, cfg_actuator_max_clamp, cfg_actuator_min_clamp};
                    8'h20: ctrl_reg_rdata <= {32'h00000000, cfg_fallback_command_value};
                    8'h40: ctrl_reg_rdata <= {59'h000000000000000, status_fault_latched, status_fallback_active, status_fsm_state};
                    8'h48: ctrl_reg_rdata <= {16'h0000, status_last_response_sequence, status_last_request_sequence, status_last_accepted_sequence};
                    8'h50: ctrl_reg_rdata <= {16'h0000, status_clamp_event_counter, status_stale_reject_counter, status_timeout_counter};
                    8'h58: ctrl_reg_rdata <= {56'h00000000000000, status_fault_cause_code};
                    default: ctrl_reg_rdata <= 64'h0000000000000000;
                endcase
            end
        end
    end
endmodule

module adaptive_aero_control_stream (
    clk,
    rst_n,
    cfg_enable,
    cfg_clear_fault,
    cfg_sequence_seed,
    cfg_request_timeout_threshold,
    cfg_response_age_threshold,
    cfg_operating_mode,
    cfg_config_valid,
    status_fsm_state,
    status_last_accepted_sequence,
    status_last_request_sequence,
    status_last_response_sequence,
    status_timeout_counter,
    status_stale_reject_counter,
    status_clamp_event_counter,
    status_fault_cause_code,
    status_fallback_active,
    status_fault_latched,
    model_req_valid,
    model_req_ready,
    model_req_data,
    model_resp_valid,
    model_resp_ready,
    model_resp_data,
    accepted_command_word
);
    input clk;
    input rst_n;
    input cfg_enable;
    input cfg_clear_fault;
    input [15:0] cfg_sequence_seed;
    input [15:0] cfg_request_timeout_threshold;
    input [15:0] cfg_response_age_threshold;
    input [3:0] cfg_operating_mode;
    input cfg_config_valid;
    output reg [2:0] status_fsm_state;
    output reg [15:0] status_last_accepted_sequence;
    output reg [15:0] status_last_request_sequence;
    output reg [15:0] status_last_response_sequence;
    output reg [15:0] status_timeout_counter;
    output reg [15:0] status_stale_reject_counter;
    output reg [15:0] status_clamp_event_counter;
    output reg [7:0] status_fault_cause_code;
    output reg status_fallback_active;
    output reg status_fault_latched;
    output reg model_req_valid;
    input model_req_ready;
    output reg [127:0] model_req_data;
    input model_resp_valid;
    output reg model_resp_ready;
    input [127:0] model_resp_data;
    output reg [31:0] accepted_command_word;

    localparam [2:0] S_RESET = 3'd0;
    localparam [2:0] S_IDLE = 3'd1;
    localparam [2:0] S_ISSUE_REQ = 3'd2;
    localparam [2:0] S_WAIT_RESP = 3'd3;
    localparam [2:0] S_VALIDATE_RESP = 3'd4;
    localparam [2:0] S_APPLY_CMD = 3'd5;
    localparam [2:0] S_FALLBACK = 3'd6;
    localparam [2:0] S_FAULT_LATCH = 3'd7;

    reg [2:0] state, next_state;
    reg [15:0] req_seq;
    reg [15:0] resp_age_counter;
    reg [15:0] timeout_counter_r;
    reg [15:0] issued_seq_r;
    reg [15:0] last_request_sequence_r;
    reg [15:0] last_response_sequence_r;
    reg [31:0] last_command_word_r;
    reg [15:0] accepted_seq_r;
    reg [127:0] req_packet;
    reg [127:0] resp_packet;
    reg [31:0] resp_cmd_word;
    reg resp_valid_flag;
    reg [7:0] resp_checksum;
    reg [7:0] req_checksum;
    reg [31:0] req_payload_word;
    reg resp_ok;
    reg timeout_hit;
    reg stale_hit;
    reg malformed_hit;
    reg [31:0] accepted_command_word_next;
    reg [31:0] fallback_word;
    reg [7:0] fault_code_next;
    reg fallback_next;
    reg fault_latched_next;
    reg [15:0] timeout_counter_next;
    reg [15:0] stale_counter_next;
    reg [15:0] clamp_counter_next;
    reg [15:0] accepted_seq_next;
    reg [15:0] last_req_next;
    reg [15:0] last_resp_next;
    reg [15:0] req_seq_next;
    reg [15:0] resp_age_next;
    reg [31:0] mode_mixed_word;
    reg [31:0] safe_cmd_word;
    reg [31:0] clamped_cmd_word;
    reg [31:0] rate_limited_cmd_word;
    reg [31:0] delta_word;
    reg signed [32:0] signed_cmd;
    reg signed [32:0] signed_prev;
    reg signed [32:0] signed_min;
    reg signed [32:0] signed_max;
    reg signed [32:0] signed_rate;
    reg signed [32:0] signed_tmp;

    always @(*) begin
        next_state = state;
        model_req_valid = 1'b0;
        model_req_data = 128'h00000000000000000000000000000000;
        model_resp_ready = 1'b0;
        accepted_command_word_next = accepted_command_word;
        fallback_word = accepted_command_word;
        fault_code_next = status_fault_cause_code;
        fallback_next = status_fallback_active;
        fault_latched_next = status_fault_latched;
        timeout_counter_next = status_timeout_counter;
        stale_counter_next = status_stale_reject_counter;
        clamp_counter_next = status_clamp_event_counter;
        accepted_seq_next = status_last_accepted_sequence;
        last_req_next = status_last_request_sequence;
        last_resp_next = status_last_response_sequence;
        req_seq_next = req_seq;
        resp_age_next = resp_age_counter;
        req_packet = 128'h00000000000000000000000000000000;
        resp_packet = model_resp_data;
        resp_cmd_word = model_resp_data[95:64];
        resp_valid_flag = model_resp_data[96];
        resp_checksum = model_resp_data[103:96];
        req_checksum = 8'h00;
        req_payload_word = 32'h00000000;
        resp_ok = 1'b0;
        timeout_hit = 1'b0;
        stale_hit = 1'b0;
        malformed_hit = 1'b0;
        mode_mixed_word = 32'h00000000;
        safe_cmd_word = accepted_command_word;
        clamped_cmd_word = accepted_command_word;
        rate_limited_cmd_word = accepted_command_word;
        delta_word = 32'h00000000;
        signed_cmd = 33'sd0;
        signed_prev = 33'sd0;
        signed_min = 33'sd0;
        signed_max = 33'sd0;
        signed_rate = 33'sd0;
        signed_tmp = 33'sd0;
        model_resp_ready = (state == S_WAIT_RESP) & ~status_fault_latched;
        mode_mixed_word = {28'h0000000, cfg_operating_mode} ^ {28'h0000000, cfg_sequence_seed[3:0]};
        req_payload_word = {cfg_sequence_seed[7:0], cfg_sequence_seed[15:8], cfg_operating_mode, 4'hA, 16'h1357};
        req_checksum = req_payload_word[7:0] ^ req_payload_word[15:8] ^ req_payload_word[23:16] ^ req_payload_word[31:24] ^ cfg_operating_mode;
        req_packet = {req_checksum, 24'h00A55A, mode_mixed_word, req_payload_word, cfg_sequence_seed, 48'h000000000000};
        fallback_word = {16'h0000, 16'h0000} | cfg_fallback_command_value;
        if (!cfg_config_valid || !cfg_enable) begin
            fallback_next = 1'b1;
        end
        case (state)
            S_RESET: begin
                next_state = S_IDLE;
            end
            S_IDLE: begin
                if (!cfg_config_valid || !cfg_enable) begin
                    next_state = S_FALLBACK;
                end else if (status_fault_latched) begin
                    next_state = S_FAULT_LATCH;
                end else begin
                    next_state = S_ISSUE_REQ;
                end
            end
            S_ISSUE_REQ: begin
                model_req_valid = 1'b1;
                model_req_data = req_packet;
                if (model_req_ready) begin
                    next_state = S_WAIT_RESP;
                end else begin
                    next_state = S_ISSUE_REQ;
                end
            end
            S_WAIT_RESP: begin
                model_resp_ready = 1'b1;
                if (model_resp_valid) begin
                    next_state = S_VALIDATE_RESP;
                end else if (status_timeout_counter >= cfg_request_timeout_threshold) begin
                    timeout_hit = 1'b1;
                    next_state = S_FALLBACK;
                end else begin
                    next_state = S_WAIT_RESP;
                end
            end
            S_VALIDATE_RESP: begin
                model_resp_ready = 1'b1;
                stale_hit = (model_resp_data[63:48] != status_last_request_sequence);
                malformed_hit = (~model_resp_data[120]) | (model_resp_data[103:96] != (model_resp_data[95:64][7:0] ^ model_resp_data[95:64][15:8] ^ model_resp_data[95:64][23:16] ^ model_resp_data[95:64][31:24] ^ model_resp_data[111:104]));
                resp_ok = model_resp_data[121] & ~stale_hit & ~malformed_hit;
                if (resp_ok) begin
                    next_state = S_APPLY_CMD;
                end else begin
                    next_state = S_FALLBACK;
                end
            end
            S_APPLY_CMD: begin
                next_state = S_IDLE;
            end
            S_FALLBACK: begin
                next_state = S_FAULT_LATCH;
            end
            S_FAULT_LATCH: begin
                if (cfg_clear_fault && cfg_config_valid && cfg_enable) begin
                    next_state = S_IDLE;
                end else begin
                    next_state = S_FAULT_LATCH;
                end
            end
            default: begin
                next_state = S_FAULT_LATCH;
            end
        endcase
        if (state == S_VALIDATE_RESP && resp_ok) begin
            accepted_command_word_next = model_resp_data[95:64];
            accepted_seq_next = model_resp_data[63:48];
            last_resp_next = model_resp_data[63:48];
            fallback_next = 1'b0;
            fault_code_next = 8'h00;
        end
        if (state == S_WAIT_RESP && timeout_hit) begin
            timeout_counter_next = status_timeout_counter + 16'd1;
            fault_code_next = 8'h01;
            fallback_next = 1'b1;
            fault_latched_next = 1'b1;
        end
        if (state == S_VALIDATE_RESP && stale_hit) begin
            stale_counter_next = status_stale_reject_counter + 16'd1;
            fault_code_next = 8'h02;
            fallback_next = 1'b1;
            fault_latched_next = 1'b1;
        end
        if (state == S_VALIDATE_RESP && malformed_hit) begin
            fault_code_next = 8'h03;
            fallback_next = 1'b1;
            fault_latched_next = 1'b1;
        end
        if (!cfg_config_valid) begin
            fault_code_next = 8'h04;
        end
        if (cfg_clear_fault) begin
            fault_latched_next = 1'b0;
            if (cfg_config_valid && cfg_enable) fault_code_next = 8'h00;
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= S_RESET;
            req_seq <= 16'h0000;
            resp_age_counter <= 16'h0000;
            timeout_counter_r <= 16'h0000;
            issued_seq_r <= 16'h0000;
            last_request_sequence_r <= 16'h0000;
            last_response_sequence_r <= 16'h0000;
            last_command_word_r <= 32'h00000000;
            accepted_seq_r <= 16'h0000;
            status_fsm_state <= S_RESET;
            status_last_accepted_sequence <= 16'h0000;
            status_last_request_sequence <= 16'h0000;
            status_last_response_sequence <= 16'h0000;
            status_timeout_counter <= 16'h0000;
            status_stale_reject_counter <= 16'h0000;
            status_clamp_event_counter <= 16'h0000;
            status_fault_cause_code <= 8'h00;
            status_fallback_active <= 1'b1;
            status_fault_latched <= 1'b0;
            accepted_command_word <= 32'h00000000;
        end else begin
            state <= next_state;
            status_fsm_state <= next_state;
            status_timeout_counter <= timeout_counter_next;
            status_stale_reject_counter <= stale_counter_next;
            status_clamp_event_counter <= clamp_counter_next;
            status_fault_cause_code <= fault_code_next;
            status_fallback_active <= fallback_next;
            status_fault_latched <= fault_latched_next;
            accepted_command_word <= accepted_command_word_next;
            status_last_accepted_sequence <= accepted_seq_next;
            status_last_request_sequence <= last_req_next;
            status_last_response_sequence <= last_resp_next;
            req_seq <= req_seq_next;
            resp_age_counter <= resp_age_next;
            last_request_sequence_r <= last_req_next;
            last_response_sequence_r <= last_resp_next;
            accepted_seq_r <= accepted_seq_next;
            if (state == S_IDLE && cfg_enable && cfg_config_valid && !status_fault_latched) begin
                issued_seq_r <= cfg_sequence_seed;
                req_seq <= cfg_sequence_seed + 16'd1;
                status_last_request_sequence <= cfg_sequence_seed;
            end
            if (state == S_ISSUE_REQ && model_req_ready) begin
                status_last_request_sequence <= req_seq;
                last_request_sequence_r <= req_seq;
            end
            if (state == S_WAIT_RESP && !model_resp_valid) begin
                resp_age_counter <= resp_age_counter + 16'd1;
                timeout_counter_r <= timeout_counter_r + 16'd1;
            end else begin
                resp_age_counter <= 16'h0000;
            end
            if (state == S_APPLY_CMD) begin
                last_command_word_r <= accepted_command_word_next;
            end
            if (cfg_clear_fault && cfg_config_valid && cfg_enable) begin
                status_fault_latched <= 1'b0;
                status_fault_cause_code <= 8'h00;
            end
        end
    end
endmodule

module adaptive_aero_control_safety (
    clk,
    rst_n,
    cfg_enable,
    cfg_clear_fault,
    cfg_actuator_min_clamp,
    cfg_actuator_max_clamp,
    cfg_actuator_rate_limit,
    cfg_fallback_command_value,
    cfg_operating_mode,
    cfg_config_valid,
    status_fsm_state,
    status_timeout_counter,
    status_stale_reject_counter,
    status_clamp_event_counter,
    status_fault_cause_code,
    status_fallback_active,
    status_fault_latched,
    accepted_command_word,
    actuator_cmd_out,
    fault_status_out
);
    input clk;
    input rst_n;
    input cfg_enable;
    input cfg_clear_fault;
    input [15:0] cfg_actuator_min_clamp;
    input [15:0] cfg_actuator_max_clamp;
    input [15:0] cfg_actuator_rate_limit;
    input [31:0] cfg_fallback_command_value;
    input [3:0] cfg_operating_mode;
    input cfg_config_valid;
    input [2:0] status_fsm_state;
    input [15:0] status_timeout_counter;
    input [15:0] status_stale_reject_counter;
    input [15:0] status_clamp_event_counter;
    input [7:0] status_fault_cause_code;
    input status_fallback_active;
    input status_fault_latched;
    input [31:0] accepted_command_word;
    output reg [63:0] actuator_cmd_out;
    output reg [31:0] fault_status_out;

    reg [31:0] bounded_cmd;
    reg [31:0] next_cmd;
    reg [31:0] prev_cmd;
    reg [31:0] min_cmd;
    reg [31:0] max_cmd;
    reg [31:0] rate_cmd;
    reg [31:0] fallback_cmd;
    reg clamp_event;
    reg policy_violation;
    reg signed [32:0] s_acc;
    reg signed [32:0] s_prev;
    reg signed [32:0] s_min;
    reg signed [32:0] s_max;
    reg signed [32:0] s_rate;
    reg signed [32:0] s_diff;

    always @(*) begin
        min_cmd = {16'h0000, cfg_actuator_min_clamp};
        max_cmd = {16'h0000, cfg_actuator_max_clamp};
        rate_cmd = {16'h0000, cfg_actuator_rate_limit};
        fallback_cmd = cfg_fallback_command_value;
        next_cmd = accepted_command_word;
        bounded_cmd = accepted_command_word;
        clamp_event = 1'b0;
        policy_violation = 1'b0;
        s_acc = {1'b0, accepted_command_word};
        s_prev = {1'b0, actuator_cmd_out[31:0]};
        s_min = {1'b0, min_cmd};
        s_max = {1'b0, max_cmd};
        s_rate = {1'b0, rate_cmd};
        s_diff = s_acc - s_prev;
        if (!cfg_config_valid || !cfg_enable || status_fault_latched || status_fallback_active) begin
            next_cmd = fallback_cmd;
        end else begin
            if (s_acc < s_min) begin
                bounded_cmd = min_cmd;
                clamp_event = 1'b1;
            end else if (s_acc > s_max) begin
                bounded_cmd = max_cmd;
                clamp_event = 1'b1;
            end else begin
                bounded_cmd = accepted_command_word;
            end
            if (s_diff > s_rate) begin
                next_cmd = actuator_cmd_out[31:0] + rate_cmd;
                clamp_event = 1'b1;
            end else if (s_diff < -s_rate) begin
                next_cmd = actuator_cmd_out[31:0] - rate_cmd;
                clamp_event = 1'b1;
            end else begin
                next_cmd = bounded_cmd;
            end
            if (next_cmd < min_cmd || next_cmd > max_cmd) begin
                policy_violation = 1'b1;
                next_cmd = fallback_cmd;
            end
        end
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            actuator_cmd_out <= 64'h0000000000000000;
            fault_status_out <= 32'h00000000;
        end else begin
            if (!cfg_config_valid || !cfg_enable || status_fault_latched || status_fallback_active || policy_violation) begin
                actuator_cmd_out <= {32'h00000000, fallback_cmd};
            end else begin
                actuator_cmd_out <= {32'h00000000, next_cmd};
            end
            fault_status_out <= {24'h000000, cfg_config_valid ? 1'b0 : 1'b1, status_fault_latched, status_fallback_active, clamp_event, status_stale_reject_counter[0], status_timeout_counter[0], status_fault_cause_code[3:0]};
            if (cfg_clear_fault && cfg_config_valid && cfg_enable) begin
                fault_status_out <= {24'h000000, 1'b0, 1'b0, 1'b0, 1'b0, status_stale_reject_counter[0], status_timeout_counter[0], 4'h0};
            end
        end
    end
endmodule

module adaptive_aero_control_top (
    clk,
    rst_n,
    ctrl_reg_addr,
    ctrl_reg_wdata,
    ctrl_reg_write_en,
    ctrl_reg_read_en,
    ctrl_reg_rdata,
    ctrl_reg_ready,
    ctrl_reg_valid,
    model_req_valid,
    model_req_ready,
    model_req_data,
    model_resp_valid,
    model_resp_ready,
    model_resp_data,
    actuator_cmd_out,
    fault_status_out
);
    input clk;
    input rst_n;
    input [7:0] ctrl_reg_addr;
    input [63:0] ctrl_reg_wdata;
    input ctrl_reg_write_en;
    input ctrl_reg_read_en;
    output [63:0] ctrl_reg_rdata;
    output ctrl_reg_ready;
    output ctrl_reg_valid;
    output model_req_valid;
    input model_req_ready;
    output [127:0] model_req_data;
    input model_resp_valid;

module adaptive_aero_staging_ram (
    clk,
    csb,
    web,
    addr,
    din,
    dout
);
    input clk;
    input csb;
    input web;
    input [7:0] addr;
    input [31:0] din;
    output reg [63:0] dout;

    reg [31:0] mem0 [0:255];
    reg [31:0] mem1 [0:255];
    integer i;

    always @(posedge clk) begin
        if (!csb) begin
            if (!web) begin
                mem0[addr] <= din;
                mem1[addr] <= din ^ 32'hA5A5A5A5;
            end
            dout <= {mem1[addr], mem0[addr]};
        end
    end

    initial begin
        for (i = 0; i < 256; i = i + 1) begin
            mem0[i] = 32'h00000000;
            mem1[i] = 32'h00000000;
        end
        dout = 64'h0000000000000000;
    end
endmodule
