module register_file_64b (
    input clk,
    input reset_n,
    input [7:0] cfg_bus_addr,
    input [63:0] cfg_bus_wdata,
    input cfg_bus_wen,
    input cfg_bus_ren,
    output [63:0] cfg_bus_rdata,
    output cfg_bus_rvalid,
    output reg cfg_enable,
    output reg cfg_soft_reset,
    output reg [2:0] cfg_mode,
    output reg [15:0] cfg_timeout_threshold,
    output reg [15:0] cfg_freshness_threshold,
    output reg [31:0] cfg_act_min,
    output reg [31:0] cfg_act_max,
    output reg [7:0] cfg_seq_seed,
    output reg [31:0] cfg_safe_fallback,
    output reg cfg_status_clear,
    output reg cfg_fault_write,
    input [31:0] status_flags_in,
    input fault_in
);
reg [63:0] cfg_bus_rdata_r;
reg cfg_bus_rvalid_r;
assign cfg_bus_rdata = cfg_bus_rdata_r;
assign cfg_bus_rvalid = cfg_bus_rvalid_r;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        cfg_enable <= 1'b0;
        cfg_soft_reset <= 1'b0;
        cfg_mode <= 3'b000;
        cfg_timeout_threshold <= 16'd20;
        cfg_freshness_threshold <= 16'd10;
        cfg_act_min <= 32'd0;
        cfg_act_max <= 32'd255;
        cfg_seq_seed <= 8'd1;
        cfg_safe_fallback <= 32'd0;
        cfg_status_clear <= 1'b0;
        cfg_fault_write <= 1'b0;
        cfg_bus_rdata_r <= 64'd0;
        cfg_bus_rvalid_r <= 1'b0;
    end else begin
        cfg_soft_reset <= 1'b0;
        cfg_status_clear <= 1'b0;
        cfg_fault_write <= 1'b0;
        cfg_bus_rvalid_r <= cfg_bus_ren;
        if (cfg_bus_wen) begin
            case (cfg_bus_addr)
                8'h00: begin
                    cfg_enable <= cfg_bus_wdata[0];
                    cfg_soft_reset <= cfg_bus_wdata[1];
                    cfg_mode <= cfg_bus_wdata[4:2];
                    cfg_status_clear <= cfg_bus_wdata[7];
                end
                8'h01: cfg_timeout_threshold[7:0] <= cfg_bus_wdata[7:0];
                8'h02: cfg_timeout_threshold[15:8] <= cfg_bus_wdata[7:0];
                8'h03: cfg_freshness_threshold[7:0] <= cfg_bus_wdata[7:0];
                8'h04: cfg_freshness_threshold[15:8] <= cfg_bus_wdata[7:0];
                8'h05: cfg_act_min[7:0] <= cfg_bus_wdata[7:0];
                8'h06: cfg_act_min[15:8] <= cfg_bus_wdata[7:0];
                8'h07: cfg_act_min[23:16] <= cfg_bus_wdata[7:0];
                8'h08: cfg_act_min[31:24] <= cfg_bus_wdata[7:0];
                8'h09: cfg_act_max[7:0] <= cfg_bus_wdata[7:0];
                8'h0A: cfg_act_max[15:8] <= cfg_bus_wdata[7:0];
                8'h0B: cfg_act_max[23:16] <= cfg_bus_wdata[7:0];
                8'h0C: cfg_act_max[31:24] <= cfg_bus_wdata[7:0];
                8'h0D: cfg_seq_seed <= cfg_bus_wdata[7:0];
                8'h0E: cfg_safe_fallback[7:0] <= cfg_bus_wdata[7:0];
                8'h0F: cfg_safe_fallback[15:8] <= cfg_bus_wdata[7:0];
                8'h10: cfg_safe_fallback[23:16] <= cfg_bus_wdata[7:0];
                8'h11: cfg_safe_fallback[31:24] <= cfg_bus_wdata[7:0];
                default: begin
                    cfg_fault_write <= 1'b1;
                end
            endcase
        end
        case (cfg_bus_addr)
            8'h00: cfg_bus_rdata_r <= {54'd0, cfg_status_clear, cfg_fault_write, 1'b0, cfg_soft_reset, cfg_enable, 2'b00, cfg_mode};
            8'h01: cfg_bus_rdata_r <= {56'd0, cfg_timeout_threshold[7:0]};
            8'h02: cfg_bus_rdata_r <= {56'd0, cfg_timeout_threshold[15:8]};
            8'h03: cfg_bus_rdata_r <= {56'd0, cfg_freshness_threshold[7:0]};
            8'h04: cfg_bus_rdata_r <= {56'd0, cfg_freshness_threshold[15:8]};
            8'h05: cfg_bus_rdata_r <= {56'd0, cfg_act_min[7:0]};
            8'h06: cfg_bus_rdata_r <= {56'd0, cfg_act_min[15:8]};
            8'h07: cfg_bus_rdata_r <= {56'd0, cfg_act_min[23:16]};
            8'h08: cfg_bus_rdata_r <= {56'd0, cfg_act_min[31:24]};
            8'h09: cfg_bus_rdata_r <= {56'd0, cfg_act_max[7:0]};
            8'h0A: cfg_bus_rdata_r <= {56'd0, cfg_act_max[15:8]};
            8'h0B: cfg_bus_rdata_r <= {56'd0, cfg_act_max[23:16]};
            8'h0C: cfg_bus_rdata_r <= {56'd0, cfg_act_max[31:24]};
            8'h0D: cfg_bus_rdata_r <= {56'd0, cfg_seq_seed};
            8'h0E: cfg_bus_rdata_r <= {56'd0, cfg_safe_fallback[7:0]};
            8'h0F: cfg_bus_rdata_r <= {56'd0, cfg_safe_fallback[15:8]};
            8'h10: cfg_bus_rdata_r <= {56'd0, cfg_safe_fallback[23:16]};
            8'h11: cfg_bus_rdata_r <= {56'd0, cfg_safe_fallback[31:24]};
            8'h20: cfg_bus_rdata_r <= {55'd0, fault_in, cfg_fault_write, status_flags_in[6:0]};
            8'h21: cfg_bus_rdata_r <= {56'd0, cfg_seq_seed};
            8'h22: cfg_bus_rdata_r <= {56'd0, status_flags_in[7:0]};
            8'h23: cfg_bus_rdata_r <= {56'd0, status_flags_in[15:8]};
            8'h24: cfg_bus_rdata_r <= {56'd0, status_flags_in[23:16]};
            8'h25: cfg_bus_rdata_r <= {56'd0, status_flags_in[31:24]};
            8'h26: cfg_bus_rdata_r <= {56'd0, cfg_seq_seed};
            8'h27: cfg_bus_rdata_r <= 64'd0;
            8'h28: cfg_bus_rdata_r <= 64'd0;
            8'h29: cfg_bus_rdata_r <= 64'd0;
            8'h2A: cfg_bus_rdata_r <= 64'd0;
            8'h2B: cfg_bus_rdata_r <= 64'd0;
            8'h2C: cfg_bus_rdata_r <= 64'd0;
            8'h2D: cfg_bus_rdata_r <= {56'd0, cfg_seq_seed};
            8'h2E: cfg_bus_rdata_r <= 64'd0;
            8'h2F: cfg_bus_rdata_r <= 64'd0;
            default: cfg_bus_rdata_r <= 64'd0;
        endcase
    end
end
endmodule

module request_fsm (
    input clk,
    input reset_n,
    input cfg_enable,
    input cfg_soft_reset,
    input [2:0] cfg_mode,
    input [7:0] cfg_seq_seed,
    input [31:0] cfg_safe_fallback,
    input [7:0] outstanding_seq,
    input busy_in,
    input req_ready,
    output reg [63:0] req_data,
    output reg req_valid,
    output reg [7:0] req_seq_out,
    output reg [15:0] req_geom_handle,
    output reg [15:0] req_flow_id,
    output reg [7:0] req_ctrl_flags,
    output reg req_launch
);
reg [7:0] seq_reg;
reg [1:0] state;
reg [1:0] next_state;
localparam S_IDLE = 2'd0;
localparam S_WAIT = 2'd1;
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        seq_reg <= 8'd0;
        state <= S_IDLE;
        req_data <= 64'd0;
        req_valid <= 1'b0;
        req_seq_out <= 8'd0;
        req_geom_handle <= 16'd0;
        req_flow_id <= 16'd0;
        req_ctrl_flags <= 8'd0;
        req_launch <= 1'b0;
    end else begin
        state <= next_state;
        if (cfg_soft_reset) begin
            state <= S_IDLE;
            req_valid <= 1'b0;
            req_launch <= 1'b0;
        end else begin
            req_launch <= 1'b0;
            case (state)
                S_IDLE: begin
                    if (cfg_enable && !busy_in && req_ready) begin
                        seq_reg <= cfg_seq_seed;
                        req_seq_out <= cfg_seq_seed;
                        req_geom_handle <= {13'd0, cfg_mode};
                        req_flow_id <= {3'd0, cfg_mode, cfg_seq_seed};
                        req_ctrl_flags <= {cfg_mode, 5'b00001};
                        req_data <= {cfg_safe_fallback, cfg_seq_seed, cfg_mode, 21'd0};
                        req_valid <= 1'b1;
                        req_launch <= 1'b1;
                    end else begin
                        req_valid <= 1'b0;
                    end
                end
                S_WAIT: begin
                    req_valid <= 1'b0;
                end
                default: begin
                    req_valid <= 1'b0;
                end
            endcase
        end
    end
end
always @(*) begin
    next_state = state;
    if (cfg_soft_reset) begin
        next_state = S_IDLE;
    end else begin
        case (state)
            S_IDLE: begin
                if (cfg_enable && !busy_in && req_ready) begin
                    next_state = S_WAIT;
                end
            end
            S_WAIT: begin
                if (!busy_in) begin
                    next_state = S_IDLE;
                end
            end
            default: begin
                next_state = S_IDLE;
            end
        endcase
    end
end
endmodule

module response_validator (
    input clk,
    input reset_n,
    input [63:0] resp_data,
    input resp_valid,
    input [7:0] outstanding_seq,
    input [15:0] cfg_freshness_threshold,
    input [15:0] cfg_timeout_threshold,
    input [15:0] elapsed_age,
    input timeout_expired,
    output reg validated_valid,
    output reg validated_seq_match,
    output reg [31:0] validated_summary,
    output reg [7:0] validated_flags,
    output reg reject_stale,
    output reg reject_fault
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        validated_valid <= 1'b0;
        validated_seq_match <= 1'b0;
        validated_summary <= 32'd0;
        validated_flags <= 8'd0;
        reject_stale <= 1'b0;
        reject_fault <= 1'b0;
    end else begin
        validated_valid <= 1'b0;
        validated_seq_match <= 1'b0;
        if (resp_valid) begin
            validated_seq_match <= (resp_data[7:0] == outstanding_seq);
            validated_summary <= resp_data[63:32];
            validated_flags <= resp_data[15:8];
            if (timeout_expired || (elapsed_age > cfg_timeout_threshold)) begin
                reject_fault <= 1'b1;
                reject_stale <= 1'b0;
            end else if (elapsed_age > cfg_freshness_threshold) begin
                reject_stale <= 1'b1;
                reject_fault <= 1'b0;
            end else if (resp_data[7:0] != outstanding_seq) begin
                reject_stale <= 1'b1;
                reject_fault <= 1'b0;
            end else begin
                validated_valid <= 1'b1;
                reject_stale <= 1'b0;
                reject_fault <= 1'b0;
            end
        end
    end
end
endmodule

module timeout_counter (
    input clk,
    input reset_n,
    input [15:0] cfg_timeout_threshold,
    input [15:0] cfg_freshness_threshold,
    input req_launch,
    input resp_accept,
    output reg [15:0] elapsed_age,
    output reg timeout_expired,
    output reg stale_expired
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        elapsed_age <= 16'd0;
        timeout_expired <= 1'b0;
        stale_expired <= 1'b0;
    end else begin
        if (req_launch || resp_accept) begin
            elapsed_age <= 16'd0;
        end else begin
            elapsed_age <= elapsed_age + 16'd1;
        end
        timeout_expired <= (elapsed_age >= cfg_timeout_threshold);
        stale_expired <= (elapsed_age >= cfg_freshness_threshold);
    end
end
endmodule

module clamp_unit (
    input [31:0] validated_summary,
    input [7:0] validated_flags,
    input [31:0] cfg_act_min,
    input [31:0] cfg_act_max,
    input [31:0] cfg_safe_fallback,
    input fallback_active,
    output reg [31:0] cmd_raw,
    output reg [31:0] cmd_clamped,
    output reg clamp_violation,
    output reg [31:0] safe_cmd
);
always @(*) begin
    cmd_raw = validated_summary ^ {24'd0, validated_flags};
    safe_cmd = cfg_safe_fallback;
    clamp_violation = 1'b0;
    if (fallback_active) begin
        cmd_clamped = cfg_safe_fallback;
    end else begin
        cmd_clamped = cmd_raw;
        if (cmd_raw < cfg_act_min) begin
            cmd_clamped = cfg_act_min;
            clamp_violation = 1'b1;
        end else if (cmd_raw > cfg_act_max) begin
            cmd_clamped = cfg_act_max;
            clamp_violation = 1'b1;
        end
    end
end
endmodule

module fallback_controller (
    input clk,
    input reset_n,
    input timeout_expired,
    input reject_stale,
    input reject_fault,
    input clamp_violation,
    input cfg_fault_write,
    input downstream_ready,
    input [31:0] safe_cmd_in,
    output reg fallback_active,
    output reg fault_out,
    output reg cmd_select_safe,
    output reg cmd_force_valid
);
always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        fallback_active <= 1'b1;
        fault_out <= 1'b1;
        cmd_select_safe <= 1'b1;
        cmd_force_valid <= 1'b1;
    end else begin
        if (timeout_expired || reject_stale || reject_fault || clamp_violation || cfg_fault_write) begin
            fallback_active <= 1'b1;
        end
        cmd_select_safe <= fallback_active | timeout_expired | reject_stale | reject_fault | clamp_violation | cfg_fault_write;
        cmd_force_valid <= fallback_active | downstream_ready;
        fault_out <= fallback_active | reject_stale | reject_fault | clamp_violation | cfg_fault_write | timeout_expired;
    end
end
endmodule

    input [2:0] cfg_mode,

    input [15:0] cfg_freshness_threshold,

    status_flags_out[7] = cfg_soft_reset;
    status_flags_out[8] = cfg_enable;
    status_flags_out[11:9] = cfg_mode;
    status_flags_out[15:12] = outstanding_seq[3:0];
    status_flags_out[31:16] = elapsed_age;
    cfg_bus_rdata = {16'd0, cfg_act_max[15:0], cfg_act_min[15:0], cfg_timeout_threshold[7:0], cfg_freshness_threshold[7:0]};
end
endmodule
