assign cfg_queue_enable = cfg_queue_enable_r;
assign cfg_status_clear = cfg_status_clear_r;
assign sticky_fault_w1c = sticky_fault_w1c_r;
assign reg_rdata = reg_rdata_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        cfg_enable_r <= 1'b0;
        cfg_mode_r <= 3'b000;
        cfg_seq_seed_r <= 16'h0000;
        cfg_timeout_thresh_r <= 16'h0000;
        cfg_freshness_thresh_r <= 8'h00;
        cfg_actuator_min_r <= 16'h0000;
        cfg_actuator_max_r <= 16'hffff;
        cfg_fallback_cmd_r <= 16'h0000;
        cfg_safe_cmd_r <= 16'h0000;
        cfg_velocity_ref_r <= 16'd3889;
        cfg_geometry_id_r <= 16'h0000;
        cfg_queue_enable_r <= 1'b0;
        cfg_status_clear_r <= 1'b0;
        sticky_fault_w1c_r <= 16'h0000;
        reg_rdata_r <= 64'h0000000000000000;
        reg_rvalid <= 1'b0;
    end else begin
        cfg_status_clear_r <= 1'b0;
        reg_rvalid <= reg_rd_en;
        if (reg_wr_en) begin
            case (reg_addr)
                16'h0000: begin
                    cfg_enable_r <= reg_wdata[0];
                    cfg_mode_r <= reg_wdata[3:1];
                    cfg_queue_enable_r <= reg_wdata[4];
                    cfg_status_clear_r <= reg_wdata[5];
                end
                16'h0001: begin
                    cfg_seq_seed_r <= reg_wdata[15:0];
                end
                16'h0002: begin
                    cfg_timeout_thresh_r <= reg_wdata[15:0];
                    cfg_freshness_thresh_r <= reg_wdata[23:16];
                    cfg_velocity_ref_r <= reg_wdata[39:24];
                    cfg_geometry_id_r <= reg_wdata[55:40];
                end
                16'h0003: begin
                    cfg_actuator_min_r <= reg_wdata[15:0];
                    cfg_actuator_max_r <= reg_wdata[31:16];
                    cfg_fallback_cmd_r <= reg_wdata[47:32];
                    cfg_safe_cmd_r <= reg_wdata[63:48];
                end
                16'h0007: begin
                    sticky_fault_w1c_r <= reg_wdata[15:0];
                end
                default: begin
                end
            endcase
        end
        case (reg_addr)
            16'h0000: reg_rdata_r <= {58'h000000000000000, cfg_status_clear_r, cfg_queue_enable_r, cfg_mode_r, cfg_enable_r};
            16'h0001: reg_rdata_r <= {48'h000000000000, cfg_seq_seed_r};
            16'h0002: reg_rdata_r <= {8'h00, cfg_geometry_id_r, cfg_velocity_ref_r, cfg_freshness_thresh_r, cfg_timeout_thresh_r};
            16'h0003: reg_rdata_r <= {cfg_safe_cmd_r, cfg_fallback_cmd_r, cfg_actuator_max_r, cfg_actuator_min_r};
            16'h0004: reg_rdata_r <= {40'h0000000000, telemetry_queue_occupancy, telemetry_last_error_code, fault_summary[7:0], 8'h00, 3'b000, 1'b0, 1'b0, 1'b0, 1'b0, 1'b0, 1'b0};
            16'h0005: reg_rdata_r <= {40'h0000000000, telemetry_last_rsp_age, telemetry_last_rsp_sequence};
            16'h0006: reg_rdata_r <= {48'h000000000000, telemetry_seq_counter};
            16'h0007: reg_rdata_r <= {48'h000000000000, fault_summary};
            16'h0008: reg_rdata_r <= {32'h00000000, telemetry_requests_issued};
            16'h0009: reg_rdata_r <= {32'h00000000, telemetry_responses_accepted};
            16'h000A: reg_rdata_r <= {32'h00000000, telemetry_stale_rejects};
            16'h000B: reg_rdata_r <= {32'h00000000, telemetry_timeout_events};
            16'h000C: reg_rdata_r <= {32'h00000000, telemetry_clamp_events};
            16'h000D: reg_rdata_r <= {32'h00000000, telemetry_fallback_activations};
            16'h000E: reg_rdata_r <= {32'h00000000, telemetry_protocol_errors};
            16'h000F: reg_rdata_r <= 64'h0000000000000000;
            16'h0010: reg_rdata_r <= 64'h0000000000000000;
            16'h0011: reg_rdata_r <= 64'h0000000000000000;
            default: reg_rdata_r <= 64'h0000000000000000;
        endcase
    end
end

module adaptive_aero_request_fsm (
    clk,
    reset_n,
    cfg_enable,
    cfg_mode,
    cfg_seq_seed,
    cfg_queue_enable,
    cfg_velocity_ref,

    busy_reject
);
input clk;
input reset_n;
input cfg_enable;
input [2:0] cfg_mode;
input [15:0] cfg_seq_seed;
input cfg_queue_enable;
input [15:0] cfg_velocity_ref;

output [15:0] request_launch_cycle;
output request_outstanding;
output [7:0] request_queue_occupancy;
output request_accepted_pulse;
output [15:0] seq_counter;
output busy_reject;

reg model_req_valid_r;
reg [63:0] model_req_data_r;
reg [15:0] request_sequence_r;
reg [15:0] request_launch_cycle_r;
reg request_outstanding_r;
reg [7:0] request_queue_occupancy_r;
reg request_accepted_pulse_r;
reg [15:0] seq_counter_r;
reg busy_reject_r;
reg [15:0] launch_cycle_cnt;

assign model_req_valid = model_req_valid_r;
assign model_req_data = model_req_data_r;
assign request_sequence = request_sequence_r;
assign request_launch_cycle = request_launch_cycle_r;
assign request_outstanding = request_outstanding_r;
assign request_queue_occupancy = request_queue_occupancy_r;
assign request_accepted_pulse = request_accepted_pulse_r;
assign seq_counter = seq_counter_r;
assign busy_reject = busy_reject_r;

always @(posedge clk or negedge reset_n) begin
    if (!reset_n) begin
        model_req_valid_r <= 1'b0;
        model_req_data_r <= 64'h0000000000000000;
        request_sequence_r <= cfg_seq_seed;
        request_launch_cycle_r <= 16'h0000;
        request_outstanding_r <= 1'b0;
        request_queue_occupancy_r <= 8'h00;
        request_accepted_pulse_r <= 1'b0;
        seq_counter_r <= cfg_seq_seed;
        busy_reject_r <= 1'b0;
        launch_cycle_cnt <= 16'h0000;
    end else begin
        request_accepted_pulse_r <= 1'b0;
        busy_reject_r <= 1'b0;
        if (!cfg_enable) begin
            model_req_valid_r <= 1'b0;
            request_outstanding_r <= 1'b0;
            request_queue_occupancy_r <= 8'h00;
        end else begin
            if (start_request) begin
                if (!request_outstanding_r || cfg_queue_enable) begin
                    model_req_valid_r <= 1'b1;
                    model_req_data_r <= {cfg_mode, cfg_geometry_id, cfg_velocity_ref, seq_counter_r, 28'h0000000};
                    request_sequence_r <= seq_counter_r;
                    request_launch_cycle_r <= launch_cycle_cnt;
                    request_outstanding_r <= 1'b1;
                    request_queue_occupancy_r <= cfg_queue_enable ? (request_queue_occupancy_r + 8'h01) : 8'h01;
                end else begin
                    busy_reject_r <= 1'b1;
                end
            end
            if (model_req_valid_r && model_req_ready) begin
                model_req_valid_r <= 1'b0;
                request_accepted_pulse_r <= 1'b1;
                request_outstanding_r <= 1'b1;
                seq_counter_r <= seq_counter_r + 16'h0001;
                if (request_queue_occupancy_r != 8'h00) begin
                    request_queue_occupancy_r <= request_queue_occupancy_r - 8'h01;
                end
            end
        end
        launch_cycle_cnt <= launch_cycle_cnt + 16'h0001;
    end
end

endmodule

        timeout_fault_r <= request_outstanding & (cfg_timeout_thresh != 16'h0000) & (cycle_count - request_launch_cycle > cfg_timeout_thresh);
        safe_mode_r <= (!cfg_enable) | timeout_fault_r | protocol_fault_r | fallback_active_r;
        fallback_active_r <= (!cfg_enable) | timeout_fault_r | protocol_fault_r | (!request_outstanding);
        if (model_rsp_valid && model_rsp_ready_r) begin
            resp_seq <= model_rsp_data[15:0];
            resp_cmd_raw <= model_rsp_data[31:16];
            age_calc <= cycle_count - request_launch_cycle;
            age_calc8 <= (cycle_count - request_launch_cycle)[7:0];
            last_rsp_sequence_r <= model_rsp_data[15:0];
            last_rsp_age_r <= age_calc8;
            if ((model_rsp_data[15:0] == request_sequence) && request_outstanding && cfg_enable) begin
                if ((cycle_count - request_launch_cycle) <= {8'h00, cfg_freshness_thresh}) begin
                    resp_accept_pulse_r <= 1'b1;
                    actuator_cmd_valid_r <= 1'b1;
                    if (model_rsp_data[31:16] < cfg_actuator_min) begin
                        actuator_cmd_r <= cfg_actuator_min;
                        clamp_event_r <= 1'b1;
                    end else if (model_rsp_data[31:16] > cfg_actuator_max) begin
                        actuator_cmd_r <= cfg_actuator_max;
                        clamp_event_r <= 1'b1;
                    end else begin
                        actuator_cmd_r <= model_rsp_data[31:16];
                    end
                    safe_mode_r <= 1'b0;
                    fallback_active_r <= 1'b0;
                    last_error_code_r <= 8'h00;
                end else begin
                    resp_stale_pulse_r <= 1'b1;
                    resp_reject_pulse_r <= 1'b1;
                    protocol_fault_r <= 1'b1;
                    sticky_fault_bits_r[0] <= 1'b1;
                    last_error_code_r <= 8'h01;
                    actuator_cmd_r <= cfg_fallback_cmd;
                    actuator_cmd_valid_r <= 1'b1;
                    safe_mode_r <= 1'b1;
                    fallback_active_r <= 1'b1;
                end
            end else begin
                resp_reject_pulse_r <= 1'b1;
                protocol_fault_r <= 1'b1;
                sticky_fault_bits_r[1] <= 1'b1;
                last_error_code_r <= 8'h02;
                actuator_cmd_r <= cfg_safe_cmd;
                actuator_cmd_valid_r <= 1'b1;
                safe_mode_r <= 1'b1;
                fallback_active_r <= 1'b1;
            end
        end else if (!cfg_enable) begin
            actuator_cmd_r <= cfg_safe_cmd;
            actuator_cmd_valid_r <= 1'b1;
            safe_mode_r <= 1'b1;
            fallback_active_r <= 1'b1;
        end else if (timeout_fault_r) begin
            resp_reject_pulse_r <= 1'b1;
            protocol_fault_r <= 1'b1;
            sticky_fault_bits_r[2] <= 1'b1;
            last_error_code_r <= 8'h03;
            actuator_cmd_r <= cfg_fallback_cmd;
            actuator_cmd_valid_r <= 1'b1;
            safe_mode_r <= 1'b1;
            fallback_active_r <= 1'b1;
