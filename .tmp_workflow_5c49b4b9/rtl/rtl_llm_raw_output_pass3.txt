module adaptive_aero_control_mmio (
    input clk,
    input rst_n,
    input csr_we,
    input csr_re,
    input [7:0] csr_addr,
    input [31:0] csr_wdata,
    output reg cfg_enable,
    output reg cfg_clear_faults,
    output reg [15:0] cfg_timeout_threshold,
    output reg [31:0] cfg_actuator_min,
    output reg [31:0] cfg_actuator_max,
    output reg [31:0] cfg_fallback_command,
    output reg cfg_fallback_valid_policy,
    output reg [15:0] cfg_velocity_scale,
    output reg [7:0] cfg_geometry_meta_default,
    output reg [7:0] last_error_code,
    output reg [15:0] last_request_id,
    output reg [15:0] last_response_id,
    output reg [7:0] status_summary_bits,
    output reg [31:0] csr_rdata
);

reg [7:0] last_error_code_next;
reg [15:0] last_request_id_next;
reg [15:0] last_response_id_next;
reg [7:0] status_summary_bits_next;
reg [31:0] csr_rdata_next;
reg cfg_clear_faults_next;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        cfg_enable <= 1'b0;
        cfg_clear_faults <= 1'b0;
        cfg_timeout_threshold <= 16'd16;
        cfg_actuator_min <= 32'd0;
        cfg_actuator_max <= 32'd255;
        cfg_fallback_command <= 32'd0;
        cfg_fallback_valid_policy <= 1'b0;
        cfg_velocity_scale <= 16'd1;
        cfg_geometry_meta_default <= 8'd0;
        last_error_code <= 8'd0;
        last_request_id <= 16'd0;
        last_response_id <= 16'd0;
        status_summary_bits <= 8'h20;
        csr_rdata <= 32'd0;
    end else begin
        cfg_enable <= cfg_enable;
        cfg_clear_faults <= cfg_clear_faults_next;
        if (csr_we) begin
            case (csr_addr)
                8'h00: cfg_enable <= csr_wdata[0];
                8'h04: cfg_clear_faults <= csr_wdata[0];
                8'h08: cfg_timeout_threshold <= csr_wdata[15:0];
                8'h09: cfg_timeout_threshold <= {csr_wdata[7:0], cfg_timeout_threshold[7:0]};
                8'h0C: cfg_actuator_min[7:0] <= csr_wdata[7:0];
                8'h0D: cfg_actuator_min[15:8] <= csr_wdata[7:0];
                8'h0E: cfg_actuator_min[23:16] <= csr_wdata[7:0];
                8'h0F: cfg_actuator_min[31:24] <= csr_wdata[7:0];
                8'h10: cfg_actuator_max[7:0] <= csr_wdata[7:0];
                8'h11: cfg_actuator_max[15:8] <= csr_wdata[7:0];
                8'h12: cfg_actuator_max[23:16] <= csr_wdata[7:0];
                8'h13: cfg_actuator_max[31:24] <= csr_wdata[7:0];
                8'h14: cfg_fallback_command[7:0] <= csr_wdata[7:0];
                8'h15: cfg_fallback_command[15:8] <= csr_wdata[7:0];
                8'h16: cfg_fallback_command[23:16] <= csr_wdata[7:0];
                8'h17: cfg_fallback_command[31:24] <= csr_wdata[7:0];
                8'h18: cfg_fallback_valid_policy <= csr_wdata[0];
                8'h1C: cfg_velocity_scale[7:0] <= csr_wdata[7:0];
                8'h1D: cfg_velocity_scale[15:8] <= csr_wdata[7:0];
                8'h20: cfg_geometry_meta_default <= csr_wdata[7:0];
                default: begin
                end
            endcase
        end
        if (cfg_clear_faults_next) begin
            last_error_code <= 8'd0;
            status_summary_bits[0] <= 1'b0;
            status_summary_bits[1] <= 1'b0;
            status_summary_bits[3] <= 1'b0;
        end
        status_summary_bits[5] <= cfg_enable;
        csr_rdata <= csr_rdata_next;
        last_error_code <= last_error_code_next;
        last_request_id <= last_request_id_next;
        last_response_id <= last_response_id_next;
        status_summary_bits <= status_summary_bits_next;
    end
end

always @(*) begin
    csr_rdata_next = 32'd0;
    last_error_code_next = last_error_code;
    last_request_id_next = last_request_id;
    last_response_id_next = last_response_id;
    status_summary_bits_next = status_summary_bits;
    cfg_clear_faults_next = 1'b0;
    case (csr_addr)
        8'h00: csr_rdata_next = {31'd0, cfg_enable};
        8'h04: csr_rdata_next = {31'd0, cfg_clear_faults};
        8'h08: csr_rdata_next = {16'd0, cfg_timeout_threshold};
        8'h09: csr_rdata_next = {24'd0, cfg_timeout_threshold[15:8]};
        8'h0C: csr_rdata_next = {24'd0, cfg_actuator_min[7:0]};
        8'h0D: csr_rdata_next = {24'd0, cfg_actuator_min[15:8]};
        8'h0E: csr_rdata_next = {24'd0, cfg_actuator_min[23:16]};
        8'h0F: csr_rdata_next = {24'd0, cfg_actuator_min[31:24]};
        8'h10: csr_rdata_next = {24'd0, cfg_actuator_max[7:0]};
        8'h11: csr_rdata_next = {24'd0, cfg_actuator_max[15:8]};
        8'h12: csr_rdata_next = {24'd0, cfg_actuator_max[23:16]};
        8'h13: csr_rdata_next = {24'd0, cfg_actuator_max[31:24]};
        8'h14: csr_rdata_next = {24'd0, cfg_fallback_command[7:0]};
        8'h15: csr_rdata_next = {24'd0, cfg_fallback_command[15:8]};
        8'h16: csr_rdata_next = {24'd0, cfg_fallback_command[23:16]};
        8'h17: csr_rdata_next = {24'd0, cfg_fallback_command[31:24]};
        8'h18: csr_rdata_next = {31'd0, cfg_fallback_valid_policy};
        8'h1C: csr_rdata_next = {16'd0, cfg_velocity_scale};
        8'h1D: csr_rdata_next = {24'd0, cfg_velocity_scale[15:8]};
        8'h20: csr_rdata_next = {24'd0, cfg_geometry_meta_default};
        8'h24: csr_rdata_next = {24'd0, last_error_code};
        8'h28: csr_rdata_next = {16'd0, last_request_id};
        8'h29: csr_rdata_next = {24'd0, last_request_id[15:8]};
        8'h2C: csr_rdata_next = {16'd0, last_response_id};
        8'h2D: csr_rdata_next = {24'd0, last_response_id[15:8]};
        8'h30: csr_rdata_next = {24'd0, status_summary_bits};
        default: csr_rdata_next = 32'd0;
    endcase
    status_summary_bits_next[0] = status_summary_bits[0];
    status_summary_bits_next[1] = status_summary_bits[1];
    status_summary_bits_next[2] = status_summary_bits[2];
    status_summary_bits_next[3] = status_summary_bits[3];
    status_summary_bits_next[4] = status_summary_bits[4];
    status_summary_bits_next[5] = cfg_enable;
    status_summary_bits_next[6] = status_summary_bits[6];
    status_summary_bits_next[7] = 1'b0;
end

endmodule

---BEGIN adaptive_aero_control_request_mgr.v---
module adaptive_aero_control_request_mgr (
    input clk,
    input rst_n,
    input cfg_enable,
    input [15:0] cfg_velocity_scale,
    input [7:0] cfg_geometry_meta_default,
    input [15:0] cfg_timeout_threshold,
    input [15:0] local_velocity_in,
    input [3:0] local_mode_in,
    input [7:0] geometry_source_sel_in,
    input req_ready,
    output reg req_valid,
    output reg [127:0] req_data,
    output reg outstanding_req_active,
    output reg request_accepted_pulse,
    output reg [15:0] request_seq_id,
    output reg [15:0] last_request_id,
    output reg request_stale_flag
);

reg [127:0] req_data_hold;
reg [15:0] velocity_scaled;
reg [7:0] geometry_meta;
reg local_policy_ok;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        req_valid <= 1'b0;
        req_data <= 128'd0;
        req_data_hold <= 128'd0;
        outstanding_req_active <= 1'b0;
        request_accepted_pulse <= 1'b0;
        request_seq_id <= 16'd0;
        last_request_id <= 16'd0;
        request_stale_flag <= 1'b0;
    end else begin
        request_accepted_pulse <= 1'b0;
        if (!cfg_enable) begin
            req_valid <= 1'b0;
            outstanding_req_active <= 1'b0;
        end else begin
            if (!outstanding_req_active && !request_stale_flag) begin
                req_valid <= 1'b1;
                req_data <= req_data_hold;
            end else if (req_valid && req_ready) begin
                req_valid <= 1'b0;
            end
            if (req_valid && req_ready) begin
                outstanding_req_active <= 1'b1;
                request_accepted_pulse <= 1'b1;
                request_seq_id <= request_seq_id + 16'd1;
                last_request_id <= request_seq_id + 16'd1;
                request_stale_flag <= 1'b0;
            end
        end
        if (request_stale_flag) begin
            outstanding_req_active <= 1'b0;
            req_valid <= 1'b0;
        end
    end
end

always @(*) begin
    local_policy_ok = cfg_enable & (cfg_timeout_threshold != 16'd0);
    velocity_scaled = local_velocity_in * cfg_velocity_scale;
    geometry_meta = geometry_source_sel_in ^ cfg_geometry_meta_default;
    req_data_hold = 128'd0;
    req_data_hold[15:0] = request_seq_id;
    req_data_hold[19:16] = local_mode_in;
    req_data_hold[35:20] = velocity_scaled;
    req_data_hold[43:36] = geometry_meta;
    req_data_hold[59:44] = local_velocity_in;
    req_data_hold[67:60] = cfg_geometry_meta_default;
    req_data_hold[127:68] = 60'd0;
    if (!local_policy_ok || outstanding_req_active) begin
        req_data_hold[127] = 1'b0;
    end else begin
        req_data_hold[127] = 1'b1;
    end
    if (!cfg_enable || outstanding_req_active || request_stale_flag) begin
        req_valid = 1'b0;
    end else begin
        req_valid = 1'b1;
    end
end

endmodule

module adaptive_aero_control_response_mgr (
    input clk,
    input rst_n,
    input rsp_valid,
    input [127:0] rsp_data,
    output reg rsp_ready,
    input outstanding_req_active,
    input [15:0] request_seq_id,
    input timeout_expired,
    input cfg_enable,
    output reg [15:0] parsed_response_seq_id,
    output reg [31:0] parsed_command_payload,
    output reg parsed_response_valid,
    output reg parsed_health_indicator,
    output reg [15:0] parsed_freshness_meta,
    output reg response_accept_pulse,
    output reg response_reject_pulse,
    output reg valid_command_available,
    output reg [31:0] validated_command_payload,
    output reg [15:0] last_response_id,
    output reg stale_response_flag,
    output reg health_fault_flag
);

reg [15:0] rsp_seq_shadow;
reg [31:0] rsp_cmd_shadow;
reg [15:0] rsp_fresh_shadow;
reg rsp_valid_shadow;
reg rsp_health_shadow;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        rsp_ready <= 1'b0;
        parsed_response_seq_id <= 16'd0;
        parsed_command_payload <= 32'd0;
        parsed_response_valid <= 1'b0;
        parsed_health_indicator <= 1'b0;
        parsed_freshness_meta <= 16'd0;
        response_accept_pulse <= 1'b0;
        response_reject_pulse <= 1'b0;
        valid_command_available <= 1'b0;
        validated_command_payload <= 32'd0;
        last_response_id <= 16'd0;
        stale_response_flag <= 1'b0;
        health_fault_flag <= 1'b0;
        rsp_seq_shadow <= 16'd0;
        rsp_cmd_shadow <= 32'd0;
        rsp_fresh_shadow <= 16'd0;
        rsp_valid_shadow <= 1'b0;
        rsp_health_shadow <= 1'b0;
    end else begin
        response_accept_pulse <= 1'b0;
        response_reject_pulse <= 1'b0;
        if (cfg_enable && outstanding_req_active && !timeout_expired) begin
            rsp_ready <= 1'b1;
        end else begin
            rsp_ready <= 1'b0;
        end
        if (rsp_valid && rsp_ready) begin
            rsp_seq_shadow <= rsp_data[15:0];
            rsp_cmd_shadow <= rsp_data[47:16];
            rsp_valid_shadow <= rsp_data[48];
            rsp_health_shadow <= rsp_data[49];
            rsp_fresh_shadow <= rsp_data[65:50];
            parsed_response_seq_id <= rsp_data[15:0];
            parsed_command_payload <= rsp_data[47:16];
            parsed_response_valid <= rsp_data[48];
            parsed_health_indicator <= rsp_data[49];
            parsed_freshness_meta <= rsp_data[65:50];
            if (rsp_data[48] && rsp_data[49] && !timeout_expired && outstanding_req_active && (rsp_data[15:0] == request_seq_id)) begin
                response_accept_pulse <= 1'b1;
                valid_command_available <= 1'b1;
                validated_command_payload <= rsp_data[47:16];
                last_response_id <= rsp_data[15:0];
                stale_response_flag <= 1'b0;
                health_fault_flag <= 1'b0;
            end else begin
                response_reject_pulse <= 1'b1;
                valid_command_available <= 1'b0;
                stale_response_flag <= timeout_expired | (outstanding_req_active == 1'b0);
                health_fault_flag <= ~rsp_data[49];
            end
        end else begin
            valid_command_available <= 1'b0;
        end
        if (timeout_expired) begin
            stale_response_flag <= 1'b1;
            valid_command_available <= 1'b0;
        end
    end
end

endmodule

---BEGIN adaptive_aero_control_safety_mgr.v---
module adaptive_aero_control_safety_mgr (
    input clk,
    input rst_n,
    input cfg_enable,
    input [31:0] cfg_actuator_min,
    input [31:0] cfg_actuator_max,
    input [31:0] cfg_fallback_command,
    input cfg_fallback_valid_policy,
    input [15:0] cfg_timeout_threshold,
    input request_accepted_pulse,
    input response_accept_pulse,
    input valid_command_available,
    input [31:0] validated_command_payload,
    input stale_response_flag,
    input health_fault_flag,
    output reg timeout_expired,
    output reg safe_fallback_active,
    output reg actuator_cmd_valid,
    output reg [31:0] actuator_cmd_data,
    output reg status_timeout_flag,
    output reg status_stale_flag,
    output reg status_fallback_flag,
    output reg status_fault_flag
);

reg [15:0] timeout_counter;
reg timeout_running;
reg [31:0] clamped_command;
reg [31:0] clamp_min;
reg [31:0] clamp_max;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        timeout_expired <= 1'b0;
        safe_fallback_active <= 1'b1;
        actuator_cmd_valid <= 1'b0;
        actuator_cmd_data <= 32'd0;
        status_timeout_flag <= 1'b0;
        status_stale_flag <= 1'b0;
        status_fallback_flag <= 1'b1;
        status_fault_flag <= 1'b0;
        timeout_counter <= 16'd0;
        timeout_running <= 1'b0;
    end else begin
        if (!cfg_enable) begin
            safe_fallback_active <= 1'b1;
            actuator_cmd_valid <= cfg_fallback_valid_policy;
            actuator_cmd_data <= cfg_fallback_command;
            status_fallback_flag <= 1'b1;
        end else if (valid_command_available) begin
            safe_fallback_active <= 1'b0;
            actuator_cmd_valid <= 1'b1;
            actuator_cmd_data <= clamped_command;
            status_fallback_flag <= 1'b0;
        end else begin
            safe_fallback_active <= 1'b1;
            actuator_cmd_valid <= cfg_fallback_valid_policy;
            actuator_cmd_data <= cfg_fallback_command;
            status_fallback_flag <= 1'b1;
        end

        if (request_accepted_pulse) begin
            timeout_running <= 1'b1;
            timeout_counter <= 16'd0;
            timeout_expired <= 1'b0;
        end else if (timeout_running && !response_accept_pulse) begin
            timeout_counter <= timeout_counter + 16'd1;
            if (timeout_counter >= cfg_timeout_threshold) begin
                timeout_expired <= 1'b1;
                timeout_running <= 1'b0;
            end
        end
        if (response_accept_pulse) begin
            timeout_running <= 1'b0;
            timeout_counter <= 16'd0;
            timeout_expired <= 1'b0;
        end
        status_timeout_flag <= timeout_expired;
        status_stale_flag <= stale_response_flag | timeout_expired;
        status_fault_flag <= health_fault_flag | timeout_expired;
    end
end

always @(*) begin
    clamp_min = cfg_actuator_min;
    clamp_max = cfg_actuator_max;
    if (validated_command_payload < clamp_min) begin
        clamped_command = clamp_min;
    end else if (validated_command_payload > clamp_max) begin
        clamped_command = clamp_max;
    end else begin
        clamped_command = validated_command_payload;
    end
end
