---BEGIN request_formatter.v---
module request_formatter (
    input clk,
    input rst_n,
    input format_enable,
    input [15:0] sequence_tag_in,
    input [7:0] payload_len_in,
    input [31:0] vehicle_meta_in,
    input [31:0] flow_meta_in,
    input [31:0] cmd_hint_in,
    input req_ready_in,
    output reg req_valid_out,
    output reg [127:0] req_data_out,
    output reg req_last_out,
    output reg request_accepted
);

reg [127:0] packet_reg;
reg packet_pending;

wire [31:0] checksum_w;
wire [31:0] meta_mix_w;
wire [127:0] packet_w;

assign meta_mix_w = vehicle_meta_in ^ flow_meta_in ^ cmd_hint_in ^ {16'h0000, sequence_tag_in};
assign checksum_w = meta_mix_w ^ {24'h000000, payload_len_in};
assign packet_w = {
    8'hA5,
    sequence_tag_in,
    payload_len_in,
    8'h01,
    vehicle_meta_in,
    flow_meta_in,
    cmd_hint_in,
    checksum_w
};

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        packet_reg <= 128'h00000000000000000000000000000000;
        packet_pending <= 1'b0;
        req_valid_out <= 1'b0;
        req_data_out <= 128'h00000000000000000000000000000000;
        req_last_out <= 1'b0;
        request_accepted <= 1'b0;
    end else begin
        request_accepted <= 1'b0;
        if (format_enable && !packet_pending) begin
            packet_reg <= packet_w;
            packet_pending <= 1'b1;
        end
        if (packet_pending && req_ready_in) begin
            req_valid_out <= 1'b1;
            req_data_out <= packet_reg;
            req_last_out <= 1'b1;
            request_accepted <= 1'b1;
            packet_pending <= 1'b0;
        end else begin
            req_valid_out <= packet_pending;
            req_last_out <= packet_pending;
            if (packet_pending) begin
                req_data_out <= packet_reg;
            end
        end
    end
end

endmodule

module response_validator (
    input clk,
    input rst_n,
    input rsp_valid_in,
    input [127:0] rsp_data_in,
    input rsp_last_in,
    input [15:0] expected_sequence_tag_in,
    input timeout_expired_in,
    input busy_in,
    output reg response_valid_out,
    output reg response_stale_out,
    output reg malformed_response_out,
    output reg sequence_error_out,
    output reg [31:0] validated_command_out,
    output reg [15:0] response_sequence_tag_out
);

wire [7:0] version_w;
wire [15:0] tag_w;
wire [7:0] len_w;
wire [31:0] command_w;
wire [31:0] parity_w;
wire [31:0] calc_parity_w;
wire frame_ok_w;
wire tag_ok_w;
wire len_ok_w;

assign version_w = rsp_data_in[127:120];
assign tag_w = rsp_data_in[119:104];
assign len_w = rsp_data_in[103:96];
assign command_w = rsp_data_in[63:32];
assign parity_w = rsp_data_in[31:0];
assign calc_parity_w = rsp_data_in[127:96] ^ rsp_data_in[95:64];
assign frame_ok_w = rsp_last_in && (version_w == 8'hA5);
assign tag_ok_w = (tag_w == expected_sequence_tag_in);
assign len_ok_w = (len_w != 8'h00);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        response_valid_out <= 1'b0;
        response_stale_out <= 1'b0;
        malformed_response_out <= 1'b0;
        sequence_error_out <= 1'b0;
        validated_command_out <= 32'h00000000;
        response_sequence_tag_out <= 16'h0000;
    end else begin
        response_valid_out <= 1'b0;
        response_stale_out <= 1'b0;
        malformed_response_out <= 1'b0;
        sequence_error_out <= 1'b0;
        if (rsp_valid_in) begin
            response_sequence_tag_out <= tag_w;
            if (timeout_expired_in) begin
                response_stale_out <= 1'b1;
            end else if (!busy_in) begin
                response_stale_out <= 1'b1;
            end else if (!rsp_last_in || !frame_ok_w || !len_ok_w) begin
                malformed_response_out <= 1'b1;
            end else if (!tag_ok_w) begin
                sequence_error_out <= 1'b1;
            end else if (parity_w != calc_parity_w) begin
                malformed_response_out <= 1'b1;
            end else begin
                response_valid_out <= 1'b1;
                validated_command_out <= command_w;
            end
        end
    end
end

endmodule

---BEGIN actuator_command_limiter.v---
module actuator_command_limiter (
    input clk,
    input rst_n,
    input [31:0] command_in,
    input limit_enable_in,
    input [15:0] rate_limit_step_in,
    input [31:0] command_min_in,
    input [31:0] command_max_in,
    input apply_in,
    output reg [31:0] command_out,
    output reg command_clamped_out
);

reg [31:0] last_command;

wire [31:0] clamp_min_w;
wire [31:0] clamp_max_w;
wire [31:0] bounded_hi_w;
wire [31:0] bounded_lo_w;
wire [31:0] rate_step_w;
wire [31:0] delta_up_w;
wire [31:0] delta_dn_w;
wire clamp_need_w;
wire [31:0] next_command_w;

assign clamp_min_w = command_min_in;
assign clamp_max_w = command_max_in;
assign bounded_hi_w = (command_in > clamp_max_w) ? clamp_max_w : command_in;
assign bounded_lo_w = (bounded_hi_w < clamp_min_w) ? clamp_min_w : bounded_hi_w;
assign rate_step_w = {16'h0000, rate_limit_step_in};
assign delta_up_w = (bounded_lo_w > last_command) ? (bounded_lo_w - last_command) : 32'h00000000;
assign delta_dn_w = (last_command > bounded_lo_w) ? (last_command - bounded_lo_w) : 32'h00000000;
assign clamp_need_w = (command_in != bounded_lo_w) || ((limit_enable_in && (rate_limit_step_in != 16'h0000) && ((delta_up_w > rate_step_w) || (delta_dn_w > rate_step_w))));
assign next_command_w = (limit_enable_in && (rate_limit_step_in != 16'h0000) && (delta_up_w > rate_step_w)) ? (last_command + rate_step_w) :
                        (limit_enable_in && (rate_limit_step_in != 16'h0000) && (delta_dn_w > rate_step_w)) ? (last_command - rate_step_w) :
                        bounded_lo_w;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        command_out <= 32'h00000000;
        command_clamped_out <= 1'b0;
        last_command <= 32'h00000000;
    end else begin
        command_clamped_out <= 1'b0;
        if (apply_in) begin
            command_out <= next_command_w;
            command_clamped_out <= (command_in != next_command_w) || clamp_need_w;
            last_command <= next_command_w;
        end
    end
end

endmodule
---END actuator_command_limiter.v---
---BEGIN stale_data_monitor.v---
module stale_data_monitor (
    input clk,
    input rst_n,
    input start_request_in,
    input abort_in,
    input [15:0] sequence_tag_in,
    input [15:0] timeout_threshold_in,
    input response_accepted_in,
    input request_busy_in,
    output reg timeout_expired_out,
    output reg stale_response_out,
    output reg [15:0] current_sequence_tag_out,
    output reg busy_out
);

reg [15:0] age_counter;
reg [15:0] threshold_reg;
reg active_reg;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        timeout_expired_out <= 1'b0;
        stale_response_out <= 1'b0;
        current_sequence_tag_out <= 16'h0000;
        busy_out <= 1'b0;
        age_counter <= 16'h0000;
        threshold_reg <= 16'h03E8;
        active_reg <= 1'b0;
    end else begin
        timeout_expired_out <= 1'b0;
        stale_response_out <= 1'b0;
        if (abort_in) begin
            busy_out <= 1'b0;
            age_counter <= 16'h0000;
            current_sequence_tag_out <= 16'h0000;
            active_reg <= 1'b0;
        end else if (start_request_in && !busy_out && !request_busy_in) begin
            busy_out <= 1'b1;
            current_sequence_tag_out <= sequence_tag_in;
            age_counter <= 16'h0000;
            threshold_reg <= timeout_threshold_in;
            active_reg <= 1'b1;
        end else if (busy_out) begin
            if (response_accepted_in) begin
                busy_out <= 1'b0;
                age_counter <= 16'h0000;
                active_reg <= 1'b0;
            end else begin
                age_counter <= age_counter + 16'h0001;
                if (age_counter >= threshold_reg) begin
                    timeout_expired_out <= 1'b1;
                    stale_response_out <= 1'b1;
                    busy_out <= 1'b0;
                    active_reg <= 1'b0;
                end
            end
        end
        if (request_busy_in) begin
            busy_out <= 1'b1;
        end
    end
end

endmodule

module mmio_register_block (
    input clk,
    input rst_n,
    input [7:0] mmio_addr,
    input [63:0] mmio_wdata,
    input [7:0] mmio_wstrb,
    input mmio_valid,
    input mmio_write,
    output reg [63:0] mmio_rdata,
    output reg mmio_ready,
    output reg control_start,
    output reg control_abort,
    output reg control_enable,
    output reg control_clear_faults,
    output reg [15:0] timeout_threshold_cfg,
    output reg [31:0] actuator_min_cfg,
    output reg [31:0] actuator_max_cfg,
    output reg [15:0] rate_limit_cfg,
    output reg [15:0] sequence_seed_cfg,
    output reg [7:0] safe_state_cfg,
    input [15:0] last_request_tag_in,
    input [15:0] last_response_tag_in,
    input [7:0] fault_code_in,
    input busy_in,
    input response_valid_in,
    input response_stale_in,
    input timeout_in,
    input malformed_response_in,
    input sequence_error_in,
    input fallback_active_in,
    input command_clamped_in,
    input [15:0] accepted_response_count_in,
    input [15:0] rejected_response_count_in,
    input [15:0] timeout_count_in,
    input [15:0] fallback_entry_count_in,
    input [15:0] clamp_event_count_in
);

reg [63:0] control_reg;
reg [63:0] status_reg;
reg [63:0] fault_reg;
reg [63:0] sequence_reg;
reg [63:0] timeout_reg;
reg [63:0] amin_reg;
reg [63:0] amax_reg;
reg [63:0] rate_reg;
reg [63:0] tags_reg;
reg [63:0] safe_reg;
reg [63:0] ctr0_reg;
reg [63:0] ctr1_reg;

wire [63:0] read_mux_w;
assign read_mux_w = (mmio_addr == 8'h00) ? control_reg :
                    (mmio_addr == 8'h08) ? status_reg :
                    (mmio_addr == 8'h10) ? fault_reg :
                    (mmio_addr == 8'h18) ? sequence_reg :
                    (mmio_addr == 8'h20) ? timeout_reg :
                    (mmio_addr == 8'h28) ? amin_reg :
                    (mmio_addr == 8'h30) ? amax_reg :
                    (mmio_addr == 8'h38) ? rate_reg :
                    (mmio_addr == 8'h40) ? tags_reg :
                    (mmio_addr == 8'h48) ? safe_reg :
                    (mmio_addr == 8'h50) ? ctr0_reg :
                    (mmio_addr == 8'h58) ? ctr1_reg : 64'h0;

integer i;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        control_reg <= 64'h0;
        status_reg <= 64'h0;
        fault_reg <= 64'h0;
        sequence_reg <= 64'h0;
        timeout_reg <= 64'h00000000000003E8;
        amin_reg <= 64'h0;
        amax_reg <= 64'h000000000000FFFF;
        rate_reg <= 64'h0;
        tags_reg <= 64'h0;
        safe_reg <= 64'h0;
        ctr0_reg <= 64'h0;
        ctr1_reg <= 64'h0;
        mmio_rdata <= 64'h0;
        mmio_ready <= 1'b1;
        control_start <= 1'b0;
        control_abort <= 1'b0;
        control_enable <= 1'b0;
        control_clear_faults <= 1'b0;
        timeout_threshold_cfg <= 16'd1000;
        actuator_min_cfg <= 32'h00000000;
        actuator_max_cfg <= 32'h0000FFFF;
        rate_limit_cfg <= 16'h0000;
        sequence_seed_cfg <= 16'h0000;
        safe_state_cfg <= 8'h00;
    end else begin
        control_start <= 1'b0;
        control_abort <= 1'b0;
        control_clear_faults <= 1'b0;
        mmio_ready <= mmio_valid;
        if (mmio_valid && mmio_write) begin
            case (mmio_addr)
                8'h00: begin
                    for (i = 0; i < 8; i = i + 1) begin
                        if (mmio_wstrb[i]) control_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    control_start <= mmio_wdata[0];
                    control_abort <= mmio_wdata[1];
                    control_enable <= mmio_wdata[2];
                    control_clear_faults <= mmio_wdata[3];
                end
                8'h18: begin
                    for (i = 0; i < 2; i = i + 1) begin
                        if (mmio_wstrb[i]) sequence_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    sequence_seed_cfg <= mmio_wdata[15:0];
                end
                8'h20: begin
                    for (i = 0; i < 2; i = i + 1) begin
                        if (mmio_wstrb[i]) timeout_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    timeout_threshold_cfg <= mmio_wdata[15:0];
                end
                8'h28: begin
                    for (i = 0; i < 4; i = i + 1) begin
                        if (mmio_wstrb[i]) amin_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    actuator_min_cfg <= mmio_wdata[31:0];
                end
                8'h30: begin
                    for (i = 0; i < 4; i = i + 1) begin
                        if (mmio_wstrb[i]) amax_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    actuator_max_cfg <= mmio_wdata[31:0];
                end
                8'h38: begin
                    for (i = 0; i < 2; i = i + 1) begin
                        if (mmio_wstrb[i]) rate_reg[i*8 +: 8] <= mmio_wdata[i*8 +: 8];
                    end
                    rate_limit_cfg <= mmio_wdata[15:0];
                end
                8'h48: begin
                    if (mmio_wstrb[0]) safe_reg[7:0] <= mmio_wdata[7:0];
                    safe_state_cfg <= mmio_wdata[7:0];
                end
                default: begin
                end
            endcase
        end
        if (control_clear_faults) begin
            fault_reg <= 64'h0;
        end
        status_reg <= {56'h0, fallback_active_in, command_clamped_in, sequence_error_in, malformed_response_in, timeout_in, response_stale_in, response_valid_in, busy_in};
        fault_reg[7:0] <= fault_code_in;
        tags_reg <= {32'h0, last_response_tag_in, last_request_tag_in};
        ctr0_reg <= {32'h0, rejected_response_count_in, accepted_response_count_in};
        ctr1_reg <= {16'h0, clamp_event_count_in, fallback_entry_count_in, timeout_count_in};
        mmio_rdata <= read_mux_w;
    end
end

endmodule
